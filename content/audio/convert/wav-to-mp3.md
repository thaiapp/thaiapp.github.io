  var RATES_V1 = [32, 40, 48, 56, 64, 80, 96, 112, 128, 160, 192, 224, 256, 320];

  var SUPPORTED_SR = [8000, 11025, 12000, 16000, 22050, 24000, 32000, 44100, 48000];

  function fitBitrate(kbps, sr) {
    var list = sr >= 32000 ? RATES_V1 : RATES_V2;
    var best = list[0];
    for (var i = 0; i < list.length; i++) { if (list[i] <= kbps) { best = list[i]; } }
    return best;
  }

  function pickSampleRate(setting, info) {
    if (setting !== 'auto') { return parseInt(setting, 10) || 44100; }
    if (info && SUPPORTED_SR.indexOf(info.sampleRate) >= 0) { return info.sampleRate; }
    return 44100;
  }

  function chooseBitrate(cfg, dur, sr) {
    var kbps = cfg.bitrate;
    if (cfg.targetMB > 0 && dur > 0) {
      kbps = Math.min(cfg.bitrate, cfg.targetMB * 1048576 * 8 / dur / 1000);
    }
    return fitBitrate(kbps, sr);
  }

  function planFor(totalDur, info, cfg) {
    var sr = pickSampleRate(cfg.rate, info);
    var dur = Math.max(0, totalDur - cfg.trimStart - cfg.trimEnd);
    var kbps = chooseBitrate(cfg, dur, sr);
    return { sr: sr, dur: dur, kbps: kbps, bytes: Math.round(dur * kbps * 125) };
  }

  function formatSize(b) {
    if (b < 1024) { return b + ' B'; }
    if (b < 1048576) { return (b / 1024).toFixed(0) + ' KB'; }
    if (b < 1073741824) { return (b / 1048576).toFixed(1) + ' MB'; }
    return (b / 1073741824).toFixed(2) + ' GB';
  }

  function formatTime(sec) {
    sec = Math.round(sec);
    var h = Math.floor(sec / 3600);
    var m = Math.floor((sec % 3600) / 60);
    var s = sec % 60;
    var ss = (s < 10 ? '0' : '') + s;
    if (h > 0) { return h + ':' + (m < 10 ? '0' : '') + m + ':' + ss; }
    return m + ':' + ss;
  }

  function baseName(name) { return name.replace(/\.[^.]+$/, ''); }

  function makeName(fileName, pattern, kbps, n) {
    var vars = { name: baseName(fileName), bitrate: String(kbps), n: String(n) };
    var out = pattern.replace(/\{(name|bitrate|n)\}/g, function (m, k) { return vars[k]; });
    out = out.replace(/[\\\/:*?"\x3C\x3E|\u0000-\u001f]/g, '').replace(/^[\s.]+|[\s.]+$/g, '');
    if (!out) { out = 'audio'; }
    if (out.length > 150) { out = out.slice(0, 150); }
    return out + '.mp3';
  }

  function readWavInfo(buf, fileSize) {
    try {
      var v = new DataView(buf);
      if (v.byteLength < 12 || v.getUint32(0, false) !== 0x52494646 || v.getUint32(8, false) !== 0x57415645) { return null; }
      var p = 12;
      var info = {};
      var total = fileSize || v.byteLength;
      while (p + 8 <= v.byteLength) {
        var id = v.getUint32(p, false);
        var size = v.getUint32(p + 4, true);
        if (id === 0x666d7420 && p + 24 <= v.byteLength) {
          info.channels = v.getUint16(p + 10, true);
          info.sampleRate = v.getUint32(p + 12, true);
          info.byteRate = v.getUint32(p + 16, true);
        }
        if (id === 0x64617461) {
          info.dataSize = Math.min(size, total - (p + 8));
          break;
        }
        p += 8 + size + (size & 1);
      }
      return info.sampleRate ? info : null;
    } catch (e) {
      return null;
    }
  }

  function mixBlock(data, chIn, chOut, from, n, L, R) {
    var i, c, s;
    if (chOut === 1) {
      if (chIn === 1) {
        var d0 = data[0];
        for (i = 0; i < n; i++) { L[i] = d0[from + i]; }
      } else {
        for (i = 0; i < n; i++) {
          s = 0;
          for (c = 0; c < chIn; c++) { s += data[c][from + i]; }
          L[i] = s / chIn;
        }
      }
    } else {
      var dl = data[0];
      var dr = chIn > 1 ? data[1] : data[0];
      for (i = 0; i < n; i++) { L[i] = dl[from + i]; R[i] = dr[from + i]; }
    }
  }

  var crcTable = null;
  function crc32(u8) {
    if (!crcTable) {
      crcTable = new Uint32Array(256);
      for (var n = 0; n < 256; n++) {
        var c = n;
        for (var k = 0; k < 8; k++) { c = (c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1); }
        crcTable[n] = c >>> 0;
      }
    }
    var crc = 0xFFFFFFFF;
    for (var i = 0; i < u8.length; i++) { crc = crcTable[(crc ^ u8[i]) & 0xFF] ^ (crc >>> 8); }
    return (crc ^ 0xFFFFFFFF) >>> 0;
  }

  /* ZIP "store" (no compression) - MP3 is already compressed. files: [{name, data: Uint8Array}] */
  function buildZip(files) {
    var enc = new TextEncoder();
    var parts = [];
    var central = [];
    var offset = 0;
    var d = new Date();
    var dosTime = ((d.getHours() << 11) | (d.getMinutes() << 5) | (d.getSeconds() >> 1)) & 0xFFFF;
    var dosDate = (((d.getFullYear() - 1980) << 9) | ((d.getMonth() + 1) << 5) | d.getDate()) & 0xFFFF;
    var cdSize = 0;
    files.forEach(function (f) {
      var nameBytes = enc.encode(f.name);
      var crc = crc32(f.data);
      var size = f.data.length;
      var lh = new DataView(new ArrayBuffer(30));
      lh.setUint32(0, 0x04034b50, true);
      lh.setUint16(4, 20, true);
      lh.setUint16(6, 0x0800, true);
      lh.setUint16(8, 0, true);
      lh.setUint16(10, dosTime, true);
      lh.setUint16(12, dosDate, true);
      lh.setUint32(14, crc, true);
      lh.setUint32(18, size, true);
      lh.setUint32(22, size, true);
      lh.setUint16(26, nameBytes.length, true);
      lh.setUint16(28, 0, true);
      parts.push(lh.buffer, nameBytes, f.data);
      var ch = new DataView(new ArrayBuffer(46));
      ch.setUint32(0, 0x02014b50, true);
      ch.setUint16(4, 20, true);
      ch.setUint16(6, 20, true);
      ch.setUint16(8, 0x0800, true);
      ch.setUint16(10, 0, true);
      ch.setUint16(12, dosTime, true);
      ch.setUint16(14, dosDate, true);
      ch.setUint32(16, crc, true);
      ch.setUint32(20, size, true);
      ch.setUint32(24, size, true);
      ch.setUint16(28, nameBytes.length, true);
      ch.setUint32(42, offset, true);
      central.push(ch.buffer, nameBytes);
      cdSize += 46 + nameBytes.length;
      offset += 30 + nameBytes.length + size;
    });
    var end = new DataView(new ArrayBuffer(22));
    end.setUint32(0, 0x06054b50, true);
    end.setUint16(8, files.length, true);
    end.setUint16(10, files.length, true);
    end.setUint32(12, cdSize, true);
    end.setUint32(16, offset, true);
    return new Blob(parts.concat(central, [end.buffer]), { type: 'application/zip' });
  }

  /* ID3v2.3 tag: TIT2 / TPE1 / TALB (UTF-16 with BOM) + APIC (cover) */
  function id3Frame(id, bytes) {
    var f = new Uint8Array(10 + bytes.length);
    for (var i = 0; i < 4; i++) { f[i] = id.charCodeAt(i); }
    var n = bytes.length;
    f[4] = (n >>> 24) & 0xFF; f[5] = (n >>> 16) & 0xFF; f[6] = (n >>> 8) & 0xFF; f[7] = n & 0xFF;
    f.set(bytes, 10);
    return f;
  }

  function id3Text(id, text) {
    var data = new Uint8Array(3 + text.length * 2);
    data[0] = 1; data[1] = 0xFF; data[2] = 0xFE;
    for (var i = 0; i < text.length; i++) {
      var c = text.charCodeAt(i);
      data[3 + i * 2] = c & 0xFF;
      data[4 + i * 2] = c >> 8;
    }
    return id3Frame(id, data);
  }

  function id3Picture(mime, bytes) {
    var head = [0];
    for (var i = 0; i < mime.length; i++) { head.push(mime.charCodeAt(i) & 0x7F); }
    head.push(0, 3, 0);
    var data = new Uint8Array(head.length + bytes.length);
    data.set(head, 0);
    data.set(bytes, head.length);
    return id3Frame('APIC', data);
  }

  function buildId3(tags) {
    var frames = [];
    if (tags.title) { frames.push(id3Text('TIT2', tags.title)); }
    if (tags.artist) { frames.push(id3Text('TPE1', tags.artist)); }
    if (tags.album) { frames.push(id3Text('TALB', tags.album)); }
    if (tags.cover) { frames.push(id3Picture(tags.cover.mime, tags.cover.bytes)); }
    if (!frames.length) { return null; }
    var total = 0;
    frames.forEach(function (f) { total += f.length; });
    var out = new Uint8Array(10 + total);
    out.set([0x49, 0x44, 0x33, 3, 0, 0], 0);
    out.set([(total >> 21) & 0x7F, (total >> 14) & 0x7F, (total >> 7) & 0x7F, total & 0x7F], 6);
    var p = 10;
    frames.forEach(function (f) { out.set(f, p); p += f.length; });
    return out;
  }

  function loadLame() {
    if (window.lamejs && window.lamejs.Mp3Encoder) { return Promise.resolve(); }
    if (lamePromise) { return lamePromise; }
    lamePromise = new Promise(function (resolve, reject) {
      var i = 0;
      function next() {
        if (i >= LAME_SOURCES.length) { lamePromise = null; reject(new Error('lame')); return; }
        var s = document.createElement('script');
        s.async = true;
        s.src = LAME_SOURCES[i++];
        s.onload = function () {
          if (window.lamejs && window.lamejs.Mp3Encoder) { resolve(); } else { s.remove(); next(); }
        };
        s.onerror = function () { s.remove(); next(); };
        document.head.appendChild(s);
      }
      next();
    });
    return lamePromise;
  }

  /* ---------- audio decode / resample ---------- */
  function decodeAudio(arrayBuffer, sr) {
    var Ctx = window.AudioContext || window.webkitAudioContext;
    if (!Ctx) { return Promise.reject(new Error('no-audio')); }
    var ctx;
    try { ctx = new Ctx({ sampleRate: sr }); } catch (e) { ctx = new Ctx(); }
    function close() { try { ctx.close(); } catch (e2) { /* ignore */ } }
    return new Promise(function (resolve, reject) {
      var p = ctx.decodeAudioData(arrayBuffer, resolve, reject);
      if (p && p.catch) { p.catch(reject); }
    }).then(function (buf) {
      close();
      return buf;
    }, function () {
      close();
      throw new Error('decode');
    });
  }

  function resample(buf, sr) {
    var Off = window.OfflineAudioContext || window.webkitOfflineAudioContext;
    var len = Math.max(1, Math.ceil(buf.duration * sr));
    var oc = new Off(buf.numberOfChannels, len, sr);
    var src = oc.createBufferSource();
    src.buffer = buf;
    src.connect(oc.destination);
    src.start(0);
    var p = oc.startRendering();
    if (p && p.then) { return p; }
    return new Promise(function (resolve) { oc.oncomplete = function (e) { resolve(e.renderedBuffer); }; });
  }

      a.onerror = function () { fin(0); };
      setTimeout(function () { fin(0); }, 8000);
      a.src = url;
    });
  }

  async function encodeMp3(it, buf, cfg, sr, cover) {
    var chIn = buf.numberOfChannels;
    var chOut = cfg.channels === 'mono' ? 1 : (cfg.channels === 'stereo' ? 2 : (chIn >= 2 ? 2 : 1));
    var total = buf.length;
    var startS = Math.min(total, Math.round(cfg.trimStart * sr));
    var endS = Math.max(startS, total - Math.round(cfg.trimEnd * sr));
    var len = endS - startS;
    if (len < sr / 20) { throw new Error('empty'); }
    var kbps = chooseBitrate(cfg, len / sr, sr);
    var data = [];
    var c, i, pos, n, v, g, k;
    for (c = 0; c < chIn; c++) { data.push(buf.getChannelData(c)); }
    var BLOCK = 1152 * 8;
    var Lf = new Float32Array(BLOCK);
    var Rf = chOut === 2 ? new Float32Array(BLOCK) : null;
    var Li = new Int16Array(BLOCK);
    var Ri = chOut === 2 ? new Int16Array(BLOCK) : null;
    var lastYield = now();
    var gain = 1;
    var base = cfg.normalize ? 0.1 : 0;

    if (cfg.normalize) {
      var peak = 0;
      for (pos = 0; pos < len; pos += BLOCK) {
        n = Math.min(BLOCK, len - pos);
        mixBlock(data, chIn, chOut, startS + pos, n, Lf, Rf);
        for (i = 0; i < n; i++) {
          v = Lf[i] < 0 ? -Lf[i] : Lf[i];
          if (v > peak) { peak = v; }
          if (Rf) {
            v = Rf[i] < 0 ? -Rf[i] : Rf[i];
            if (v > peak) { peak = v; }
          }
        }
        if (now() - lastYield > 50) {
          checkStop();
          setProgress(it, base * pos / len, 'กำลังวิเคราะห์ระดับเสียง…');
          await yieldUI();
          lastYield = now();
        }
      }
      if (peak > 0.0001) { gain = Math.min(20, 0.891 / peak); }
    }

    var fi = Math.round(cfg.fadeIn * sr);
    var fo = Math.round(cfg.fadeOut * sr);
    if (fi + fo > len) {
      var sc = len / (fi + fo);
      fi = Math.floor(fi * sc);
      fo = Math.floor(fo * sc);
    }

    var enc = new window.lamejs.Mp3Encoder(chOut, sr, kbps);
    var chunks = [];
    for (pos = 0; pos < len; pos += BLOCK) {
      n = Math.min(BLOCK, len - pos);
      mixBlock(data, chIn, chOut, startS + pos, n, Lf, Rf);
      for (i = 0; i < n; i++) {
        k = pos + i;
        g = gain;
        if (k < fi) { g *= k / fi; }
        if (k >= len - fo) { g *= (len - k) / fo; }
        v = Lf[i] * g;
        if (v > 1) { v = 1; } else if (v < -1) { v = -1; }
        Li[i] = v < 0 ? v * 32768 : v * 32767;
        if (Rf) {
          v = Rf[i] * g;
          if (v > 1) { v = 1; } else if (v < -1) { v = -1; }
          Ri[i] = v < 0 ? v * 32768 : v * 32767;
        }
      }
      var out = Rf ? enc.encodeBuffer(Li.subarray(0, n), Ri.subarray(0, n)) : enc.encodeBuffer(Li.subarray(0, n));
      if (out.length > 0) { chunks.push(new Uint8Array(out)); }
      if (now() - lastYield > 50) {
        checkStop();
        setProgress(it, base + (1 - base) * pos / len, 'กำลังแปลง… ' + Math.round((base + (1 - base) * pos / len) * 100) + '%');
        await yieldUI();
        lastYield = now();
      }
    }
    var last = enc.flush();
    if (last.length > 0) { chunks.push(new Uint8Array(last)); }

    var tags = null;
    if (cfg.title || cfg.artist || cfg.album || cover) {
      tags = { title: cfg.title || baseName(it.file.name), artist: cfg.artist, album: cfg.album, cover: cover };
    }
    var id3 = tags ? buildId3(tags) : null;
    var parts = id3 ? [id3].concat(chunks) : chunks;
    return { blob: new Blob(parts, { type: 'audio/mpeg' }), kbps: kbps };
  }

  async function convertItem(it, cfg, cover) {
    it.state = 'working';
    it.li.classList.remove('is-error');
    setProgress(it, 0, 'กำลังอ่านไฟล์…');
    refreshAll();
    var ab = await readBuffer(it.file);
    var info = readWavInfo(ab, it.file.size) || it.info;
    var sr = pickSampleRate(cfg.rate, info);
    setProgress(it, 0, 'กำลังถอดรหัสเสียง…');
    await yieldUI();
    checkStop();
    var buf = await decodeAudio(ab, sr);
    ab = null;
    if (buf.sampleRate !== sr) { buf = await resample(buf, sr); }
    checkStop();
    var res = await encodeMp3(it, buf, cfg, sr, cover);
    showResult(it, res, cfg);
  }

    if (it.outUrl) { URL.revokeObjectURL(it.outUrl); }

    it.outName = makeName(it.file.name, cfg.pattern, res.kbps, items.indexOf(it) + 1);
    it.state = 'done';

    it.dl.href = it.outUrl;
    it.dl.download = it.outName;
    var saved = it.file.size > 0 ? Math.round((1 - res.blob.size / it.file.size) * 100) : 0;
    var txt = it.outName + ' · ' + formatSize(res.blob.size) + ' · ' + res.kbps + ' kbps';
    if (saved > 0) { txt += ' · เล็กลง ' + saved + '%'; }

    it.resText.textContent = txt;
    it.resultBox.hidden = false;
    setProgress(it, 1, 'เสร็จแล้ว');
  }

    it.li.classList.add('is-error');

  function removeItem(it) {
    if (running) { say('ไม่สามารถลบไฟล์ระหว่างแปลง — กด "หยุด" ก่อน', 'err'); return; }
    URL.revokeObjectURL(it.url);

      URL.revokeObjectURL(it.url);
      if (it.outUrl) { URL.revokeObjectURL(it.outUrl); }
    });

    els.list.textContent = '';

  }

  function cfgKey(cfg) { return JSON.stringify(cfg); }

  function refreshAll() {
    var cfg = readSettings();
    var key = cfgKey(cfg);
    var totalBytes = 0, estBytes = 0, estCount = 0, done = 0, doneBytes = 0, doneOrig = 0;
    items.forEach(function (it) {
      totalBytes += it.file.size;
      var txt = formatSize(it.file.size);
      if (it.dur) { txt += ' · ' + formatTime(it.dur); }
      if (it.state === 'done' && it.blob) {
        done++;
        doneBytes += it.blob.size;
        doneOrig += it.file.size;
        if (it.cfgKey !== key) { it.statusEl.textContent = 'แปลงแล้ว (ตั้งค่าเปลี่ยนไป — กดแปลงอีกครั้งเพื่อใช้ค่าล่าสุด)'; }
        else { it.statusEl.textContent = 'เสร็จแล้ว'; }
      } else if (it.dur) {
        var p = planFor(it.dur, it.info, cfg);
        estBytes += p.bytes;
        estCount++;
        txt += ' · MP3 ≈ ' + formatSize(p.bytes) + ' (' + p.kbps + ' kbps)';
      }
      it.metaEl.textContent = txt;
    });
    var s = '';
    if (items.length) {
      s = items.length + ' ไฟล์ · ต้นฉบับรวม ' + formatSize(totalBytes);
      if (estCount) { s += ' · MP3 ที่ยังไม่แปลงประมาณ ' + formatSize(estBytes); }
      if (done) {
        var pct = doneOrig > 0 ? Math.round((1 - doneBytes / doneOrig) * 100) : 0;
        s += ' · แปลงแล้ว ' + done + ' ไฟล์ รวม ' + formatSize(doneBytes) + (pct > 0 ? ' (เล็กลง ' + pct + '%)' : '');
      }
    }
    els.summary.textContent = s;
    els.go.disabled = running || !items.length;
    els.stop.hidden = !running;
    els.zip.disabled = running || done === 0;
    els.clear.disabled = running || !items.length;
    els.pick.disabled = running;
    els.pickdir.disabled = running;
  }

  async function probe(it) {
    try {
      var head = await readBuffer(it.file.slice(0, 65536));
      it.info = readWavInfo(head, it.file.size);
    } catch (e) { /* ignore */ }
    it.dur = await probeDuration(it.url);
    if (!it.dur && it.info && it.info.byteRate && it.info.dataSize) { it.dur = it.info.dataSize / it.info.byteRate; }
    refreshAll();
  }

  function isAudioFile(f) {
    return (f.type && f.type.indexOf('audio/') === 0) || /\.(wav|wave|mp3|m4a|aac|ogg|oga|opus|flac|wma|aif|aiff)$/i.test(f.name);
  }

  function addFiles(list) {
    var added = 0, skipped = 0, dup = 0;
    list.forEach(function (f) {
      if (!isAudioFile(f)) { skipped++; return; }
      var key = f.name + '|' + f.size + '|' + f.lastModified;
      for (var i = 0; i < items.length; i++) { if (items[i].key === key) { dup++; return; } }
      var it = createItem(f, key);
      items.push(it);
      els.list.appendChild(it.li);
      added++;
      probe(it);
    });
    var notes = [];
    if (skipped) { notes.push('ข้ามไฟล์ที่ไม่ใช่เสียง ' + skipped + ' ไฟล์'); }
    if (dup) { notes.push('ข้ามไฟล์ซ้ำ ' + dup + ' ไฟล์'); }
    if (added) { notes.unshift('เพิ่ม ' + added + ' ไฟล์'); loadLame().catch(function () { /* retried on convert */ }); }
    if (notes.length) { say(notes.join(' · '), skipped && !added ? 'err' : ''); }
    refreshAll();
  }

  /* ---------- drop / paste / pick ---------- */
  function walkEntry(entry) {
    if (entry.isFile) {
      return new Promise(function (resolve) { entry.file(function (f) { resolve([f]); }, function () { resolve([]); }); });
    }
    if (entry.isDirectory) {
      var reader = entry.createReader();
      var all = [];
      return new Promise(function (resolve) {
        (function readMore() {
          reader.readEntries(function (ents) {
            if (!ents.length) {
              Promise.all(all.map(walkEntry)).then(function (lists) { resolve([].concat.apply([], lists)); });
              return;
            }
            all = all.concat(Array.prototype.slice.call(ents));
            readMore();
          }, function () { resolve([]); });
        })();
      });
    }
    return Promise.resolve([]);
  }

  function collectDropped(dt) {
    var files = [];
    var entries = [];
    if (dt.items && dt.items.length) {
      for (var i = 0; i < dt.items.length; i++) {
        var item = dt.items[i];
        if (item.kind !== 'file') { continue; }
        var en = item.webkitGetAsEntry ? item.webkitGetAsEntry() : null;
        if (en) { entries.push(en); }
        else { var f = item.getAsFile(); if (f) { files.push(f); } }
      }
    } else {
      files = Array.prototype.slice.call(dt.files || []);
    }
    return Promise.all(entries.map(walkEntry)).then(function (lists) {
      lists.forEach(function (l) { files = files.concat(l); });
      return files;
    });
  }

  async function convertAll() {
    if (running) { return; }
    var cover = null;
    try { cover = await readCover(); }
    catch (e) {
      say(e.message === 'cover-big' ? 'ไฟล์ภาพปกใหญ่เกิน 3 MB' : 'ภาพปกต้องเป็นไฟล์ JPG หรือ PNG', 'err');
      return;
    }
    var cfg = readSettings();
    var key = cfgKey(cfg);
    var pending = items.filter(function (it) { return it.state !== 'done' || it.cfgKey !== key; });
    if (!pending.length) { say(items.length ? 'ทุกไฟล์แปลงตามการตั้งค่าล่าสุดแล้ว' : 'เพิ่มไฟล์ WAV ก่อนเริ่มแปลง', ''); return; }
    running = true;
    stopFlag = false;
    refreshAll();
    say('กำลังโหลดตัวเข้ารหัส MP3 (ครั้งแรกเท่านั้น)…', '');
    try { await loadLame(); }
    catch (e2) {
      running = false;
      say('โหลดตัวเข้ารหัส MP3 ไม่สำเร็จ — ตรวจสอบอินเทอร์เน็ตหรือการบล็อกสคริปต์จากภายนอกแล้วลองใหม่', 'err');
      refreshAll();
      return;
    }
    say('กำลังแปลง… โปรดเปิดแท็บนี้ค้างไว้จนเสร็จ', '');
    var ok = 0, fail = 0;
    for (var i = 0; i < pending.length; i++) {
      var it = pending[i];
      if (stopFlag) { break; }
      try {
        await convertItem(it, cfg, cover);
        it.cfgKey = key;
        ok++;
      } catch (e3) {
        if (e3 && e3.message === 'stopped') {

          break;
        }
        fail++;
        showError(it, e3);
      }
    }
    var stopped = stopFlag;
    running = false;
    stopFlag = false;
    say(stopped ? 'หยุดการแปลงแล้ว (สำเร็จ ' + ok + ' ไฟล์)' : 'เสร็จสิ้น: สำเร็จ ' + ok + ' ไฟล์' + (fail ? ' · ล้มเหลว ' + fail + ' ไฟล์' : ''), fail ? 'err' : 'ok');
    refreshAll();
  }

  async function downloadZip() {
    var done = items.filter(function (it) { return it.state === 'done' && it.blob; });
    if (!done.length) { return; }

    try {
      var used = {};
      var files = [];
      for (var i = 0; i < done.length; i++) {
        var nm = done[i].outName;
        var base = nm.replace(/\.mp3$/i, '');
        var cand = nm;
        var k = 2;
        while (used[cand.toLowerCase()]) { cand = base + ' (' + k + ').mp3'; k++; }
        used[cand.toLowerCase()] = true;
        files.push({ name: cand, data: new Uint8Array(await readBuffer(done[i].blob)) });
      }
      var zip = buildZip(files);
      var url = URL.createObjectURL(zip);
      var a = document.createElement('a');
      a.href = url;
      a.download = 'wav-to-mp3.zip';
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      setTimeout(function () { URL.revokeObjectURL(url); }, 60000);
      say('ดาวน์โหลด Zip แล้ว (' + files.length + ' ไฟล์)', 'ok');
    } catch (e) {
      say('สร้างไฟล์ Zip ไม่สำเร็จ — ลองดาวน์โหลดทีละไฟล์แทน', 'err');
    }
  }

## วิธีใช้งานเครื่องมือแปลง WAV เป็น MP3

1. **เพิ่มไฟล์เสียง** — ลากไฟล์ WAV มาวางในกรอบด้านบน, คลิกเพื่อเลือกไฟล์, กด `Ctrl+V` เพื่อวางไฟล์ที่คัดลอกไว้ หรือกด **เลือกโฟลเดอร์** เพื่อเพิ่มทั้งโฟลเดอร์ในครั้งเดียว
2. **เลือกพรีเซ็ตหรือตั้งค่าเอง** — เลือกพรีเซ็ตสำเร็จรูป (เพลง, พอดแคสต์, ข้อความเสียง, ไฟล์เล็กที่สุด, คุณภาพสูงสุด) หรือปรับ Bitrate, Sample rate และช่องเสียงด้วยตัวเอง
3. **ดูขนาดไฟล์โดยประมาณ** — เครื่องมือจะแสดงความยาวและขนาด MP3 โดยประมาณของแต่ละไฟล์ทันที หากต้องการจำกัดขนาด ให้กรอก **ขนาดสูงสุดต่อไฟล์ (MB)** ระบบจะลด Bitrate ให้พอดีโดยอัตโนมัติ
4. **ปรับแต่งเพิ่มเติม (ไม่บังคับ)** — เปิด *ตัวเลือกขั้นสูง* เพื่อตัดต้น/ท้ายไฟล์, ใส่ Fade, ปรับระดับเสียง (Normalize), ตั้งรูปแบบชื่อไฟล์ และฝังแท็ก ID3 พร้อมภาพปก
5. **กด "แปลงทั้งหมดเป็น MP3"** — ดูความคืบหน้าของแต่ละไฟล์ได้ทันที เปิดแท็บนี้ค้างไว้จนเสร็จ และกด **หยุด** ได้ทุกเมื่อ
6. **ฟังตัวอย่างและดาวน์โหลด** — เปิดฟังไฟล์ต้นฉบับเทียบกับ MP3 ที่ได้ แล้วดาวน์โหลดทีละไฟล์ หรือรวมทั้งหมดเป็นไฟล์ `.zip` ในคลิกเดียว

> เปลี่ยนการตั้งค่าหลังแปลงเสร็จแล้ว? กด **แปลงทั้งหมดเป็น MP3** อีกครั้ง ระบบจะแปลงใหม่เฉพาะไฟล์ที่ใช้ค่าเก่า

## อธิบายศัพท์ที่ใช้ในเครื่องมือ

- **WAV** — ไฟล์เสียงแบบไม่บีบอัด คุณภาพเต็ม แต่ขนาดใหญ่มาก (ราว 10 MB ต่อนาทีสำหรับคุณภาพระดับซีดี)
- **MP3** — ไฟล์เสียงแบบบีบอัดที่เล่นได้เกือบทุกอุปกรณ์ ขนาดเล็กลงหลายเท่า เหมาะกับการส่งต่อและฟังทั่วไป
- **Lossless / Lossy** — Lossless เก็บข้อมูลเสียงครบทุกส่วน (เช่น WAV, FLAC) ส่วน Lossy ตัดรายละเอียดที่หูได้ยินยากออกเพื่อลดขนาด (เช่น MP3)
- **Bitrate (kbps)** — ปริมาณข้อมูลเสียงต่อวินาที ยิ่งสูงยิ่งคมชัดแต่ไฟล์ยิ่งใหญ่
- **CBR / VBR** — CBR คือ Bitrate คงที่ตลอดไฟล์ (เครื่องมือนี้ใช้ CBR จึงคำนวณขนาดล่วงหน้าได้แม่นยำ) ส่วน VBR คือ Bitrate ที่เปลี่ยนตามความซับซ้อนของเสียง
- **Sample rate (Hz)** — จำนวนครั้งที่สุ่มเก็บเสียงต่อวินาที 44,100 Hz คือมาตรฐานซีดี, 48,000 Hz นิยมในงานวิดีโอ, 22,050 Hz พอสำหรับเสียงพูด
- **ช่องเสียง (Mono / Stereo)** — โมโนคือ 1 ช่อง เหมาะกับเสียงพูด ส่วนสเตอริโอคือ 2 ช่อง (ซ้าย-ขวา) เหมาะกับเพลง
- **พรีเซ็ต (Preset)** — ชุดค่าที่ตั้งไว้ให้พร้อมใช้ตามประเภทงาน เลือกแล้วปรับต่อเองได้
- **ขนาดสูงสุดต่อไฟล์** — ตั้งเพดานขนาดไฟล์ MP3 เช่น 10 MB สำหรับแนบอีเมล เครื่องมือจะเลือก Bitrate ที่สูงที่สุดที่ยังไม่เกินเพดาน (ไม่เกินค่า Bitrate ที่เลือก)
- **Normalize** — ปรับความดังของทั้งไฟล์ให้จุดที่ดังที่สุดอยู่ใกล้ระดับสูงสุดโดยไม่ทำให้เสียงแตก ช่วยให้ไฟล์เบาเกินไปฟังชัดขึ้น
- **Fade in / Fade out** — ค่อย ๆ เพิ่มเสียงตอนเริ่ม หรือค่อย ๆ ลดเสียงตอนจบ เพื่อไม่ให้เสียงตัดกระทันหัน
- **ตัดต้น/ท้ายไฟล์ (Trim)** — ตัดช่วงเงียบหรือส่วนที่ไม่ต้องการออกเป็นจำนวนวินาทีก่อนแปลง
- **ID3 tag** — ข้อมูลที่ฝังในไฟล์ MP3 เช่น ชื่อเพลง ศิลปิน อัลบั้ม เพื่อให้เครื่องเล่นแสดงผลได้ถูกต้อง
- **ภาพปก (Cover art)** — รูปหน้าปกที่ฝังในไฟล์ MP3 และแสดงในเครื่องเล่นเพลง
- **รูปแบบชื่อไฟล์** — กำหนดชื่อไฟล์ผลลัพธ์ด้วย `{name}` (ชื่อไฟล์เดิม), `{bitrate}` (ค่า kbps ที่ใช้) และ `{n}` (ลำดับไฟล์) เช่น `{name}_{bitrate}k` ได้ `เพลง_192k.mp3`
- **ไฟล์ Zip** — ไฟล์รวมที่บรรจุ MP3 ทุกไฟล์ไว้ด้วยกัน สะดวกเมื่อแปลงหลายไฟล์พร้อมกัน

## เลือก Bitrate อย่างไรดี

| Bitrate | ขนาดโดยประมาณต่อ 1 นาที | เหมาะกับ |
|---|---|---|
| 320 kbps | 2.3 MB | เก็บเพลงคุณภาพสูง ใกล้เคียงต้นฉบับ |
| 192 kbps | 1.4 MB | เพลงทั่วไป สมดุลระหว่างคุณภาพและขนาด |
| 128 kbps | 0.9 MB | ฟังบนมือถือหรือลำโพงทั่วไป |
| 96 kbps | 0.7 MB | พอดแคสต์ เสียงพูด |
| 64 kbps | 0.5 MB | ข้อความเสียง บันทึกการประชุม |
| 32 kbps | 0.2 MB | ไฟล์เล็กที่สุด เหมาะกับเสียงพูดที่ไม่ต้องการคุณภาพสูง |

เพื่อเปรียบเทียบ ไฟล์ WAV คุณภาพซีดี (44,100 Hz, 16 บิต, สเตอริโอ) มีขนาดราว 10 MB ต่อนาที ดังนั้นการแปลงเป็น MP3 ที่ 192 kbps ลดขนาดลงได้ประมาณ 7 เท่า

## คำถามที่พบบ่อย

### ไฟล์ WAV ของฉันถูกอัปโหลดไปยังเซิร์ฟเวอร์หรือไม่?

ไม่ การถอดรหัสและแปลงเป็น MP3 ทำในเบราว์เซอร์ของคุณทั้งหมด ไฟล์เสียงไม่ถูกส่งออกจากอุปกรณ์ มีเพียงสคริปต์ตัวเข้ารหัส MP3 ขนาดเล็กที่ดาวน์โหลดมาครั้งแรกเมื่อเริ่มแปลง

### แปลง WAV เป็น MP3 แล้วคุณภาพเสียงลดลงหรือไม่?

MP3 เป็นรูปแบบแบบ Lossy จึงมีรายละเอียดบางส่วนถูกตัดออก แต่ที่ 192–320 kbps ผู้ฟังทั่วไปแทบแยกไม่ออกจากต้นฉบับ หากต้องการเก็บคุณภาพเต็มให้เก็บไฟล์ WAV ไว้เป็นต้นฉบับ เพราะการแปลง MP3 กลับเป็น WAV จะไม่ได้รายละเอียดที่หายไปคืน

### ควรเลือก Bitrate เท่าไร?

สำหรับเพลงแนะนำ 192 kbps หรือ 320 kbps หากต้องการคุณภาพสูงสุด สำหรับพอดแคสต์หรือเสียงพูดใช้ 96 kbps แบบโมโนก็เพียงพอ และสำหรับข้อความเสียงใช้ 64 kbps หรือต่ำกว่าได้ หากต้องการให้ไฟล์ไม่เกินขนาดที่กำหนด ให้กรอกช่อง “ขนาดสูงสุดต่อไฟล์”

### แปลงไฟล์ใหญ่หรือไฟล์ยาวได้แค่ไหน?

ขึ้นอยู่กับหน่วยความจำของอุปกรณ์ เสียงสเตอริโอ 44,100 Hz ยาว 10 นาทีใช้หน่วยความจำราว 200 MB ระหว่างประมวลผล ไฟล์ที่ยาวเกิน 1 ชั่วโมงควรแบ่งไฟล์ก่อน หรือใช้ตัวเลือกตัดต้น/ท้ายไฟล์ หากขึ้นข้อความว่าหน่วยความจำไม่พอ ให้ลองแปลงทีละไฟล์

### ใช้บนมือถือได้หรือไม่?

ได้ บนเบราว์เซอร์รุ่นใหม่ เช่น Chrome, Safari, Edge และ Firefox แต่ควรเปิดหน้านี้ค้างไว้และไม่ล็อกหน้าจอระหว่างแปลง เพราะระบบอาจหยุดการทำงานของแท็บที่อยู่เบื้องหลัง

### ทำไมแปลงแล้วขึ้นข้อผิดพลาด?

สาเหตุที่พบบ่อยคือไฟล์เสียหายหรือเบราว์เซอร์ไม่รองรับรูปแบบนั้น หน่วยความจำไม่พอสำหรับไฟล์ที่ยาวมาก หรือโหลดตัวเข้ารหัส MP3 ไม่สำเร็จเพราะอินเทอร์เน็ตหลุดหรือส่วนขยายบล็อกสคริปต์ ลองรีเฟรชหน้า ปิดตัวบล็อกโฆษณา แล้วกดแปลงอีกครั้ง
