  function pad(n) { return n < 10 ? '0' + n : String(n); }

  function fmtTime(s) {
    if (!isFinite(s) || s < 0) { return '--:--'; }
    s = Math.round(s);
    var h = Math.floor(s / 3600), m = Math.floor((s % 3600) / 60), x = s % 60;
    return h > 0 ? h + ':' + pad(m) + ':' + pad(x) : m + ':' + pad(x);
  }

  function fmtSize(b) {
    if (b < 1024) { return Math.round(b) + ' B'; }
    if (b < 1048576) { return (b / 1024).toFixed(1) + ' KB'; }
    return (b / 1048576).toFixed(2) + ' MB';
  }

  // คืนค่า null = เว่าง, NaN = รูปแบบผิด, ตัวเลข = วินาที
  function parseTime(v) {
    v = String(v == null ? '' : v).trim();
    if (!v) { return null; }
    var p = v.split(':');
    if (p.length > 3) { return NaN; }
    var t = 0;
    for (var i = 0; i < p.length; i++) {
      if (!/^\d+(\.\d+)?$/.test(p[i])) { return NaN; }
      t = t * 60 + parseFloat(p[i]);
    }
    return t;
  }

  function ascii(s) {
    var o = new Uint8Array(s.length);
    for (var i = 0; i < s.length; i++) { o[i] = s.charCodeAt(i) & 127; }
    return o;
  }

  function concatBytes(arrs) {
    var n = 0, i;
    for (i = 0; i < arrs.length; i++) { n += arrs[i].length; }
    var out = new Uint8Array(n), o = 0;
    for (i = 0; i < arrs.length; i++) { out.set(arrs[i], o); o += arrs[i].length; }
    return out;
  }

  var crcTable = (function () {
    var t = new Uint32Array(256);
    for (var n = 0; n < 256; n++) {
      var c = n;
      for (var k = 0; k < 8; k++) { c = (c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1); }
      t[n] = c >>> 0;
    }
    return t;
  })();

  function crc32(u8) {
    var c = 0xFFFFFFFF;
    for (var i = 0; i < u8.length; i++) { c = crcTable[(c ^ u8[i]) & 0xFF] ^ (c >>> 8); }
    return (c ^ 0xFFFFFFFF) >>> 0;
  }

  // สร้างไฟล์ ZIP แบบไม่บีบอัด (store) — MP3 บีบอัดต่อไม่ได้อยู่แล้ว
  function makeZip(files) {
    var enc = new TextEncoder(), parts = [], central = [], offset = 0, cdSize = 0;
    var d = new Date();
    var dosTime = (d.getHours() << 11) | (d.getMinutes() << 5) | (d.getSeconds() >> 1);
    var dosDate = ((d.getFullYear() - 1980) << 9) | ((d.getMonth() + 1) << 5) | d.getDate();
    files.forEach(function (f) {
      var nameB = enc.encode(f.name), crc = crc32(f.data), size = f.data.length;
      var lh = new DataView(new ArrayBuffer(30));
      lh.setUint32(0, 0x04034b50, true); lh.setUint16(4, 20, true); lh.setUint16(6, 0x0800, true);
      lh.setUint16(8, 0, true); lh.setUint16(10, dosTime, true); lh.setUint16(12, dosDate, true);
      lh.setUint32(14, crc, true); lh.setUint32(18, size, true); lh.setUint32(22, size, true);
      lh.setUint16(26, nameB.length, true); lh.setUint16(28, 0, true);
      parts.push(new Uint8Array(lh.buffer), nameB, f.data);
      var ch = new DataView(new ArrayBuffer(46));
      ch.setUint32(0, 0x02014b50, true); ch.setUint16(4, 20, true); ch.setUint16(6, 20, true);
      ch.setUint16(8, 0x0800, true); ch.setUint16(10, 0, true); ch.setUint16(12, dosTime, true);
      ch.setUint16(14, dosDate, true); ch.setUint32(16, crc, true); ch.setUint32(20, size, true);
      ch.setUint32(24, size, true); ch.setUint16(28, nameB.length, true); ch.setUint16(30, 0, true);
      ch.setUint16(32, 0, true); ch.setUint16(34, 0, true); ch.setUint16(36, 0, true);
      ch.setUint32(38, 0, true); ch.setUint32(42, offset, true);
      central.push(new Uint8Array(ch.buffer), nameB);
      cdSize += 46 + nameB.length;
      offset += 30 + nameB.length + size;
    });
    var end = new DataView(new ArrayBuffer(22));
    end.setUint32(0, 0x06054b50, true); end.setUint16(4, 0, true); end.setUint16(6, 0, true);
    end.setUint16(8, files.length, true); end.setUint16(10, files.length, true);
    end.setUint32(12, cdSize, true); end.setUint32(16, offset, true); end.setUint16(20, 0, true);
    return new Blob(parts.concat(central, [new Uint8Array(end.buffer)]), { type: 'application/zip' });
  }

  // ---- ID3v2.3 (UTF-16 รองรับภาษาไทย) ----
  function id3Frame(id, body) {
    var h = new Uint8Array(10), sz = body.length;
    h.set(ascii(id), 0);
    h[4] = (sz >>> 24) & 255; h[5] = (sz >>> 16) & 255; h[6] = (sz >>> 8) & 255; h[7] = sz & 255;
    return concatBytes([h, body]);
  }

  function id3Text(id, text) {
    var b = new Uint8Array(3 + text.length * 2);
    b[0] = 1; b[1] = 0xFF; b[2] = 0xFE;
    for (var i = 0; i < text.length; i++) {
      var c = text.charCodeAt(i);
      b[3 + i * 2] = c & 255; b[4 + i * 2] = c >> 8;
    }
    return id3Frame(id, b);
  }

  function id3Apic(cover) {
    var head = concatBytes([new Uint8Array([0]), ascii(cover.mime), new Uint8Array([0, 3, 0])]);
    return id3Frame('APIC', concatBytes([head, cover.bytes]));
  }

  function buildId3(tags, cover) {
    var frames = [];
    if (tags.title) { frames.push(id3Text('TIT2', tags.title)); }
    if (tags.artist) { frames.push(id3Text('TPE1', tags.artist)); }
    if (tags.album) { frames.push(id3Text('TALB', tags.album)); }
    if (cover && cover.bytes) { frames.push(id3Apic(cover)); }
    if (!frames.length) { return null; }
    var body = concatBytes(frames), s = body.length, h = new Uint8Array(10);
    h.set(ascii('ID3'), 0); h[3] = 3; h[4] = 0; h[5] = 0;
    h[6] = (s >>> 21) & 127; h[7] = (s >>> 14) & 127; h[8] = (s >>> 7) & 127; h[9] = s & 127;
    return concatBytes([h, body]);
  }

  // ---- ประมวลผลเสียง ----
  function toI16(src, off, n, dst) {
    for (var i = 0; i < n; i++) {
      var s = src[off + i];
      s = s > 1 ? 1 : (s < -1 ? -1 : s);
      dst[i] = s < 0 ? s * 32768 : s * 32767;
    }
    return dst.subarray(0, n);
  }

  function pickChannels(buf, s0, s1, mono) {
    var n = buf.numberOfChannels, len = s1 - s0, c, i;
    if (n === 1) { return [buf.getChannelData(0).subarray(s0, s1)]; }
    if (mono) {
      var m = new Float32Array(len);
      for (c = 0; c < n; c++) {
        var d = buf.getChannelData(c);
        for (i = 0; i < len; i++) { m[i] += d[s0 + i]; }
      }
      for (i = 0; i < len; i++) { m[i] /= n; }
      return [m];
    }
    return [buf.getChannelData(0).subarray(s0, s1), buf.getChannelData(1).subarray(s0, s1)];
  }

  function normalize(chs) {
    var peak = 0, c, i;
    for (c = 0; c < chs.length; c++) {
      for (i = 0; i < chs[c].length; i++) {
        var a = chs[c][i] < 0 ? -chs[c][i] : chs[c][i];
        if (a > peak) { peak = a; }
      }
    }
    if (peak < 0.0001) { return; }
    var gain = Math.min(30, 0.944 / peak);
    if (gain > 0.99 && gain < 1.01) { return; }
    for (c = 0; c < chs.length; c++) {
      for (i = 0; i < chs[c].length; i++) { chs[c][i] *= gain; }
    }
  }

  function fades(chs, rate, fin, fout) {
    var len = chs[0].length, n1 = Math.min(len, Math.round(fin * rate)), n2 = Math.min(len, Math.round(fout * rate)), c, i;
    for (c = 0; c < chs.length; c++) {
      for (i = 0; i < n1; i++) { chs[c][i] *= i / n1; }
      for (i = 0; i < n2; i++) { chs[c][len - 1 - i] *= i / n2; }
    }
  }

  function wavHeader(nch, rate, dataBytes) {
    var v = new DataView(new ArrayBuffer(44));
    function w(o, s) { for (var i = 0; i < s.length; i++) { v.setUint8(o + i, s.charCodeAt(i)); } }
    w(0, 'RIFF'); v.setUint32(4, 36 + dataBytes, true); w(8, 'WAVE'); w(12, 'fmt ');
    v.setUint32(16, 16, true); v.setUint16(20, 1, true); v.setUint16(22, nch, true);
    v.setUint32(24, rate, true); v.setUint32(28, rate * nch * 2, true);
    v.setUint16(32, nch * 2, true); v.setUint16(34, 16, true);
    w(36, 'data'); v.setUint32(40, dataBytes, true);
    return new Uint8Array(v.buffer);
  }

  function interleave(chs, off, n) {
    var nch = chs.length, out = new Int16Array(n * nch), j, c;
    for (j = 0; j < n; j++) {
      for (c = 0; c < nch; c++) {
        var s = chs[c][off + j];
        s = s > 1 ? 1 : (s < -1 ? -1 : s);
        out[j * nch + c] = s < 0 ? s * 32768 : s * 32767;
      }
    }
    return out;
  }

  async function encodeMp3(chs, rate, kbps, it) {
    var nch = chs.length, enc = new window.lamejs.Mp3Encoder(nch, rate, kbps);
    var len = chs[0].length, BLOCK = 1152 * 10, parts = [], last = Date.now();
    var l16 = new Int16Array(BLOCK), r16 = nch > 1 ? new Int16Array(BLOCK) : null;
    for (var i = 0; i < len; i += BLOCK) {
      var n = Math.min(BLOCK, len - i);
      var l = toI16(chs[0], i, n, l16), out;
      if (r16) { out = enc.encodeBuffer(l, toI16(chs[1], i, n, r16)); } else { out = enc.encodeBuffer(l); }
      if (out.length > 0) { parts.push(new Uint8Array(out)); }
      if (Date.now() - last > 14) {
        setProgress(it, 0.3 + 0.68 * (i / len), 'กำลังเข้ารหัส MP3… ' + Math.round(100 * i / len) + '%');
        await yieldNow();
        checkCancel();
        last = Date.now();
      }
    }
    var tail = enc.flush();
    if (tail.length > 0) { parts.push(new Uint8Array(tail)); }
    return parts;
  }

  async function encodeWav(chs, rate, it) {
    var nch = chs.length, len = chs[0].length, bytes = len * nch * 2;
    if (bytes > 4000000000) { throw ue('ไฟล์ WAV ที่ได้ใหญ่เกิน 4 GB ลองตัดเป็นช่วงสั้นลง หรือเลือก MP3'); }
    var parts = [wavHeader(nch, rate, bytes)], BLOCK = 44100, last = Date.now();
    for (var i = 0; i < len; i += BLOCK) {
      parts.push(interleave(chs, i, Math.min(BLOCK, len - i)));
      if (Date.now() - last > 14) {
        setProgress(it, 0.3 + 0.68 * (i / len), 'กำลังสร้างไฟล์ WAV… ' + Math.round(100 * i / len) + '%');
        await yieldNow();
        checkCancel();
        last = Date.now();
      }
    }
    return new Blob(parts, { type: 'audio/wav' });
  }

  async function convertOne(it) {
    var ui = it.ui, o = readOpts();
    ui.err.hidden = true; ui.out.hidden = true; ui.bar.hidden = true; ui.state.textContent = '';
    var t0 = parseTime(ui.start.value), t1 = parseTime(ui.end.value);
    if (isNaN(t0) || isNaN(t1)) { fail(it, 'รูปแบบเวลาไม่ถูกต้อง ใช้ได้เช่น 1:30 หรือ 90 (วินาที)'); return; }
    it.state = 'working';
    try {
      if (o.fmt === 'mp3') { setProgress(it, 0.02, 'กำลังโหลดตัวเข้ารหัส MP3…'); await loadLame(); }
      checkCancel();
      setProgress(it, 0.06, 'กำลังอ่านไฟล์…');
      var ab;
      try { ab = await readFile(it.file); } catch (e1) { throw tag(e1, 'read'); }
      checkCancel();
      setProgress(it, 0.12, 'กำลังถอดรหัสเสียง…');
      var buf;
      try { buf = await decodeAudio(ab, o.rate); } catch (e2) { throw tag(e2, 'decode'); }
      ab = null;
      checkCancel();
      var rate = buf.sampleRate, dur = buf.duration;
      it.duration = dur;
      var start = t0 === null ? 0 : t0, end = t1 === null ? dur : Math.min(t1, dur);
      if (start >= end) { throw ue('ช่วงเวลาไม่ถูกต้อง: เวลาเริ่มต้องน้อยกว่าเวลาสิ้นสุด และไม่เกินความยาวไฟล์ (' + fmtTime(dur) + ')'); }
      var s0 = Math.floor(start * rate), s1 = Math.min(buf.length, Math.ceil(end * rate));
      setProgress(it, 0.25, 'กำลังปรับแต่งเสียง…');
      await yieldNow();
      var chs = pickChannels(buf, s0, s1, o.mono);
      buf = null;
      if (o.norm) { normalize(chs); }
      if (o.fin > 0 || o.fout > 0) { fades(chs, rate, o.fin, o.fout); }
      checkCancel();
      var blob, ext;
      if (o.fmt === 'mp3') {
        var parts = await encodeMp3(chs, rate, o.kbps, it);
        var id3 = buildId3({ title: o.tags.title || it.base, artist: o.tags.artist, album: o.tags.album }, cover);
        blob = new Blob(id3 ? [id3].concat(parts) : parts, { type: 'audio/mpeg' });
        ext = '.mp3';
      } else {
        blob = await encodeWav(chs, rate, it);
        ext = '.wav';
      }
      chs = null;
      if (it.url) { URL.revokeObjectURL(it.url); }
      it.blob = blob; it.url = URL.createObjectURL(blob); it.outName = it.base + ext;
      it.state = 'done';
      ui.res.src = it.url;
      ui.dl.href = it.url; ui.dl.download = it.outName;
      ui.dl.textContent = 'ดาวน์โหลด ' + it.outName;
      ui.sizes.textContent = fmtSize(it.file.size) + ' → ' + fmtSize(blob.size) + ' (' + Math.round(blob.size / it.file.size * 100) + '% ของไฟล์ต้นฉบับ)';
      ui.out.hidden = false;
      setProgress(it, 1, 'เสร็จแล้ว');
      ui.one.textContent = 'แปลงใหม่';
      updateMeta(it);
    } catch (e) {
      if (e === CANCEL) {
        it.state = 'idle'; ui.bar.hidden = true; ui.state.textContent = 'ยกเลิกแล้ว';
      } else {
        fail(it, friendly(e));
      }
    }
  }

  async function acquireLock() {
    try {
      if (navigator.wakeLock && navigator.wakeLock.request) { return await navigator.wakeLock.request('screen'); }
    } catch (e) { /* ไม่รองรับก็ข้ามไป */ }

  async function runItems(list) {
    if (running || !list.length) { return; }
    running = true; cancelAll = false; setBusy(true);
    var lock = await acquireLock(), ok = 0;
    try {
      for (var i = 0; i < list.length; i++) {
        if (cancelAll) { break; }
        say('กำลังแปลงไฟล์ ' + (i + 1) + ' จาก ' + list.length + '…');
        await convertOne(list[i]);
        if (list[i].state === 'done') { ok++; }
      }
    } finally {
      if (lock && lock.release) { try { lock.release(); } catch (e) { /* ignore */ } }
      running = false; setBusy(false);
    }
    if (cancelAll) { say('ยกเลิกแล้ว'); }
    else if (ok === list.length) { say('แปลงเสร็จ ' + ok + ' ไฟล์ — ฟังตัวอย่างหรือกดดาวน์โหลดได้เลย'); }
    else { say('แปลงสำเร็จ ' + ok + ' จาก ' + list.length + ' ไฟล์ (ดูข้อความในแต่ละไฟล์)', true); }
  }

  async function downloadZip() {
    var done = items.filter(function (x) { return x.state === 'done' && x.blob; });
    if (done.length < 2 || running) { return; }
    running = true; setBusy(true);
    say('กำลังสร้างไฟล์ ZIP…');
    try {
      var used = {}, files = [];
      for (var i = 0; i < done.length; i++) {
        var nm = done[i].outName, base = nm.replace(/\.[^.]+$/, ''), ext = nm.slice(base.length), k = nm, n = 1;

    } catch (e) {

## วิธีใช้งาน

1. **เพิ่มไฟล์** — ลากไฟล์ `.m4a` (หรือไฟล์เสียง/วิดีโออื่น ๆ) มาวางในกรอบด้านบน คลิกเพื่อเลือกไฟล์ หรือกด Ctrl+V (⌘+V บน Mac) เพื่อวางไฟล์ที่คัดลอกไว้ เพิ่มได้หลายไฟล์พร้อมกัน
2. **เลือกพรีเซ็ตหรือตั้งค่าเอง** — พรีเซ็ต "มาตรฐาน" เหมาะกับเพลงทั่วไป ส่วน "เสียงพูด/พอดแคสต์" ให้ไฟล์เล็กและฟังชัด หากต้องการควบคุมละเอียดขึ้น เลือก bitrate เองหรือเปิด "ตั้งค่าเสียงขั้นสูง"
3. **ตัดช่วงเสียง (ไม่บังคับ)** — ในแต่ละไฟล์ใส่เวลาเริ่มและสิ้นสุด เช่น `0:30` และ `2:15` หรือใส่เป็นวินาที เช่น `90` เว้นว่างไว้ถ้าต้องการทั้งไฟล์ กดเล่นตัวอย่างต้นฉบับเพื่อหาจุดตัดได้
4. **ใส่ข้อมูลเพลง (ไม่บังคับ)** — เปิด "ข้อมูลเพลง (ID3) และภาพปก" เพื่อกรอกชื่อเพลง ศิลปิน อัลบั้ม และเลือกภาพปก หากเว้นชื่อเพลงไว้ ระบบจะใช้ชื่อไฟล์
5. **กด "แปลงทั้งหมด"** — ดูความคืบหน้าของแต่ละไฟล์ได้ทันที และกด "ยกเลิก" ได้ตลอดเวลา ถ้าต้องการแปลงเฉพาะบางไฟล์ ให้กด "แปลงไฟล์นี้" ที่ไฟล์นั้น
6. **ฟังตัวอย่างและดาวน์โหลด** — เล่นไฟล์ที่แปลงแล้วเพื่อเช็กเสียง จากนั้นกดดาวน์โหลดทีละไฟล์ หรือกด "ดาวน์โหลดทั้งหมด (ZIP)" เมื่อแปลงตั้งแต่ 2 ไฟล์ขึ้นไป

## คำอธิบายศัพท์ที่ใช้ในเครื่องมือ

| คำศัพท์ | ความหมายโดยย่อ |
| --- | --- |
| M4A | ไฟล์เสียงนามสกุล `.m4a` ของ Apple ใช้ใน iTunes, Apple Music และแอป Voice Memos มักบรรจุเสียงที่บีบอัดด้วย AAC |
| MP3 | รูปแบบไฟล์เสียงบีบอัดที่เล่นได้เกือบทุกอุปกรณ์ ทั้งมือถือ วิทยุรถยนต์ และเครื่องเล่นรุ่นเก่า |
| AAC | วิธีบีบอัดเสียง (codec) ที่ใหม่กว่า MP3 ให้คุณภาพใกล้เคียงกันที่ไฟล์เล็กกว่าเล็กน้อย |
| WAV | ไฟล์เสียงดิบที่ไม่บีบอัด คุณภาพเต็มแต่ไฟล์ใหญ่มาก (ประมาณ 10 MB ต่อนาที ที่ 44.1 kHz สเตอริโอ) |
| Lossy / Lossless | Lossy คือบีบอัดโดยตัดข้อมูลที่หูแยกยากออก (MP3, AAC) ส่วน Lossless คือบีบอัดโดยไม่สูญเสียข้อมูล (FLAC, ALAC) |
| Bitrate (kbps) | ปริมาณข้อมูลเสียงต่อวินาที ยิ่งสูงยิ่งละเอียดและไฟล์ใหญ่ขึ้น 1 kbps เท่ากับ 1,000 บิตต่อวินาที |
| CBR | Constant Bitrate คือใช้ bitrate คงที่ตลอดไฟล์ ทำให้คาดเดาขนาดไฟล์ได้ เครื่องมือนี้ใช้แบบนี้ |
| Sample rate (Hz) | จำนวนครั้งที่สุ่มสัญญาณเสียงต่อวินาที 44.1 kHz เป็นมาตรฐานซีดี 48 kHz พบมากในวิดีโอ ส่วน 32 kHz ให้ไฟล์เล็กลงและเหมาะกับเสียงพูด |
| โมโน / สเตอริโอ | โมโนคือเสียงช่องเดียว สเตอริโอคือสองช่อง (ซ้าย-ขวา) การแปลงเป็นโมโนช่วยลดขนาดไฟล์ได้ประมาณครึ่งหนึ่ง |
| Normalize | ปรับให้ส่วนที่ดังที่สุดของไฟล์อยู่ใกล้ระดับสูงสุดโดยไม่ทำให้เสียงแตก เหมาะกับไฟล์ที่เสียงเบา |
| Fade in / Fade out | ค่อย ๆ เพิ่มเสียงตอนเริ่ม หรือค่อย ๆ ลดเสียงตอนจบ เพื่อไม่ให้เสียงตัดกะทันหัน |
| ตัดช่วงเสียง (Trim) | เลือกเฉพาะช่วงเวลาที่ต้องการ โดยกำหนดเวลาเริ่มและเวลาสิ้นสุด |
| ID3 tag | ข้อมูลที่ฝังในไฟล์ MP3 เช่น ชื่อเพลง ศิลปิน อัลบั้ม และภาพปก ที่เครื่องเล่นนำไปแสดง |
| ภาพปก (Cover art) | รูปที่ฝังในไฟล์ MP3 เพื่อให้เครื่องเล่นแสดงเป็นปกอัลบั้ม |
| Voice Memos | แอปบันทึกเสียงใน iPhone และ Mac ซึ่งบันทึกเป็นไฟล์ M4A |
| ประมวลผลบนอุปกรณ์ | การแปลงทำงานในเบราว์เซอร์ของคุณเอง ไฟล์เสียงไม่ถูกส่งไปที่เซิร์ฟเวอร์ |
| ZIP | ไฟล์รวมหลายไฟล์เป็นไฟล์เดียว สะดวกเมื่อดาวน์โหลดหลายไฟล์พร้อมกัน |

## เลือก bitrate อย่างไรดี

| การใช้งาน | bitrate ที่แนะนำ | หมายเหตุ |
| --- | --- | --- |
| บันทึกเสียงพูด ประชุม ห้องเรียน | 64–96 kbps โมโน | ไฟล์เล็กมาก ฟังคำพูดได้ชัด |
| พอดแคสต์ หรือส่งผ่านแอปแชต | 96–128 kbps | สมดุลระหว่างขนาดไฟล์และคุณภาพ |
| เพลงทั่วไปบนมือถือหรือรถยนต์ | 192 kbps | ค่าเริ่มต้นของเครื่องมือนี้ |
| เพลงคุณภาพสูง ฟังด้วยหูฟังดี ๆ | 256–320 kbps | ไฟล์ใหญ่ขึ้นแต่เก็บรายละเอียดได้ครบกว่า |

## M4A กับ MP3 ต่างกันอย่างไร

| หัวข้อ | M4A | MP3 |
| --- | --- | --- |
| ความเข้ากันได้ | ดีบนอุปกรณ์ Apple และแอปรุ่นใหม่ แต่เครื่องเล่นรุ่นเก่าและวิทยุรถยนต์บางรุ่นเล่นไม่ได้ | เล่นได้เกือบทุกอุปกรณ์ |
| คุณภาพต่อขนาดไฟล์ | โดยทั่วไปดีกว่าเล็กน้อยที่ bitrate เท่ากัน (เมื่อใช้ AAC) | ต้องใช้ bitrate สูงกว่าเล็กน้อยเพื่อคุณภาพใกล้เคียงกัน |
| ข้อมูลเพลง | ใช้ metadata แบบ MP4 | ใช้ ID3 tag |
| เหมาะกับ | iTunes, Apple Music, Voice Memos | เครื่องเล่นทั่วไป USB ในรถ และการแชร์ไฟล์ |

การแปลงจาก M4A เป็น MP3 คือการบีบอัดแบบสูญเสียข้อมูลซ้ำอีกรอบ จึงไม่ทำให้คุณภาพดีขึ้น หากต้นฉบับเป็นเสียงสำคัญ เช่น งานบันทึกเสียงหรือเพลงที่ต้องการเก็บไว้นาน ควรเก็บไฟล์ M4A ต้นฉบับไว้ด้วย และเลือก bitrate 192 kbps ขึ้นไปเพื่อให้คุณภาพใกล้เคียงต้นฉบับ

## คำถามที่พบบ่อย

### ไฟล์ของฉันถูกอัปโหลดไปที่เซิร์ฟเวอร์หรือไม่

ไม่ การถอดรหัสและเข้ารหัสเสียงทั้งหมดทำในเบราว์เซอร์ของคุณ ไฟล์เสียง ภาพปก และข้อมูลเพลงไม่ถูกส่งออกจากอุปกรณ์ สิ่งเดียวที่ดาวน์โหลดเพิ่มคือสคริปต์ตัวเข้ารหัส MP3 ซึ่งโหลดเมื่อกดแปลงเป็น MP3 ครั้งแรก

### ใช้ฟรีหรือไม่ ต้องสมัครสมาชิกหรือไม่

ใช้ฟรี ไม่ต้องลงทะเบียน และไม่ต้องติดตั้งโปรแกรมหรือแอปเพิ่ม

### แปลงเป็น MP3 แล้วคุณภาพจะลดลงหรือไม่

เนื่องจากทั้ง M4A (AAC) และ MP3 เป็นการบีบอัดแบบสูญเสียข้อมูล การแปลงอาจทำให้คุณภาพลดลงเล็กน้อย โดยเฉพาะถ้าเลือก bitrate ต่ำ แนะนำ 192 kbps ขึ้นไปสำหรับเพลง และ 96 kbps สำหรับเสียงพูด

### แปลงไฟล์ Voice Memos จาก iPhone ได้อย่างไร

เปิดแอป Voice Memos เลือกการบันทึก กดแชร์แล้วเลือก "บันทึกลงไฟล์" จากนั้นเปิดหน้านี้ในเบราว์เซอร์ กรอบอัปโหลดจะให้เลือกไฟล์จากแอป "ไฟล์" ที่บันทึกไว้ แล้วกดแปลงได้เลย

### ทำไมแปลงไม่สำเร็จ หรือขึ้นข้อความผิดพลาด

สาเหตุที่พบบ่อยคือไฟล์เสียหาย มีการป้องกันลิขสิทธิ์ (DRM) เช่น เพลงที่ซื้อจากร้านค้าบางแห่ง เป็นไฟล์ Apple Lossless ที่เบราว์เซอร์บางตัวถอดรหัสไม่ได้ หรือหน่วยความจำของอุปกรณ์ไม่พอสำหรับไฟล์ยาวมาก ลองใช้ Safari หรือ Chrome รุ่นล่าสุด ตัดไฟล์เป็นช่วงสั้นลง หรือปิดแท็บอื่นก่อน

### ไฟล์ใหญ่หรือยาวสุดเท่าไร

ไม่มีขีดจำกัดจากเว็บไซต์ แต่เสียงที่ถอดรหัสแล้วต้องเก็บไว้ในหน่วยความจำของอุปกรณ์ ไฟล์ที่ยาวหลายชั่วโมงอาจใช้หน่วยความจำมาก โดยเฉพาะบนมือถือ ถ้าเจอปัญหาให้ตัดทีละช่วงด้วยช่อง "เริ่มที่" และ "สิ้นสุดที่"

### ใช้งานแบบออฟไลน์ได้หรือไม่

ต้องออนไลน์ในครั้งแรกเพื่อโหลดตัวเข้ารหัส MP3 หากเลือกรูปแบบ WAV จะไม่ต้องโหลดสคริปต์เพิ่ม

### ทำไมต้องเปิดแท็บค้างไว้ระหว่างแปลง

เบราว์เซอร์บนมือถือมักจำกัดการทำงานของแท็บที่อยู่เบื้องหลังหรือเมื่อหน้าจอดับ ทำให้การแปลงช้าลงหรือหยุดชั่วคราว เครื่องมือนี้พยายามขอให้หน้าจอไม่ดับระหว่างแปลงในเบราว์เซอร์ที่รองรับ แต่ถ้าสลับแอป การแปลงอาจหยุดรอจนกว่าจะกลับมาที่แท็บนี้
