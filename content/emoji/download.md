  }

  function makeImg(ch, style, useSvg) {
    var img = document.createElement('img');
    var u = urls(ch, style);
    img.alt = ch;
    img.decoding = 'async';
    img.src = useSvg ? u.svg : u.png;
    img.addEventListener('error', function () {
      if (img.parentNode) {
        img.parentNode.replaceChild(document.createTextNode(ch), img);
      }
    });
    return img;
  }

  function renderGrid() {
    var wrap = $('emj-grid');
    wrap.textContent = '';
    CATS.forEach(function (cat) {
      if (state.cat !== 'all' && state.cat !== cat.id) { return; }
      var sec = document.createElement('section');
      sec.className = 'emj-sec';
      sec.id = 'emj-' + cat.id;
      var h = document.createElement('h3');
      h.textContent = cat.name + ' (' + cat.items.length + ')';
      var grid = document.createElement('div');
      grid.className = 'emj-g';
      cat.items.forEach(function (ch) {
        var b = document.createElement('button');
        b.type = 'button';
        b.className = 'emj-b';
        b.setAttribute('data-e', ch);
        b.setAttribute('aria-label', ch + ' ' + codeLabel(ch));
        var img = makeImg(ch, state.style, false);
        img.loading = 'lazy';
        b.appendChild(img);
        grid.appendChild(b);
      });
      sec.appendChild(h);
      sec.appendChild(grid);
      wrap.appendChild(sec);
    });
  }

  function renderCats() {
    var box = $('emj-cats');
    box.textContent = '';
    var all = [{ id: 'all', name: 'ทั้งหมด' }].concat(CATS);
    all.forEach(function (c) {
      var b = document.createElement('button');
      b.type = 'button';
      b.className = 'emj-chip';
      b.textContent = c.name;
      b.setAttribute('data-c', c.id);
      b.setAttribute('aria-pressed', String(state.cat === c.id));
      box.appendChild(b);
    });
  }

  function say(id, text) { $(id).textContent = text; }

  function copyText(text) {
    if (navigator.clipboard && navigator.clipboard.writeText) {
      return navigator.clipboard.writeText(text);
    }
    return new Promise(function (resolve, reject) {
      var ta = document.createElement('textarea');
      ta.value = text;
      ta.style.position = 'fixed';
      ta.style.opacity = '0';
      document.body.appendChild(ta);
      ta.select();
      var ok = false;
      try { ok = document.execCommand('copy'); } catch (e) { ok = false; }
      document.body.removeChild(ta);
      if (ok) { resolve(); } else { reject(new Error('copy failed')); }
    });
  }

  function openModal(ch) {
    state.cur = ch;
    $('emj-mt').textContent = 'ดาวน์โหลด ' + ch;
    $('emj-code').textContent = codeLabel(ch);
    ['tw', 'go'].forEach(function (s) {
      var el = $('emj-pv-' + s);
      el.alt = ch;
      el.src = urls(ch, s).svg;
    });
    say('emj-mmsg', '');
    $('emj-ov').classList.add('emj-show');
  }
  function closeModal() { $('emj-ov').classList.remove('emj-show'); }

  function fetchBlob(url) {
    return fetch(url).then(function (r) {
      if (!r.ok) { throw new Error('HTTP ' + r.status); }
      return r.blob();
    });
  }
  function saveBlob(blob, name) {
    var a = document.createElement('a');
    var u = URL.createObjectURL(blob);
    a.href = u;
    a.download = name;
    document.body.appendChild(a);
    a.click();
    setTimeout(function () {
      URL.revokeObjectURL(u);
      document.body.removeChild(a);
    }, 1500);
  }
  function svgToPng(svgBlob, size) {
    return svgBlob.text().then(function (txt) {
      var doc = new DOMParser().parseFromString(txt, 'image/svg+xml');
      var root = doc.documentElement;
      if (!root || root.nodeName.toLowerCase() !== 'svg') { throw new Error('bad svg'); }
      if (!root.getAttribute('viewBox')) {
        var w = parseFloat(root.getAttribute('width')) || 128;
        var h = parseFloat(root.getAttribute('height')) || 128;
        root.setAttribute('viewBox', '0 0 ' + w + ' ' + h);
      }
      root.setAttribute('width', String(size));
      root.setAttribute('height', String(size));
      var xml = new XMLSerializer().serializeToString(root);
      var blob = new Blob([xml], { type: 'image/svg+xml;charset=utf-8' });
      var objUrl = URL.createObjectURL(blob);
      return new Promise(function (resolve, reject) {
        var img = new Image();
        img.onload = function () {
          var cv = document.createElement('canvas');
          cv.width = size;
          cv.height = size;
          cv.getContext('2d').drawImage(img, 0, 0, size, size);
          URL.revokeObjectURL(objUrl);
          cv.toBlob(function (b) {
            if (b) { resolve(b); } else { reject(new Error('toBlob failed')); }
          }, 'image/png');
        };
        img.onerror = function () {
          URL.revokeObjectURL(objUrl);
          reject(new Error('image load failed'));
        };
        img.src = objUrl;
      });
    });
  }

  function download(style, fmt, btn) {
    var ch = state.cur;
    if (!ch) { return; }
    var u = urls(ch, style);
    var size = parseInt($('emj-size').value, 10) || 512;
    var base = 'emoji-' + u.id + '-' + (style === 'tw' ? 'twitter' : 'google');
    var failMsg = 'ไม่พบไฟล์สำหรับอิโมจิตัวนี้ในสไตล์ที่เลือก ลองสไตล์อื่นหรือคัดลอกอิโมจิแทน';
    btn.disabled = true;
    say('emj-mmsg', 'กำลังเตรียมไฟล์...');
    var job;
    if (fmt === 'svg') {
      job = fetchBlob(u.svg).then(function (b) { saveBlob(b, base + '.svg'); });
    } else {
      job = fetchBlob(u.svg).then(function (b) {
        return svgToPng(b, size).then(function (png) { saveBlob(png, base + '-' + size + 'px.png'); });
      }).catch(function () {
        return fetchBlob(u.png).then(function (b) { saveBlob(b, base + '.png'); });
      });
    }
    job.then(function () {
      say('emj-mmsg', 'ดาวน์โหลดแล้ว: ' + base + '.' + fmt);
    }).catch(function () {
      say('emj-mmsg', failMsg);
    }).then(function () {
      btn.disabled = false;
    });
  }

  function firstEmoji(s) {
    s = s.replace(/\s+/g, '');
    if (!s) { return ''; }
    if (typeof Intl !== 'undefined' && Intl.Segmenter) {
      var it = new Intl.Segmenter(undefined, { granularity: 'grapheme' }).segment(s)[Symbol.iterator]().next();
      return it.value ? it.value.segment : '';
    }
    return s;
  }
  function looksLikeEmoji(s) {
    try {
      return new RegExp('\\p{Extended_Pictographic}|\\p{Regional_Indicator}|\\u20E3', 'u').test(s);
    } catch (e) {
      return s.codePointAt(0) > 0x7F;
    }
  }

  function onPick(ch) {
    if (state.click === 'copy') {
      copyText(ch).then(function () {
        say('emj-msg', 'คัดลอก ' + ch + ' แล้ว');
      }).catch(function () {
        say('emj-msg', 'คัดลอกไม่สำเร็จ กรุณาลองอีกครั้ง');
      });
    } else {
      openModal(ch);
    }
  }

  document.querySelectorAll('input[name="emj-style"]').forEach(function (r) {
    r.addEventListener('change', function () {
      if (r.checked) { state.style = r.value; renderGrid(); }
    });
  });
  document.querySelectorAll('input[name="emj-click"]').forEach(function (r) {
    r.addEventListener('change', function () {
      if (r.checked) { state.click = r.value; say('emj-msg', ''); }
    });
  });
  $('emj-cats').addEventListener('click', function (e) {
    var b = e.target.closest('.emj-chip');
    if (!b) { return; }
    state.cat = b.getAttribute('data-c');
    renderCats();
    renderGrid();
  });
  $('emj-grid').addEventListener('click', function (e) {
    var b = e.target.closest('.emj-b');
    if (b) { onPick(b.getAttribute('data-e')); }
  });
  $('emj-open').addEventListener('click', function () {
    var ch = firstEmoji($('emj-in').value);
    if (!ch || !looksLikeEmoji(ch)) {
      say('emj-msg', 'กรุณาวางอิโมจิ 1 ตัวในช่องก่อน');
      return;
    }
    say('emj-msg', '');
    onPick(ch);
  });
  $('emj-in').addEventListener('keydown', function (e) {
    if (e.key === 'Enter') { $('emj-open').click(); }
  });
  $('emj-close').addEventListener('click', closeModal);
  $('emj-ov').addEventListener('click', function (e) {
    if (e.target === $('emj-ov')) { closeModal(); }
  });
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape') { closeModal(); }
  });
  document.querySelectorAll('#emj-app .emj-d').forEach(function (b) {
    b.addEventListener('click', function () {
      download(b.getAttribute('data-s'), b.getAttribute('data-f'), b);
    });
  });
  $('emj-copy').addEventListener('click', function () {
    copyText(state.cur).then(function () {
      say('emj-mmsg', 'คัดลอก ' + state.cur + ' แล้ว');
    }).catch(function () {
      say('emj-mmsg', 'คัดลอกไม่สำเร็จ กรุณาลองอีกครั้ง');
    });
  });

  renderCats();
  renderGrid();
})();
</script>

## วิธีใช้งานเครื่องมือ

1. **เลือกสไตล์ภาพ** ที่ด้านบนสุด ได้แก่ Twitter Emoji (Twemoji) ที่ดีไซน์เรียบ ไม่มีเส้นขอบ หรือ Google Emoji (Noto) ที่มีเส้นขอบและรายละเอียดมากกว่า ตารางอิโมจิจะเปลี่ยนตามสไตล์ที่เลือกทันที
2. **เลือกพฤติกรรมเมื่อคลิก** โดยเลือก "ป๊อปอัพรายการดาวน์โหลด" หากต้องการไฟล์ภาพ หรือเลือก "คัดลอกอิโมจิ" หากต้องการนำตัวอักษรอิโมจิไปวางในแชตหรือโซเชียลมีเดีย
3. **เลือกหมวดหมู่** ด้วยปุ่มกลมๆ เช่น หน้ายิ้มและผู้คน สัตว์และธรรมชาติ หรือธง เพื่อแสดงเฉพาะหมวดที่ต้องการ กด "ทั้งหมด" เพื่อดูครบทุกหมวด
4. **คลิกที่อิโมจิ** ที่ต้องการ หากอยู่ในโหมดป๊อปอัพ จะมีหน้าต่างแสดงตัวอย่างภาพทั้งสองสไตล์พร้อมรหัส Unicode
5. **ปรับขนาดไฟล์ PNG** ในหน้าต่างป๊อปอัพ (64 ถึง 1024 พิกเซล) แล้วกดลูกศรลงในช่อง .PNG หรือ .SVG ของสไตล์ที่ต้องการเพื่อดาวน์โหลด ไฟล์ SVG จะคมชัดทุกขนาด ส่วน PNG จะถูกสร้างตามขนาดที่เลือก
6. **หาอิโมจิที่ไม่อยู่ในรายการ** ได้ด้วยการวางอิโมจิลงในช่องข้อความด้านบน แล้วกดปุ่ม "เปิดตัวเลือกดาวน์โหลด" (หรือกด Enter)
7. **ปิดหน้าต่างป๊อปอัพ** ด้วยปุ่ม × การคลิกพื้นที่ว่างด้านนอก หรือการกดปุ่ม Esc
