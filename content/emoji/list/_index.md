  function loadState() {
    var r = lsGet(LS_RECENT);

      try {
        var a = JSON.parse(r);

  function withTone(e) {
    var ch = e[0];
    if (!tone || !e[3]) { return ch; }
    var arr = Array.from(ch);
    var pos = e[3].split(',').map(Number);
    var mod = String.fromCodePoint(tone);
    var out = '';
    for (var i = 0; i < arr.length; i++) {
      out += arr[i];
      if (pos.indexOf(i) !== -1) { out += mod; }
    }
    return out;
  }

  function renderSearch() {

    var html = '';

    for (var i = 1; i < CATS.length; i++) {

  function observe() {
    if (!('IntersectionObserver' in window)) { return; }
    if (io) { io.disconnect(); }
    io = new IntersectionObserver(function (entries) {
      entries.forEach(function (en) {
        if (en.isIntersecting) {
          var id = en.target.getAttribute('data-cat');
          var tabs = document.querySelectorAll('#emo-nav .emo-tab');
          for (var i = 0; i < tabs.length; i++) {
            tabs[i].classList.toggle('on', tabs[i].getAttribute('data-go') === id);

        }
      });
    }, { rootMargin: '-20% 0px -70% 0px' });
    var secs = body.querySelectorAll('section.cat');
    for (var i = 0; i < secs.length; i++) { io.observe(secs[i]); }

  function copyBox() {
    var text = box.value;
    if (!text) { toast('ยังไม่มี Emoji ให้คัดลอก'); return; }
    var done = function (ok) { toast(ok ? 'คัดลอกแล้ว' : 'คัดลอกไม่สำเร็จ ลองกดค้างที่ช่องข้อความแล้วเลือกคัดลอก'); };
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(text).then(function () { done(true); }, function () { done(legacyCopy()); });
    } else {
      done(legacyCopy());
    }
  }

## วิธีใช้งาน

1. **เลือกหมวดหมู่** แตะไอคอนบนแถบด้านบน เช่น 😀 หน้ายิ้มและผู้คน หรือ 🚩 ธง เพื่อเลื่อนไปยังหมวดนั้นทันที
2. **ค้นหา Emoji** พิมพ์ชื่อภาษาไทยหรืออังกฤษในช่องค้นหา เช่น "หัวใจ" "ไฟ" หรือ "cat" ระบบจะแสดงเฉพาะรายการที่ตรงกัน
3. **เลือกโทนสีผิว (ถ้าต้องการ)** กดวงกลมสีผิวเพื่อปรับ Emoji รูปมือและผู้คนทั้งหน้า ระบบจะจำค่าที่เลือกไว้ให้
4. **คลิก Emoji** ระบบจะเพิ่มลงในช่องข้อความด้านบนและคัดลอกให้ทันที
5. **คลิกต่อเนื่องเพื่อเรียงเป็นชุด** ทุกครั้งที่คลิก ระบบจะคัดลอกทั้งชุดที่อยู่ในช่องข้อความ หรือกดปุ่ม "คัดลอก" ซ้ำได้เสมอ
6. **นำไปวาง** ใช้ Ctrl+V (Windows) Command+V (Mac) หรือกดค้างแล้วเลือก "วาง" บนมือถือ
7. **ใช้ซ้ำง่ายขึ้น** ตัวที่คลิกล่าสุดจะอยู่ในหมวด 🕓 ที่ใช้ล่าสุด ส่วนปุ่ม "ล้าง" จะเคลียร์ช่องข้อความเพื่อเริ่มชุดใหม่

## คำอธิบายศัพท์ที่ควรรู้

- **Emoji (อิโมจิ)** คือภาพสัญลักษณ์ขนาดเล็กที่พิมพ์ได้เหมือนตัวอักษร ใช้แทนอารมณ์ สิ่งของ สัตว์ หรือแนวคิดในข้อความ
- **Unicode** คือมาตรฐานสากลที่กำหนดรหัสให้ตัวอักษรและ Emoji จึงส่งข้ามแอปและอุปกรณ์ได้ และมีการเพิ่ม Emoji ใหม่เป็นระยะ
- **หมวดหมู่ (Category)** คือกลุ่มที่แบ่งตามเนื้อหา เช่น หน้ายิ้มและผู้คน สัตว์และธรรมชาติ อาหารและเครื่องดื่ม กิจกรรม การเดินทางและสถานที่ สิ่งของ สัญลักษณ์ และธง
- **ที่ใช้ล่าสุด (Recent)** คือรายการ Emoji ที่คุณคลิกล่าสุด เก็บไว้ในเบราว์เซอร์ของคุณเท่านั้น
- **ช่องข้อความ** คือพื้นที่ด้านบนที่รวบรวม Emoji ที่คุณคลิก เพื่อคัดลอกเป็นชุดเดียวกัน
- **โทนสีผิว (Skin Tone Modifier)** คือตัวปรับสีผิว 5 ระดับ ใช้กับ Emoji รูปมือและผู้คน
- **ZWJ (Zero Width Joiner)** คืออักขระที่มองไม่เห็น ใช้เชื่อม Emoji หลายตัวให้กลายเป็นตัวเดียว เช่น ครอบครัว หรือรูปคนทำอาชีพต่าง ๆ
- **ธง (Flag)** คือธงประเทศที่สร้างจากอักษรบ่งชี้ภูมิภาค 2 ตัว บางระบบ เช่น Windows แสดงเป็นอักษรย่อประเทศแทนรูปธง
- **อิโมจิขาวดำ (Text Style)** คือ Emoji ที่แสดงเป็นเส้นสีเดียวตามสีตัวอักษร ไม่มีสีสัน เหมาะกับงานที่ต้องการลุคเรียบง่าย
- **Variation Selector (VS15 และ VS16)** คืออักขระที่มองไม่เห็นและบอกว่าสัญลักษณ์ควรแสดงเป็นแบบข้อความขาวดำ (VS15) หรือแบบรูปสี (VS16)
- **Emoticon (อีโมติคอน)** คือหน้าตาที่สร้างจากตัวอักษรธรรมดา เช่น :) ต่างจาก Emoji ที่เป็นภาพสัญลักษณ์
