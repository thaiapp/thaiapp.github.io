function crc32(u8) {

    for (var n = 0; n < 256; n++) { var c = n; for (var k = 0; k < 8; k++) { c = (c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1); } CRC[n] = c >>> 0; }

## วิธีใช้งานเครื่องมือแปลงวิดีโอเป็น GIF

1. **เพิ่มวิดีโอ** ลากไฟล์มาวางในกรอบด้านบน คลิกเพื่อเลือกไฟล์ หรือกด Ctrl+V (⌘+V บน Mac) เพื่อวางไฟล์ที่คัดลอกไว้ เลือกได้หลายไฟล์พร้อมกัน ไฟล์จะถูกประมวลผลในเบราว์เซอร์ของคุณเอง ไม่ถูกส่งไปยังเซิร์ฟเวอร์
2. **เลือกช่วงที่ต้องการ** เลื่อนแถบ "จุดเริ่มต้น" และ "จุดสิ้นสุด" หรือเล่นวิดีโอไปยังตำแหน่งที่ต้องการแล้วกดปุ่มตั้งจุดเริ่ม/จุดสิ้นสุด กด "เล่นเฉพาะช่วงที่เลือก" เพื่อดูตัวอย่างก่อนสร้างไฟล์ ช่วงที่แนะนำคือไม่เกิน 10–15 วินาที
3. **เลือก Preset หรือปรับค่าเอง** Preset จะตั้งความกว้าง FPS จำนวนสี การบีบอัด และขนาดไฟล์เป้าหมายให้ทันที หากต้องการควบคุมเอง ให้ปรับแต่ละตัวเลือกได้อิสระ
4. **ครอปและใส่ข้อความ (ถ้าต้องการ)** เลือกสัดส่วนภาพ เช่น 1:1 หรือ 9:16 แล้วเลื่อนตำแหน่งครอปให้ตรงส่วนที่สำคัญ หรือพิมพ์ข้อความเพื่อใส่แคปชันบนภาพ
5. **ตรวจสอบค่าประมาณแล้วกด "สร้าง GIF"** ระบบจะแสดงจำนวนเฟรม ขนาดภาพ และขนาดไฟล์โดยประมาณ ระหว่างสร้างไฟล์ให้เปิดแท็บนี้ไว้ และกด "ยกเลิก" ได้ทุกเมื่อ
6. **ดูตัวอย่างและดาวน์โหลด** เมื่อเสร็จแล้วจะมีตัวอย่าง GIF พร้อมขนาดไฟล์ กด "ดาวน์โหลด GIF" หรือหากแปลงหลายไฟล์ ให้กด "ดาวน์โหลดทั้งหมดเป็น ZIP"

### เคล็ดลับทำ GIF ให้ไฟล์เล็กแต่ยังคมชัด

- **ตัดให้สั้นที่สุด** ขนาดไฟล์เพิ่มขึ้นตามจำนวนเฟรม ตัดเฉพาะช่วงสำคัญจะช่วยได้มากที่สุด
- **ลดความกว้างก่อนลด FPS** ความกว้าง 320–480 px เพียงพอสำหรับแชทและโซเชียลส่วนใหญ่
- **ใช้ FPS 10–15** ให้ความลื่นไหลพอดูและไฟล์ไม่ใหญ่เกินไป
- **ตั้งขนาดไฟล์เป้าหมาย** หากไฟล์แรกเกินเป้าหมาย ระบบจะลดความกว้าง FPS และจำนวนสีให้อัตโนมัติจนถึง 4 รอบ
- **เลือก Lossy ระดับต่ำถึงกลาง** ช่วยลดขนาดได้มากโดยภาพเปลี่ยนไปเล็กน้อย
- **ภาพที่เปลี่ยนน้อยจะได้ไฟล์เล็ก** เช่น วิดีโอแคปหน้าจอ เมื่อเทียบกับวิดีโอที่กล้องขยับตลอดเวลา

## ความหมายของแต่ละตัวเลือก

- **GIF** รูปแบบไฟล์ภาพเคลื่อนไหวที่เล่นวนได้เองโดยไม่ต้องมีเสียง เปิดได้เกือบทุกแอปและเว็บ แต่มีสีได้สูงสุด 256 สี
- **เฟรม (Frame)** ภาพนิ่งแต่ละภาพที่นำมาเรียงต่อกันเป็นภาพเคลื่อนไหว ยิ่งมีเฟรมมาก ไฟล์ยิ่งใหญ่
- **เฟรมเรต (FPS)** จำนวนเฟรมต่อวินาที ค่าสูงให้ภาพลื่นกว่าแต่ไฟล์ใหญ่กว่า ค่า 10–15 เหมาะกับงานทั่วไป
- **ความกว้าง (px)** ความกว้างของ GIF เป็นพิกเซล ระบบคำนวณความสูงให้ตามสัดส่วน และจะไม่ขยายใหญ่กว่าวิดีโอต้นฉบับ
- **จุดเริ่มต้น / จุดสิ้นสุด** ช่วงเวลาของวิดีโอที่จะนำมาทำ GIF
- **ความเร็ว** ปรับให้เล่นเร็วขึ้นหรือช้าลงเมื่อเทียบกับวิดีโอต้นฉบับ เช่น 2x คือเร็วขึ้นเท่าตัว
- **ทิศทางการเล่น** "ปกติ" เล่นจากต้นไปท้าย "ย้อนกลับ" เล่นจากท้ายไปต้น และ "ไปกลับ (Boomerang)" เล่นไปแล้วย้อนกลับ
- **จำนวนรอบ (Loop)** ให้ GIF เล่นวนไม่รู้จบ หรือเล่นจำนวนครั้งที่กำหนดแล้วหยุด
- **สัดส่วนภาพ (ครอป)** ตัดขอบภาพให้ได้สัดส่วนที่ต้องการ เช่น 1:1 สำหรับโปรไฟล์ หรือ 9:16 สำหรับสตอรี่
- **ตำแหน่งครอป** เลื่อนกรอบครอปไปทางซ้าย/ขวา หรือบน/ล่าง เพื่อเลือกส่วนของภาพที่จะเก็บไว้
- **จำนวนสี (Color palette)** จำนวนสีที่ใช้ในภาพ ยิ่งน้อยไฟล์ยิ่งเล็ก แต่ไล่สีจะหยาบลง
- **ดิเธอริง (Dithering)** เทคนิคใช้จุดสีผสมกันเพื่อให้ดูเหมือนมีสีมากขึ้น "แบบเรียงตัว" ได้ไฟล์เล็กกว่า ส่วน Floyd–Steinberg ไล่สีเนียนกว่าแต่ไฟล์ใหญ่กว่า
- **การบีบอัดแบบสูญเสีย (Lossy)** ยอมให้สีที่ใกล้เคียงกันในเฟรมถัดไปใช้สีเดิมได้ ทำให้ไฟล์เล็กลงโดยภาพเปลี่ยนไปเล็กน้อย
- **ขนาดไฟล์เป้าหมาย** กำหนดขนาดสูงสุดที่ต้องการ หากเกิน ระบบจะลดค่าต่าง ๆ ให้อัตโนมัติ
- **ข้อความบนภาพ** ใส่แคปชันลงในทุกเฟรม มีขอบตัวอักษรเพื่อให้อ่านง่ายบนพื้นหลังทุกสี
- **Preset** ชุดค่าที่ตั้งไว้ล่วงหน้าตามการใช้งาน เช่น ไฟล์เล็กสำหรับอีโมจิ หรือคุณภาพสูงสำหรับงานแสดงผล
- **ประมาณการขนาดไฟล์** ค่าคาดการณ์ก่อนสร้างจริง ขนาดจริงขึ้นกับเนื้อหาของภาพ
- **ไฟล์ ZIP** ไฟล์บีบรวมที่ใช้ดาวน์โหลด GIF หลายไฟล์ในครั้งเดียว

## คำถามที่พบบ่อย

### วิดีโอของฉันถูกอัปโหลดไปที่เซิร์ฟเวอร์หรือไม่

ไม่ การแปลงทั้งหมดเกิดขึ้นในเบราว์เซอร์บนอุปกรณ์ของคุณ ไฟล์วิดีโอและ GIF ที่สร้างขึ้นจึงไม่ถูกส่งออกไปที่ใด และปิดแท็บเมื่อใดไฟล์ก็หายไปจากหน่วยความจำ

### รองรับไฟล์วิดีโอชนิดใดบ้าง

ขึ้นอยู่กับเบราว์เซอร์ที่ใช้ โดยทั่วไปเปิดไฟล์ MP4 (H.264), WebM และ MOV ได้ ไฟล์บางประเภท เช่น AVI หรือ MKV บางชนิดอาจเปิดไม่ได้ ให้ลองแปลงเป็น MP4 ก่อน

### ทำไมไฟล์ GIF ถึงใหญ่กว่าวิดีโอต้นฉบับ

GIF เก็บภาพทีละเฟรมโดยมีสีจำกัด 256 สีและไม่มีการบีบอัดข้ามเฟรมที่ซับซ้อนแบบวิดีโอ ลองลดความกว้าง ลด FPS ตัดช่วงให้สั้นลง หรือเปิดใช้ Lossy และกำหนดขนาดไฟล์เป้าหมาย

### ทำ GIF ได้ยาวสูงสุดเท่าไร

ไม่มีขีดจำกัดตายตัว แต่เพื่อป้องกันเบราว์เซอร์ใช้หน่วยความจำมากเกินไป เครื่องมือจะจำกัดจำนวนเฟรมไม่เกิน 500 เฟรมต่อ GIF และปรับค่าให้อัตโนมัติเมื่อเกิน แนะนำให้ใช้ช่วงสั้น ๆ ไม่เกิน 10–15 วินาที

### มีลายน้ำหรือต้องสมัครสมาชิกหรือไม่

ไม่มี ใช้ได้ฟรี ไม่ต้องสมัครสมาชิก และไฟล์ที่ได้ไม่มีลายน้ำ

### ใช้บนมือถือได้หรือไม่

ได้ แต่วิดีโอยาวหรือความละเอียดสูงอาจใช้เวลานานหรือหน่วยความจำไม่พอ แนะนำให้เลือกช่วงสั้น ๆ และใช้ Preset "เล็กมาก" หรือ "แชททั่วไป"

### สร้าง GIF ไม่สำเร็จหรือค้าง ต้องทำอย่างไร

ลองตัดช่วงให้สั้นลง ลดความกว้าง หรือลด FPS แล้วสร้างใหม่ หากใช้บนมือถือให้ปิดแท็บอื่นก่อน และตรวจสอบว่าเบราว์เซอร์เปิดไฟล์วิดีโอนั้นได้ตามปกติ
