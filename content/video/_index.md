      cv.captureStream(P.fps).getVideoTracks().forEach(function (t) { tracks.push(t); });

        var AC = window.AudioContext || window.webkitAudioContext;

        if (!alive) { return; }

        var f = 1;

          if (S.cancelled || tNow >= segEnd - 0.03 || pv.ended || recErr) { var r = waiting; waiting = null; r(); }
          else if (tNow !== lastT) { lastT = tNow; lastMove = performance.now(); }
          else if (!pv.paused && performance.now() - lastMove > 12000) { var r2 = waiting; waiting = null; stall = true; r2(); }
        }

      }

        if (S.cancelled) { throw CANCEL; }

        await seek(pv, sg[0]);

    }
  }

  async function sniffExt(blob, fallback) {

  }

  function medianCut(hist, n) {
    var keys = [];
    for (var i = 0; i < 32768; i++) { if (hist[i]) { keys.push(i); } }
    if (!keys.length) { return [[0, 0, 0]]; }
    function stat(list) {
      var a = 31, A = 0, b = 31, B = 0, c = 31, C = 0, cnt = 0;
      for (var j = 0; j < list.length; j++) {
        var k = list[j], r = k >> 10, g = (k >> 5) & 31, bl = k & 31;
        if (r < a) { a = r; } if (r > A) { A = r; }
        if (g < b) { b = g; } if (g > B) { B = g; }
        if (bl < c) { c = bl; } if (bl > C) { C = bl; }
        cnt += hist[k];
      }
      var dr = A - a, dg = B - b, db = C - c;
      return { list: list, cnt: cnt, axis: (dr >= dg && dr >= db) ? 0 : (dg >= db ? 1 : 2), rng: Math.max(dr, dg, db) };
    }
    var boxes = [stat(keys)];
    while (boxes.length < n) {
      var bi = -1, bs = -1;
      for (var q = 0; q < boxes.length; q++) {
        var bx = boxes[q];
        if (bx.list.length > 1) { var sc = bx.cnt * (bx.rng + 1); if (sc > bs) { bs = sc; bi = q; } }
      }
      if (bi < 0) { break; }
      var box = boxes[bi], sh = box.axis === 0 ? 10 : (box.axis === 1 ? 5 : 0);
      var L = box.list.slice().sort(function (p, o) { return ((p >> sh) & 31) - ((o >> sh) & 31); });
      var acc = 0, cut = 1;
      for (var z = 0; z < L.length - 1; z++) { acc += hist[L[z]]; cut = z + 1; if (acc >= box.cnt / 2) { break; } }
      boxes.splice(bi, 1, stat(L.slice(0, cut)), stat(L.slice(cut)));
    }
    return boxes.map(function (bx2) {
      var r = 0, g = 0, b = 0, c = 0;
      for (var y = 0; y < bx2.list.length; y++) { var k = bx2.list[y], w = hist[k]; r += (k >> 10) * w; g += ((k >> 5) & 31) * w; b += (k & 31) * w; c += w; }
      return [Math.round(r / c * 255 / 31), Math.round(g / c * 255 / 31), Math.round(b / c * 255 / 31)];
    });
  }
  function makeMapper(pal, nPal) {
    var cache = new Int16Array(32768).fill(-1);
    return function (key) {
      var c = cache[key];
      if (c >= 0) { return c; }
      var r = key >> 10, g = (key >> 5) & 31, b = key & 31;
      var R8 = (r << 3) | (r >> 2), G8 = (g << 3) | (g >> 2), B8 = (b << 3) | (b >> 2);
      var best = 0, bd = 1e9;
      for (var i = 0; i < nPal; i++) {
        var p = pal[i], dr = R8 - p[0], dg = G8 - p[1], db = B8 - p[2];
        var d = dr * dr * 3 + dg * dg * 4 + db * db * 2;
        if (d < bd) { bd = d; best = i; }
      }
      cache[key] = best;
      return best;
    };
  }
  function indexFrame(data, w, h, map, dith, amp) {
    var out = new Uint8Array(w * h), p = 0, o = 0;
    for (var y = 0; y < h; y++) {
      for (var x = 0; x < w; x++, p += 4, o++) {
        var r = data[p], g = data[p + 1], b = data[p + 2];
        if (dith) {
          var d = (BAYER[((y & 3) << 2) | (x & 3)] - 7.5) * amp;
          r += d; g += d; b += d;
          r = r < 0 ? 0 : (r > 255 ? 255 : r); g = g < 0 ? 0 : (g > 255 ? 255 : g); b = b < 0 ? 0 : (b > 255 ? 255 : b);
        }
        out[o] = map(((r >> 3) << 10) | ((g >> 3) << 5) | (b >> 3));
      }
    }
    return out;
  }

  function lzwEncode(idx, minCode) {
    if (!LZ.tbl) { LZ.tbl = new Int32Array(1 << 20); LZ.used = new Int32Array(4100); }
    var tbl = LZ.tbl, used = LZ.used, nu = 0;
    var clear = 1 << minCode, eoi = clear + 1;
    var out = [], cur = 0, nb = 0, size = minCode + 1, next = eoi + 1;
    function emit(c) { cur |= c << nb; nb += size; while (nb >= 8) { out.push(cur & 255); cur >>>= 8; nb -= 8; } }
    emit(clear);
    var pre = idx[0];
    for (var i = 1; i < idx.length; i++) {
      var k = idx[i], key = (pre << 8) | k, c = tbl[key];
      if (c) { pre = c; continue; }
      emit(pre);
      if (next === 4096) {
        emit(clear);
        for (var j = 0; j < nu; j++) { tbl[used[j]] = 0; }
        nu = 0; next = eoi + 1; size = minCode + 1;
      } else {
        if (next >= (1 << size)) { size++; }
        tbl[key] = next++; used[nu++] = key;
      }
      pre = k;
    }
    emit(pre); emit(eoi);
    if (nb > 0) { out.push(cur & 255); }
    for (var z = 0; z < nu; z++) { tbl[used[z]] = 0; }
    var res = [minCode];
    for (var s = 0; s < out.length; s += 255) {
      var len = Math.min(255, out.length - s);
      res.push(len);
      for (var q = 0; q < len; q++) { res.push(out[s + q]); }
    }
    res.push(0);
    return Uint8Array.from(res);
  }

## วิธีใช้งานเครื่องมือวิดีโอ

ทั้งสามเครื่องมือใช้ขั้นตอนเริ่มต้นเหมือนกัน แล้วค่อยแยกตามงานที่ต้องการ

### 1. เลือกไฟล์วิดีโอ

ลากไฟล์วิดีโอมาวางในกรอบด้านบน คลิกเพื่อเลือกไฟล์ หรือกด Ctrl+V เพื่อวางไฟล์จากคลิปบอร์ด เมื่อโหลดสำเร็จ ระบบจะแสดงชื่อไฟล์ ความละเอียด ความยาว และขนาดไฟล์ พร้อมตัวเล่นสำหรับดูตัวอย่าง จากนั้นเลือกแท็บเครื่องมือที่ต้องการ

### 2. กำหนดช่วงเวลา (ใช้ร่วมกันทุกเครื่องมือ)

พิมพ์เวลาเริ่มต้นและสิ้นสุดในรูปแบบ นาที:วินาที เช่น `1:23.5` หรือเลื่อนแถบสไลเดอร์ หากต้องการความแม่นยำ ให้เล่นวิดีโอไปยังตำแหน่งที่ต้องการแล้วกด "ใช้ตำแหน่งที่เล่นอยู่" และใช้ปุ่ม −0.1 / +0.1 วินาทีปรับละเอียด จากนั้นกด "ดูตัวอย่างช่วงที่เลือก" เพื่อตรวจสอบก่อนเริ่มประมวลผล

### วิธีตัดวิดีโอ

1. เลือกแท็บ **ตัดวิดีโอ** แล้วกำหนดช่วงเวลา
2. เลือกโหมด **เก็บช่วงที่เลือก** เพื่อตัดหัวท้ายที่ไม่ต้องการออก หรือ **ลบช่วงที่เลือก** เพื่อตัดส่วนกลางออกแล้วต่อส่วนหน้ากับส่วนหลังเข้าด้วยกัน
3. ปรับความเร็ว เฟดเข้า/ออก คุณภาพ และรูปแบบไฟล์ตามต้องการ (ไม่ปรับก็ได้)
4. กด **เริ่มตัดวิดีโอ** รอจนแถบความคืบหน้าเต็ม แล้วกดดาวน์โหลด

### วิธีบีบอัดวิดีโอให้ได้ขนาดตามต้องการ

1. เลือกแท็บ **บีบอัดวิดีโอ**
2. ถ้าต้องการให้ไฟล์ไม่เกินขนาดที่กำหนด ให้กดปุ่มตั้งเป้าหมายด่วน (อีเมล, WhatsApp, Discord) หรือเลือก **ตามขนาดไฟล์เป้าหมาย** แล้วกรอกจำนวน MB ถ้าต้องการควบคุมเอง ให้เลือก **ตามระดับคุณภาพ**
3. ลดความละเอียดหรือเฟรมเรตเพื่อให้ไฟล์เล็กลงอีก และดูตัวเลข "ประมาณการ" ที่อัปเดตทันทีก่อนเริ่ม
4. กด **เริ่มบีบอัดวิดีโอ** เมื่อเสร็จ ระบบจะแสดงขนาดก่อนและหลังบีบอัด ถ้าไฟล์ยังใหญ่กว่าเป้าหมายเล็กน้อย ให้กด **ลองอีกครั้งให้เล็กลง** เพื่อให้ระบบปรับบิตเรตเอง

### วิธีแปลงวิดีโอเป็น GIF

1. เลือกแท็บ **วิดีโอเป็น GIF** และกำหนดช่วงสั้น ๆ ที่ต้องการ (แนะนำไม่เกิน 10 วินาที)
2. เลือกความลื่นไหล (fps) ความกว้าง จำนวนสี และการวนลูป ยิ่งกว้าง สีเยอะ และเฟรมเรตสูง ไฟล์ยิ่งใหญ่
3. เปิดตัวเลือก **เก็บเฉพาะส่วนที่เปลี่ยนในแต่ละเฟรม** เพื่อลดขนาดไฟล์ เหมาะกับคลิปที่พื้นหลังนิ่ง เช่น การบันทึกหน้าจอ
4. กด **สร้าง GIF** แล้วดาวน์โหลดผลลัพธ์

### ทำงานต่อเนื่อง

หลังประมวลผลเสร็จ กด **ใช้ไฟล์นี้ทำต่อ** เพื่อนำผลลัพธ์กลับมาเป็นไฟล์ตั้งต้น เช่น ตัดให้สั้นก่อน แล้วบีบอัดต่อ หรือตัดเป็นช่วงสั้น ๆ ก่อนแปลงเป็น GIF

| ตัดวิดีโอ (Trim) | เก็บเฉพาะช่วงเวลาที่ต้องการ และตัดส่วนหัวหรือท้ายที่ไม่ใช้ออก |
| ตัดส่วนกลาง | ลบช่วงตรงกลางของวิดีโอออก แล้วต่อส่วนที่เหลือเข้าด้วยกัน |
| บีบอัดวิดีโอ (Compress) | ทำให้ไฟล์เล็กลงโดยลดบิตเรต ความละเอียด หรือเฟรมเรต |
| ขนาดไฟล์เป้าหมาย | ขนาดสูงสุดที่ต้องการ เครื่องมือคำนวณบิตเรตให้พอดี ผลจริงอาจคลาดเคลื่อนเล็กน้อย |
| บิตเรต (Bitrate) | ปริมาณข้อมูลต่อวินาที หน่วย kbps หรือ Mbps ยิ่งสูงภาพยิ่งคมแต่ไฟล์ใหญ่ |
| ความละเอียด (Resolution) | จำนวนพิกเซลของภาพ เช่น 1080p, 720p, 480p ยิ่งต่ำไฟล์ยิ่งเล็ก |
| เฟรมเรต (fps) | จำนวนภาพต่อวินาที 30 fps ลื่นไหลทั่วไป การลดเหลือ 15–24 fps ช่วยลดขนาดไฟล์ |
| ระดับคุณภาพ | ค่าตั้งล่วงหน้าของบิตเรตต่อพิกเซล เลือกตั้งแต่สูงไปถึงต่ำสุด |
| โคเดก (Codec) | วิธีบีบอัดและถอดรหัสภาพ/เสียง เช่น H.264, VP9, VP8 |
| H.264 | โคเดกที่เล่นได้เกือบทุกอุปกรณ์ ใช้กับไฟล์ MP4 |
| VP9 / VP8 | โคเดกแบบเปิดที่ใช้กับไฟล์ WebM โดย VP9 ให้ไฟล์เล็กกว่า VP8 |
| MP4 / WebM | รูปแบบไฟล์ (คอนเทนเนอร์) ที่ห่อภาพและเสียงไว้ด้วยกัน MP4 เข้ากันได้ดีที่สุด |
| เฟดเข้า / เฟดออก | ภาพและเสียงค่อย ๆ ปรากฏตอนเริ่มและค่อย ๆ จางหายตอนจบ |
| ความเร็ว | 0.5x คือเล่นช้าลงครึ่งหนึ่ง 2x คือเล่นเร็วขึ้นเท่าตัว |
| บิตเรตเสียง | ปริมาณข้อมูลของเสียงต่อวินาที ยิ่งต่ำไฟล์ยิ่งเล็กแต่เสียงยิ่งด้อยลง |
| GIF | ภาพเคลื่อนไหวแบบไม่มีเสียง เล่นวนซ้ำได้ เหมาะกับคลิปสั้น |
| จำนวนสี (พาเลตต์) | GIF ใช้สีได้สูงสุด 256 สี การลดเหลือ 128 หรือ 64 สีช่วยลดขนาดไฟล์ |
| ลดแถบสี (Dither) | การกระจายจุดสีเพื่อให้การไล่เฉดดูเนียนขึ้นเมื่อมีสีจำกัด แต่ไฟล์จะใหญ่ขึ้น |
| การวนลูป | จำนวนรอบที่ GIF เล่นซ้ำ ตั้งเป็นไม่จำกัดหรือเล่นครั้งเดียวได้ |
| เก็บเฉพาะส่วนที่เปลี่ยน | บันทึกเฉพาะบริเวณที่เปลี่ยนในแต่ละเฟรมของ GIF แทนการเก็บภาพเต็มทุกเฟรม |
| เข้ารหัสใหม่ (Re-encode) | การสร้างวิดีโอใหม่จากภาพต้นฉบับ ทำให้ใช้เวลานานกว่าการคัดลอกไฟล์ |
| ประมวลผลในเครื่อง | งานทั้งหมดทำในเบราว์เซอร์ของคุณ ไฟล์ไม่ถูกส่งไปยังเซิร์ฟเวอร์ |

## คำถามที่พบบ่อย

### ไฟล์วิดีโอของฉันถูกอัปโหลดไปที่เซิร์ฟเวอร์หรือไม่

ไม่ ทุกขั้นตอนทำงานในเบราว์เซอร์ของคุณ ตรวจสอบเองได้โดยเปิดแท็บ Network ในเครื่องมือนักพัฒนาของเบราว์เซอร์ขณะประมวลผล จะไม่พบการส่งไฟล์วิดีโอออกไป

### ทำไมการตัดและการบีบอัดใช้เวลาประมาณเท่าความยาวคลิป

เครื่องมือนี้เข้ารหัสวิดีโอใหม่ขณะเล่นวิดีโอในเบราว์เซอร์ จึงใช้เวลาใกล้เคียงกับความยาวช่วงที่เลือก ถ้าต้องการให้เร็วขึ้นในโหมดตัดวิดีโอ ให้เลือกความเร็ว 2x หรือเลือกเฉพาะช่วงที่ต้องการ และควรเปิดแท็บนี้ค้างไว้จนเสร็จ

### ทำไมได้ไฟล์ WebM แทน MP4

รูปแบบไฟล์ที่บันทึกได้ขึ้นอยู่กับเบราว์เซอร์ Chrome และ Edge รุ่นใหม่ รวมถึง Safari มักบันทึกเป็น MP4 (H.264) ได้ ส่วนบางเบราว์เซอร์รองรับเฉพาะ WebM ลองเปลี่ยนรูปแบบไฟล์ในตัวเลือก หรือใช้เบราว์เซอร์รุ่นล่าสุด

### จะทำให้วิดีโอเล็กพอส่งอีเมลหรือแชทได้อย่างไร

อีเมลส่วนใหญ่จำกัดไฟล์แนบที่ราว 25 MB และแอปแชทแต่ละแอปมีเพดานต่างกัน ซึ่งอาจเปลี่ยนแปลงได้ ควรตรวจสอบเงื่อนไขล่าสุดของแอปที่ใช้ แล้วเลือกขนาดเป้าหมายให้ต่ำกว่าเพดานเล็กน้อย หากยังใหญ่เกิน ให้ลดความละเอียดเป็น 720p หรือ 480p และลดเฟรมเรตเป็น 24 หรือ 15 fps

### ขนาดไฟล์ไม่ตรงกับเป้าหมายเป๊ะ ๆ ใช่หรือไม่

ใช่ เบราว์เซอร์ควบคุมบิตเรตเป็นค่าประมาณ ระบบจึงเผื่อขนาดไว้แล้ว หากผลลัพธ์ยังเกินเป้าหมาย ปุ่ม "ลองอีกครั้งให้เล็กลง" จะปรับค่าให้อัตโนมัติ

### เปิดไฟล์ MKV, AVI หรือ HEVC ไม่ได้

เครื่องมือนี้ใช้ตัวถอดรหัสของเบราว์เซอร์ ไฟล์บางรูปแบบหรือบางโคเดกจึงอาจเปิดไม่ได้ ลองใช้ Chrome หรือ Safari รุ่นล่าสุด หรือแปลงไฟล์เป็น MP4 (H.264) ก่อน

### ทำไมไฟล์ GIF ถึงใหญ่มาก

GIF เก็บข้อมูลเป็นภาพทีละเฟรมและจำกัด 256 สี คลิปยาว กว้าง และเฟรมเรตสูงจะทำให้ไฟล์ใหญ่เร็ว ควรทำ GIF ให้ยาวไม่กี่วินาที ลดความกว้างเหลือ 320–480 px ลดจำนวนสี และเปิดตัวเลือกเก็บเฉพาะส่วนที่เปลี่ยน

### ใช้บนมือถือได้หรือไม่

ใช้ได้บน iPhone และ Android ที่ใช้เบราว์เซอร์รุ่นใหม่ แต่ไฟล์ขนาดใหญ่หรือความละเอียดสูงอาจใช้หน่วยความจำมาก ควรเปิดหน้าจอและแท็บนี้ค้างไว้ขณะประมวลผล

### ผลลัพธ์ตัดวิดีโอเป็นกี่ fps

โหมดตัดวิดีโอบันทึกผลลัพธ์สูงสุด 30 fps หากวิดีโอต้นฉบับเป็น 60 fps ผลลัพธ์จะลดเหลือ 30 fps
