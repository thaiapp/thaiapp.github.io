  function fmtSize(b) {
    if (b >= 1073741824) return (b / 1073741824).toFixed(2) + ' GB';

  function fmtTime(s) {
    s = Math.max(0, Math.round(s));

  function plan(it, o) {
    var w = it.w, h = it.h, dur = it.dur;
    if (o.h && h > o.h) { w = Math.round(w * o.h / h); h = o.h; }
    w = Math.max(2, w - (w % 2));
    h = Math.max(2, h - (h % 2));
    var ab = o.mute ? 0 : 96000, vb;
    if (o.mode === 'size') vb = (o.mb * 1048576 * 8 * 0.9) / dur - ab;
    else vb = BPP[o.q] * w * h * o.fps;
    var orig = (it.file.size * 8) / dur;
    vb = Math.max(60000, Math.min(vb, Math.max(orig * 0.95, 60000)));
    return {
      w: w, h: h, fps: o.fps, vb: vb, ab: ab,
      est: ((vb + ab) * dur) / 8,
      skip: o.mode === 'size' && it.file.size <= o.mb * 1048576
    };
  }

  function probe(file) {

        done();
        resolve(r);
      };
      v.onerror = function () { done(); reject(new Error('probe')); };
      v.src = u;
    });
  }

  function updateItem(it) {
    var e = it.el;
    if (!e) return;
    var m = [];
    if (it.w) m.push(it.w + '×' + it.h);
    if (it.dur) m.push(fmtTime(it.dur));
    m.push(fmtSize(it.file.size));
    if (it.est && (it.status === 'wait' || it.status === 'cancel')) m.push('ประมาณ ' + fmtSize(it.est) + ' หลังบีบอัด');
    e.meta.textContent = m.join(' · ');
    e.fill.style.width = (it.status === 'done' || it.status === 'kept' ? 100 : Math.round((it.progress || 0) * 100)) + '%';
    var s = '';
    if (it.status === 'wait') s = it.warn || 'รอประมวลผล';
    else if (it.status === 'run') s = 'กำลังบีบอัด ' + Math.round((it.progress || 0) * 100) + '%' + (it.eta ? ' · เหลือประมาณ ' + fmtTime(it.eta) : '');

  function addFiles(fl) {

  async function compressOne(it, o) {
    var p = plan(it, o);
    it.status = 'run'; it.progress = 0; it.eta = null; it.err = ''; it.out = null; it.kept = false;
    if (it.previewUrl) { URL.revokeObjectURL(it.previewUrl); it.previewUrl = null; }
    while (it.el.act.firstChild) it.el.act.removeChild(it.el.act.firstChild);
    updateItem(it);
    if (p.skip) {
      it.status = 'kept'; it.kept = true; it.out = it.file; it.ext = o.codec.ext;
      updateItem(it); showActions(it);
      return;
    }
    var url = URL.createObjectURL(it.file);
    var v = document.createElement('video');
    var ac = null, rec = null, mixed = null, running = true, timer = null, finish, fail;
    var onVis = function () {
      if (document.hidden) {
        try { v.pause(); } catch (e) { /* ignore */ }
        if (rec && rec.state === 'recording') rec.pause();
      } else {
        if (rec && rec.state === 'paused') rec.resume();
        if (!v.ended) v.play().catch(function () { /* ignore */ });
      }
    };
    var cleanup = function () {
      running = false;
      clearInterval(timer);
      document.removeEventListener('visibilitychange', onVis);
      try { v.pause(); } catch (e) { /* ignore */ }
      try { if (rec && rec.state !== 'inactive') rec.stop(); } catch (e) { /* ignore */ }
      try { if (mixed) mixed.getTracks().forEach(function (t) { t.stop(); }); } catch (e) { /* ignore */ }
      try { if (ac) ac.close(); } catch (e) { /* ignore */ }
      URL.revokeObjectURL(url);
      v.removeAttribute('src');
      try { v.load(); } catch (e) { /* ignore */ }
      active = null;
    };
    try {
      v.preload = 'auto';
      v.playsInline = true;
      v.src = url;
      await new Promise(function (res, rej) {
        v.onloadeddata = res;
        v.onerror = function () { rej(new Error('เบราว์เซอร์เปิดไฟล์นี้ไม่ได้ ลองแปลงเป็น MP4 (H.264) ก่อน')); };
      });
      var c = document.createElement('canvas');
      c.width = p.w; c.height = p.h;
      var ctx = c.getContext('2d', { alpha: false });
      var tracks = c.captureStream(p.fps).getVideoTracks();
      if (!o.mute) {
        try {
          var AC = window.AudioContext || window.webkitAudioContext;
          ac = new AC();
          var src = ac.createMediaElementSource(v);
          var dst = ac.createMediaStreamDestination();
          src.connect(dst);
          tracks = tracks.concat(dst.stream.getAudioTracks());
          if (ac.state === 'suspended') ac.resume();
        } catch (e) { ac = null; }
      }
      if (o.mute || !ac) v.muted = true;
      mixed = new MediaStream(tracks);
      var ropt = { mimeType: o.codec.mime, videoBitsPerSecond: Math.round(p.vb) };
      if (p.ab && ac) ropt.audioBitsPerSecond = p.ab;
      rec = new MediaRecorder(mixed, ropt);
      var chunks = [];
      rec.ondataavailable = function (e) { if (e.data && e.data.size) chunks.push(e.data); };
      var stopped = new Promise(function (res, rej) {
        rec.onstop = res;
        rec.onerror = function () { rej(new Error('การบันทึกวิดีโอผิดพลาด')); };
      });
      var ended = new Promise(function (res, rej) { finish = res; fail = rej; });
      v.onended = function () { finish(); };
      active = { stop: function () { finish(); } };
      document.addEventListener('visibilitychange', onVis);

      var t0 = Date.now(), lastT = -1, stall = 0;
      var draw = function () {
        if (!running) return;
        ctx.drawImage(v, 0, 0, p.w, p.h);
        var pr = Math.min(1, v.currentTime / it.dur), el = (Date.now() - t0) / 1000;
        it.progress = pr;
        it.eta = pr > 0.03 ? el / pr - el : null;
        updateItem(it);
        if (v.requestVideoFrameCallback) v.requestVideoFrameCallback(draw);
        else requestAnimationFrame(draw);
      };
      timer = setInterval(function () {
        if (document.hidden || v.ended) return;
        if (v.currentTime === lastT) {
          stall++;
          if (stall >= 30) fail(new Error('วิดีโอหยุดค้าง ลองบีบอัดใหม่อีกครั้ง'));
        } else { stall = 0; lastT = v.currentTime; }
      }, 1000);

      ctx.drawImage(v, 0, 0, p.w, p.h);
      rec.start(1000);
      draw();
      try { await v.play(); } catch (e) { throw new Error('เล่นวิดีโอเพื่อบีบอัดไม่ได้ ลองกดปุ่มบีบอัดอีกครั้ง'); }
      await ended;
      running = false;
      try { ctx.drawImage(v, 0, 0, p.w, p.h); } catch (e) { /* ignore */ }
      await new Promise(function (r) { setTimeout(r, 250); });
      if (rec.state !== 'inactive') rec.stop();
      await stopped;
      if (cancelled) throw new Error('cancelled');
      var blob = new Blob(chunks, { type: o.codec.mime.split(';')[0] });
      if (!blob.size) throw new Error('ไม่ได้ข้อมูลวิดีโอ ลองเปลี่ยนรูปแบบ/Codec');
      it.ext = o.codec.ext;
      if (blob.size >= it.file.size) {
        it.status = 'kept'; it.kept = true; it.out = it.file;
      } else {
        it.status = 'done'; it.out = blob;
      }
      it.progress = 1;
      updateItem(it); showActions(it);
    } finally {
      cleanup();
    }
  }

  var crcTable = (function () {
    var t = [], c, n, k;
    for (n = 0; n < 256; n++) {
      c = n;
      for (k = 0; k < 8; k++) c = (c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1);
      t[n] = c >>> 0;
    }
    return t;
  })();
  function crc32(u8) {
    var c = 0xFFFFFFFF;
    for (var i = 0; i < u8.length; i++) c = crcTable[(c ^ u8[i]) & 255] ^ (c >>> 8);
    return (c ^ 0xFFFFFFFF) >>> 0;
  }
  async function buildZip(entries) {
    var enc = new TextEncoder(), parts = [], cen = [], off = 0, d = new Date();
    var dt = ((d.getHours() << 11) | (d.getMinutes() << 5) | (d.getSeconds() >> 1)) & 0xFFFF;
    var dd = (((d.getFullYear() - 1980) << 9) | ((d.getMonth() + 1) << 5) | d.getDate()) & 0xFFFF;
    var csize = 0;
    for (var i = 0; i < entries.length; i++) {
      var data = new Uint8Array(await entries[i].blob.arrayBuffer());
      var nm = enc.encode(entries[i].name), crc = crc32(data);
      var lh = new DataView(new ArrayBuffer(30));
      lh.setUint32(0, 0x04034b50, true); lh.setUint16(4, 20, true); lh.setUint16(6, 0x0800, true);
      lh.setUint16(8, 0, true); lh.setUint16(10, dt, true); lh.setUint16(12, dd, true);
      lh.setUint32(14, crc, true); lh.setUint32(18, data.length, true); lh.setUint32(22, data.length, true);
      lh.setUint16(26, nm.length, true); lh.setUint16(28, 0, true);
      parts.push(lh.buffer, nm, data);
      var ch = new DataView(new ArrayBuffer(46));
      ch.setUint32(0, 0x02014b50, true); ch.setUint16(4, 20, true); ch.setUint16(6, 20, true);
      ch.setUint16(8, 0x0800, true); ch.setUint16(10, 0, true); ch.setUint16(12, dt, true); ch.setUint16(14, dd, true);
      ch.setUint32(16, crc, true); ch.setUint32(20, data.length, true); ch.setUint32(24, data.length, true);
      ch.setUint16(28, nm.length, true); ch.setUint16(30, 0, true); ch.setUint16(32, 0, true);
      ch.setUint16(34, 0, true); ch.setUint16(36, 0, true); ch.setUint32(38, 0, true); ch.setUint32(42, off, true);
      cen.push(ch.buffer, nm);
      csize += 46 + nm.length;
      off += 30 + nm.length + data.length;
    }
    var end = new DataView(new ArrayBuffer(22));
    end.setUint32(0, 0x06054b50, true); end.setUint16(4, 0, true); end.setUint16(6, 0, true);
    end.setUint16(8, entries.length, true); end.setUint16(10, entries.length, true);
    end.setUint32(12, csize, true); end.setUint32(16, off, true); end.setUint16(20, 0, true);
    return new Blob(parts.concat(cen, [end.buffer]), { type: 'application/zip' });
  }

## วิธีใช้งานเครื่องมือบีบอัดวิดีโอ

1. **เพิ่มวิดีโอ** — ลากไฟล์มาวางในกรอบด้านบน คลิกเพื่อเลือกไฟล์ หรือกด `Ctrl`+`V` เพื่อวางไฟล์ที่คัดลอกไว้ เพิ่มได้หลายไฟล์พร้อมกัน
2. **เลือกวิธีลดขนาด** — กดพรีเซ็ตสำหรับอีเมล WhatsApp หรือ Discord เพื่อให้ไฟล์พอดีกับขีดจำกัด หรือเลือก **เลือกตามคุณภาพ** แล้วปรับระดับคุณภาพเอง
3. **กำหนดขนาดไฟล์เป้าหมาย** (ถ้าต้องการ) — เลือก **กำหนดขนาดไฟล์เป้าหมาย** แล้วพิมพ์ขนาดเป็น MB ระบบจะคำนวณบิตเรตให้ใกล้เคียงขนาดนั้น
4. **ปรับความละเอียดและเฟรมเรต** — การลดความละเอียดเป็น 720p หรือ 480p ช่วยลดขนาดได้มากที่สุด ส่วนการลด FPS ช่วยลดได้เพิ่มอีกเล็กน้อย
5. **เลือกรูปแบบ / Codec** — เลือก MP4 เพื่อให้เล่นได้ทุกอุปกรณ์ หรือ WebM หากเบราว์เซอร์ไม่มีตัวเลือก MP4 และติ๊ก **ตัดเสียงออก** หากไม่ต้องการเสียง
6. **กด "บีบอัดทั้งหมด"** — ดูความคืบหน้าและเวลาที่เหลือของแต่ละไฟล์ กด **ยกเลิก** ได้ทุกเมื่อ
7. **ตรวจผลลัพธ์และบันทึก** — กด **ดูตัวอย่าง** เพื่อเช็กภาพก่อน แล้วกด **บันทึก** ทีละไฟล์ หรือ **ดาวน์โหลด ZIP** เพื่อรับทุกไฟล์พร้อมกัน

หากผลลัพธ์ยังใหญ่เกินไป ให้ลดความละเอียดลงอีกขั้นหรือกำหนดขนาดเป้าหมายให้เล็กลงแล้วบีบอัดใหม่ ระบบจะเก็บไฟล์ต้นฉบับไว้ให้แทน หากผลลัพธ์ไม่เล็กกว่าต้นฉบับ

## ความหมายของคำศัพท์ที่ใช้ในเครื่องมือ

| คำศัพท์ | ความหมายโดยย่อ |
| --- | --- |
| บีบอัดวิดีโอ | การเข้ารหัสวิดีโอใหม่ให้ไฟล์เล็กลง โดยแลกกับความคมชัดที่ลดลงเล็กน้อย |
| คุณภาพ | ระดับรายละเอียดของภาพ ยิ่งสูงไฟล์ยิ่งใหญ่ ยิ่งต่ำไฟล์ยิ่งเล็กแต่ภาพอาจเบลอหรือเป็นบล็อก |
| ขนาดไฟล์เป้าหมาย | ขนาดสูงสุดที่ต้องการ (MB) ระบบใช้ค่านี้คำนวณบิตเรตจากความยาววิดีโอ ผลลัพธ์จริงอาจคลาดเคลื่อนเล็กน้อย |
| ความละเอียด | จำนวนพิกเซลของภาพ เช่น 1080p, 720p, 480p ยิ่งต่ำไฟล์ยิ่งเล็ก |
| FPS (เฟรมเรต) | จำนวนภาพต่อวินาที 30 FPS เหมาะกับวิดีโอทั่วไป 60 FPS ลื่นกว่าแต่ไฟล์ใหญ่กว่า |
| บิตเรต (Bitrate) | ปริมาณข้อมูลต่อวินาทีของวิดีโอ เป็นตัวกำหนดขนาดไฟล์โดยตรง |
| Codec | วิธีเข้ารหัสและถอดรหัสวิดีโอ เช่น H.264, VP8, VP9, AV1 |
| H.264 | Codec ที่เล่นได้เกือบทุกอุปกรณ์ เหมาะกับไฟล์ MP4 |
| VP8 / VP9 / AV1 | Codec ของไฟล์ WebM VP9 และ AV1 บีบอัดได้ดีกว่า แต่บางโปรแกรมหรืออุปกรณ์เก่าอาจเล่นไม่ได้ |
| MP4 | รูปแบบไฟล์วิดีโอที่ใช้กันมากที่สุด เหมาะกับการส่งและเผยแพร่ |
| WebM | รูปแบบไฟล์วิดีโอสำหรับเว็บ เล่นได้ในเบราว์เซอร์สมัยใหม่ |
| ตัดเสียงออก | ลบแทร็กเสียงเพื่อให้ไฟล์เล็กลงอีก เหมาะกับวิดีโอที่ไม่ต้องการเสียง |
| ประมวลผลบนอุปกรณ์ | การทำงานทั้งหมดเกิดในเบราว์เซอร์ของคุณ ไฟล์ไม่ถูกส่งไปยังเซิร์ฟเวอร์ |
| ZIP | ไฟล์รวมหลายไฟล์ไว้ด้วยกัน ใช้ดาวน์โหลดผลลัพธ์ทั้งหมดในครั้งเดียว |

## ข้อควรรู้ก่อนใช้งาน

- เครื่องมือนี้เล่นวิดีโอและบันทึกใหม่ในเบราว์เซอร์ จึงใช้เวลาประมาณเท่ากับความยาววิดีโอ
- ควรเปิดแท็บนี้ค้างไว้และไม่ล็อกหน้าจอระหว่างบีบอัด เพราะการสลับแอปอาจทำให้หยุดชั่วคราว
- รูปแบบไฟล์ที่เปิดได้ขึ้นอยู่กับเบราว์เซอร์ MP4 (H.264), MOV และ WebM รองรับดีที่สุด ส่วน MKV และ AVI อาจเปิดไม่ได้
- ขีดจำกัดขนาดไฟล์ของแอปต่าง ๆ อาจเปลี่ยนแปลงได้ โปรดตรวจสอบก่อนส่ง
- หากต้องการไฟล์สั้นลง ลองตัดวิดีโอให้สั้นก่อนบีบอัด

## คำถามที่พบบ่อย

### วิดีโอของฉันถูกอัปโหลดไปที่เซิร์ฟเวอร์หรือไม่

ไม่ การบีบอัดทำงานในเบราว์เซอร์บนอุปกรณ์ของคุณทั้งหมด ไฟล์จึงไม่ออกจากเครื่องและไม่มีลายน้ำ

### บีบอัดแล้วคุณภาพวิดีโอจะลดลงไหม

ภาพจะคมชัดน้อยลงเล็กน้อยตามระดับที่เลือก โหมดคุณภาพ "ดี" ที่ความละเอียด 720p เป็นจุดสมดุลที่เหมาะกับการส่งต่อส่วนใหญ่ และดูตัวอย่างก่อนบันทึกได้เสมอ

### ทำไมการบีบอัดใช้เวลานานเท่าความยาววิดีโอ

เพราะเบราว์เซอร์ต้องเล่นวิดีโอไปพร้อมกับบันทึกใหม่ วิดีโอสั้นจึงเสร็จเร็ว ส่วนวิดีโอยาวต้องรอนานกว่า

### ทำอย่างไรให้วิดีโอเล็กพอส่งทาง WhatsApp อีเมล หรือ Discord

กดพรีเซ็ตของแพลตฟอร์มนั้น แล้วกดบีบอัด หากยังใหญ่เกินไป ให้ลดขนาดเป้าหมายหรือเลือกความละเอียดที่ต่ำลง

### ทำไมระบบเก็บไฟล์ต้นฉบับไว้แทนผลลัพธ์

หากไฟล์ต้นฉบับถูกบีบอัดมาอย่างดีอยู่แล้ว การเข้ารหัสใหม่อาจได้ไฟล์ที่ใหญ่กว่าเดิม ระบบจึงเก็บต้นฉบับไว้เพื่อไม่ให้ขนาดเพิ่มขึ้น

### ข้อมูลเมตา เช่น ตำแหน่ง GPS ในวิดีโอเดิมจะยังอยู่ไหม

โดยทั่วไปไฟล์ที่ได้เป็นการเข้ารหัสใหม่ จึงไม่มีข้อมูลเมตาเดิมติดไปด้วย แต่หากต้องการความมั่นใจ โปรดตรวจสอบไฟล์ก่อนเผยแพร่
