function search(raw) {
var q = raw.trim().toLowerCase();
var out = [];
var got = {};
function add(s) { if (!has(got, s)) { got[s] = true; out.push(s); } }
var rawChars = Array.from(raw.trim());
rawChars.forEach(function (ch) { if (has(HOME, ch) && (ch.charCodeAt(0) > 127 || rawChars.length === 1)) { add(ch); } });
if (/^(u\+|0x|&#x)/.test(q) || /^[0-9a-f]{4,5}$/.test(q)) {
var m = q.replace(/^u\+/, '').replace(/^0x/, '').replace(/^&#x/, '').replace(/;$/, '').replace(/^0+/, '');
if (/^[0-9a-f]{2,6}$/.test(m)) {
ALL.forEach(function (s) { if (cpsOf(s)[0].toString(16).indexOf(m) === 0) { add(s); } });
}
}
var parts = q.split(/\s+/).filter(function (x) { return x !== ''; });
if (parts.length === 0) { return out; }
function every(h) { return parts.every(function (p) { return h.indexOf(p) !== -1; }); }
ALL.forEach(function (s) { if (has(NM, s) && every(NM[s].toLowerCase())) { add(s); } });
CATS.forEach(function (c) {
if (every((c.n + ' ' + c.k).toLowerCase())) { c.list.forEach(function (s) { add(s); }); }
});
return out;
}

## วิธีใช้งานเครื่องมือสัญลักษณ์พิเศษ

1. เลือกหมวดจากแท็บด้านบน เช่น ยอดนิยม หัวใจ ลูกศร ดาว หรือพิมพ์คำในช่องค้นหา เช่น “หัวใจ”, “arrow”, “฿” หรือรหัส “U+2764”
2. คลิกหรือแตะที่สัญลักษณ์ที่ต้องการ ระบบจะคัดลอกไปที่คลิปบอร์ดทันทีพร้อมขึ้นข้อความ “คัดลอก ... แล้ว”
3. ไปที่แอปที่ต้องการ แล้ววาง (Ctrl + V บนคอมพิวเตอร์ หรือ Command + V บน Mac หรือกดค้างแล้วเลือก “วาง” บนมือถือ)
4. หากต้องการหลายสัญลักษณ์ในข้อความเดียว ให้ติ๊ก “โหมดรวมหลายสัญลักษณ์” แล้วคลิกสัญลักษณ์ต่อกันไปเรื่อยๆ สัญลักษณ์จะเรียงอยู่ในกล่องรวมสัญลักษณ์ ใช้ปุ่ม “เว้นวรรค” “ย้อนกลับ” “ล้างกล่อง” ช่วยจัดข้อความ แล้วกด “คัดลอกทั้งหมด”
5. ดูชื่อและรหัสของสัญลักษณ์ได้ในกล่องรายละเอียดที่ขึ้นหลังจากคลิก มีรหัส Unicode, HTML, CSS และ JavaScript ให้กด “คัดลอก” ข้างรหัสได้ทันที เหมาะสำหรับนักพัฒนาเว็บและนักออกแบบ
6. หากสัญลักษณ์ที่วางกลายเป็นอีโมจิสีบนมือถือ ให้ติ๊ก “บังคับแสดงแบบข้อความ (ไม่เป็นอีโมจิ)” ก่อนคัดลอก
7. กด “☆ เพิ่มในรายการโปรด” เพื่อเก็บสัญลักษณ์ที่ใช้บ่อย ส่วนแท็บ “ที่เพิ่งใช้” จะบันทึกให้อัตโนมัติ ข้อมูลทั้งสองเก็บในเบราว์เซอร์ของคุณเท่านั้น และกด “ลบข้อมูลทั้งหมด” เพื่อล้างได้ทุกเมื่อ
8. ปรับแถบเลื่อน “ขนาด” เพื่อขยายสัญลักษณ์ให้ดูชัดขึ้นหรือย่อให้เห็นได้มากขึ้นในหน้าจอเดียว

## คำอธิบายคำศัพท์ที่ควรรู้

- **สัญลักษณ์พิเศษ (Special Characters):** ตัวอักษรหรือเครื่องหมายที่ไม่มีบนแป้นพิมพ์ปกติ เช่น ♡ ★ → ฿ แต่เป็นข้อความจริงที่คัดลอกและวางได้เหมือนตัวอักษรทั่วไป
- **คัดลอกวาง (Copy and Paste):** การส่งข้อความไปเก็บไว้ที่คลิปบอร์ดก่อน แล้วนำไปวางในแอปปลายทาง
- **คลิปบอร์ด (Clipboard):** พื้นที่พักข้อมูลชั่วคราวของอุปกรณ์ ที่ใช้เก็บสิ่งที่คัดลอกไว้ก่อนวาง
- **Unicode:** มาตรฐานสากลที่กำหนดหมายเลขให้ตัวอักษรและสัญลักษณ์แทบทุกชนิด ทำให้แสดงผลตรงกันข้ามอุปกรณ์และแพลตฟอร์ม
- **Code point (U+XXXX):** หมายเลขประจำตัวของสัญลักษณ์ใน Unicode เช่น ❤ คือ U+2764 ใช้ค้นหาสัญลักษณ์ในเครื่องมือนี้ได้
- **HTML Entity:** รหัสที่ใช้เขียนสัญลักษณ์ลงในโค้ด HTML เช่น `&#x2764;` (เลขฐานสิบหก) หรือ `&#10084;` (เลขฐานสิบ)
- **CSS Escape:** รหัสสำหรับใส่สัญลักษณ์ในพร็อพเพอร์ตี content ของ CSS เช่น `\2764`
- **JavaScript Escape:** รหัสสำหรับใส่สัญลักษณ์ในสตริงของ JavaScript เช่น `\u2764`
- **อีโมจิ (Emoji):** รูปภาพสีสันสดใสที่แทนอารมณ์หรือวัตถุ ต่างจากสัญลักษณ์ข้อความที่เป็นขาวดำและเปลี่ยนสีตามข้อความ
- **Variation Selector (U+FE0E และ U+FE0F):** อักขระล่องหนที่ต่อท้ายสัญลักษณ์เพื่อกำหนดรูปแบบการแสดงผล FE0E ให้แสดงแบบข้อความ ส่วน FE0F ให้แสดงแบบอีโมจิสี ตัวเลือก “บังคับแสดงแบบข้อความ” ของเครื่องมือนี้จะเติม FE0E ให้
- **Glyph:** รูปร่างที่มองเห็นจริงของตัวอักษรหรือสัญลักษณ์ ซึ่งขึ้นอยู่กับฟอนต์ที่อุปกรณ์ใช้
- **Font Fallback:** กลไกที่ระบบสลับไปใช้ฟอนต์อื่นเมื่อฟอนต์หลักไม่มีสัญลักษณ์นั้น
- **Tofu (กล่องสี่เหลี่ยม):** เครื่องหมายสี่เหลี่ยมว่างที่ปรากฏแทนสัญลักษณ์เมื่ออุปกรณ์ไม่มีฟอนต์รองรับ
- **โหมดรวมหลายสัญลักษณ์:** โหมดที่คลิกสัญลักษณ์แล้วนำไปเรียงในกล่องข้อความก่อน แทนการคัดลอกทีละตัว
- **ที่เพิ่งใช้ และ รายการโปรด:** รายการสัญลักษณ์ที่บันทึกไว้ในเบราว์เซอร์ของคุณ (localStorage) ไม่ถูกส่งไปที่เซิร์ฟเวอร์
- **Alt Code:** วิธีพิมพ์สัญลักษณ์บน Windows ด้วยการกด Alt ค้างแล้วพิมพ์ตัวเลขบนแป้นตัวเลข (Numpad)
- **ไม้ยมก (ๆ), ไปยาลน้อย (ฯ), ฟองมัน (๏), โคมูตร (๛):** อักขระพิเศษของภาษาไทยที่รวมอยู่ในหมวด “อักขระภาษาไทย”

## ใช้สัญลักษณ์พิเศษบน Facebook, Instagram, LINE, TikTok และเกม

- **Facebook และ Instagram:** วางได้ในชื่อโปรไฟล์ ไบโอ แคปชัน และคอมเมนต์ ควรดูตัวอย่างก่อนกดบันทึก เพราะอุปกรณ์เก่าอาจแสดงสัญลักษณ์ที่พบน้อยเป็นกล่องสี่เหลี่ยม
- **LINE:** วางได้ในชื่อที่แสดง ข้อความสถานะ และแชต สัญลักษณ์ในหมวดยอดนิยมมักแสดงผลได้ตรงกันบน iOS และ Android
- **TikTok และ YouTube:** ใช้ตกแต่งชื่อช่อง คำอธิบายวิดีโอ และแคปชัน แนะนำให้เลือกสัญลักษณ์ที่นิยมเพื่อให้ผู้ชมเห็นตรงกันทุกเครื่อง
- **เกม (เช่น ROV, Free Fire, PUBG Mobile, Roblox):** แต่ละเกมมีรายการอักขระที่อนุญาตไม่เหมือนกันและอาจเปลี่ยนแปลงได้ หากชื่อถูกปฏิเสธ ให้ลองลดจำนวนสัญลักษณ์หรือสลับไปใช้ตัวที่พบบ่อยกว่า
- **Microsoft Word, Google Docs และอีเมล:** วางได้ตามปกติ ส่วนการใส่ลงในโค้ดเว็บ ให้ใช้ HTML Entity หรือรหัสจากกล่องรายละเอียดเพื่อให้แสดงผลถูกต้องทุกเบราว์เซอร์

## วิธีพิมพ์สัญลักษณ์ด้วยแป้นพิมพ์ในอุปกรณ์ต่างๆ

- **Windows:** กด Win + จุด (.) หรือ Win + เซมิโคลอน (;) เพื่อเปิดแผงอีโมจิและสัญลักษณ์ หรือใช้ Alt Code เช่น กด Alt ค้างแล้วพิมพ์ 3 บน Numpad จะได้ ♥ (ใช้ได้ในหลายโปรแกรม)
- **Microsoft Word:** พิมพ์รหัสเลขฐานสิบหก เช่น 2764 แล้วกด Alt + X เพื่อเปลี่ยนเป็น ❤
- **Mac:** กด Control + Command + Space เพื่อเปิดตัวแสดงอักขระ (Character Viewer)
- **iPhone และ iPad:** แป้นพิมพ์มีสัญลักษณ์ให้เลือกจำกัด การคัดลอกจากเครื่องมือนี้ช่วยให้เข้าถึงสัญลักษณ์ได้ครบกว่า
- **Android:** แป้นพิมพ์ส่วนใหญ่มีหน้าสัญลักษณ์เพิ่มเติม แต่ชุดสัญลักษณ์ที่ให้มาไม่เท่ากัน การคัดลอกวางจากเครื่องมือนี้จึงสะดวกกว่า

## วิธีแก้ปัญหาที่พบบ่อย

- **เห็นเป็นกล่องสี่เหลี่ยมหรือเครื่องหมายคำถาม:** อุปกรณ์ปลายทางไม่มีฟอนต์ที่รองรับสัญลักษณ์นั้น ให้เลือกสัญลักษณ์จากหมวดยอดนิยม หรืออัปเดตระบบปฏิบัติการและแอป
- **สัญลักษณ์กลายเป็นอีโมจิสี:** ติ๊ก “บังคับแสดงแบบข้อความ (ไม่เป็นอีโมจิ)” แล้วคัดลอกใหม่ สัญลักษณ์บางตัวอาจยังแสดงเป็นอีโมจิในบางแอป
- **กดแล้วคัดลอกไม่สำเร็จ:** เบราว์เซอร์บางตัวจำกัดการเข้าถึงคลิปบอร์ด ให้เปิดโหมดรวมหลายสัญลักษณ์ แล้วกด “คัดลอกทั้งหมด” หรือเลือกข้อความในกล่องแล้วคัดลอกด้วยตนเอง
- **วางแล้วแอปไม่รับหรืออักขระหาย:** แอปหรือเกมนั้นอาจจำกัดอักขระที่ใช้ได้ ลองใช้สัญลักษณ์อื่นหรือลดจำนวนตัวที่วาง
- **ค้นหาแล้วไม่เจอ:** ลองใช้คำสั้นลง เช่น “หัวใจ” แทน “สัญลักษณ์หัวใจสวยๆ” หรือค้นหาด้วยรหัส เช่น “U+2764” หรือ “2764”

## คำถามที่พบบ่อย

### คัดลอกสัญลักษณ์จากหน้านี้แล้วใช้ได้ทุกที่หรือไม่

ส่วนใหญ่ใช้ได้ เพราะเป็นข้อความตามมาตรฐาน Unicode แต่การแสดงผลขึ้นกับฟอนต์ของอุปกรณ์ผู้ดู สัญลักษณ์ที่พบบ่อย เช่น ♡ ★ → ✓ จะแสดงผลได้กว้างกว่าสัญลักษณ์ที่พบน้อย

### เครื่องมือนี้ฟรีและต้องสมัครสมาชิกหรือไม่

ใช้ได้ฟรี ไม่ต้องสมัครสมาชิกและไม่ต้องติดตั้งโปรแกรมใดๆ เปิดหน้านี้ในเบราว์เซอร์ของคุณแล้วใช้งานได้ทันที

### ข้อมูลที่เพิ่งใช้และรายการโปรดถูกเก็บไว้ที่ไหน

เก็บในเบราว์เซอร์ของคุณเอง (localStorage) ไม่ถูกส่งไปที่เซิร์ฟเวอร์ หากล้างข้อมูลเบราว์เซอร์หรือกด “ลบข้อมูลทั้งหมด” รายการเหล่านี้จะหายไป

### รหัส Unicode ใช้ทำอะไร

ใช้ระบุสัญลักษณ์ได้แม่นยำ เช่น ค้นหาด้วย U+2764 ใช้พิมพ์ผ่าน Alt + X ใน Word หรือใช้เขียนสัญลักษณ์ในโค้ด HTML, CSS และ JavaScript โดยไม่ต้องวางตัวอักษรตรงๆ

### ทำไมสัญลักษณ์บางตัวบนมือถือเป็นอีโมจิสี

สัญลักษณ์บางตัว เช่น ☀ ☁ ❤ มีทั้งแบบข้อความและแบบอีโมจิ มือถือจำนวนมากเลือกแสดงเป็นอีโมจิสีโดยอัตโนมัติ ติ๊ก “บังคับแสดงแบบข้อความ” เพื่อเติม Variation Selector U+FE0E ซึ่งช่วยให้แสดงเป็นแบบข้อความในแอปที่รองรับ

### ใช้ตกแต่งชื่อเกมหรือชื่อโปรไฟล์ได้หรือไม่

ได้ ลองใช้หมวด “ตกแต่งชื่อเกม” หรือพิมพ์คำค้นหา เช่น “ชื่อเกม” แต่บางเกมและบางแพลตฟอร์มจำกัดอักขระที่ใช้ได้ จึงควรทดลองวางและดูตัวอย่างก่อนบันทึก
