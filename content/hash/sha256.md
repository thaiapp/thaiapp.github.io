(function () {
  var input = document.getElementById('sha-input');
  var output = document.getElementById('sha-output');
  var bytesLabel = document.getElementById('sha-bytes');
  var statusLabel = document.getElementById('sha-status');
  var copyBtn = document.getElementById('sha-copy');
  var clearBtn = document.getElementById('sha-clear');
  var encoder = new TextEncoder();
  var currentHash = '';
  var token = 0;
  var hintText = 'ป้อนข้อความด้านบนเพื่อสร้างแฮช';
  function toHex(buffer) {
    var bytes = new Uint8Array(buffer);
    var hex = '';
    for (var i = 0; i != bytes.length; i++) {
      hex += ('0' + bytes[i].toString(16)).slice(-2);
    }
    return hex;
  }
  function reset(message) {
    currentHash = '';
    output.textContent = '...';
    statusLabel.textContent = message;
  }
  function update() {
    var text = input.value;
    var data = encoder.encode(text);
    bytesLabel.textContent = data.length + ' ไบต์ · SHA-256';
    if (text === '') {
      token++;
      reset(hintText);
      return;
    }
    if (!window.crypto || !window.crypto.subtle) {
      reset('เบราว์เซอร์นี้ไม่รองรับ SubtleCrypto (ต้องเปิดผ่าน HTTPS)');
      return;
    }
    var current = ++token;
    window.crypto.subtle.digest('SHA-256', data).then(function (buffer) {
      if (current !== token) { return; }
      currentHash = toHex(buffer);
      output.textContent = currentHash;
      statusLabel.textContent = 'ความยาว ' + currentHash.length + ' ตัวอักษร (256 บิต)';
    }, function () {
      reset('เกิดข้อผิดพลาดในการสร้างแฮช');
    });
  }
  function fallbackCopy(value) {
    var area = document.createElement('textarea');
    area.value = value;
    area.style.position = 'fixed';
    area.style.opacity = '0';
    document.body.appendChild(area);
    area.select();
    try { document.execCommand('copy'); } catch (e) {}
    document.body.removeChild(area);
  }
  function flash(label) {
    copyBtn.textContent = label;
    setTimeout(function () { copyBtn.textContent = 'คัดลอก'; }, 1500);
  }
  copyBtn.addEventListener('click', function () {
    if (currentHash === '') { return; }
    if (navigator.clipboard) {
      navigator.clipboard.writeText(currentHash).then(function () {
        flash('คัดลอกแล้ว ✓');
      }, function () {
        fallbackCopy(currentHash);
        flash('คัดลอกแล้ว ✓');
      });
    } else {
      fallbackCopy(currentHash);
      flash('คัดลอกแล้ว ✓');
    }
  });
  clearBtn.addEventListener('click', function () {
    input.value = '';
    update();
    input.focus();
  });
  input.addEventListener('input', update);
  update();

## วิธีใช้งานเครื่องมือสร้างแฮช SHA-256

1. พิมพ์หรือวางข้อความที่ต้องการลงในช่องข้อความด้านบน
2. ค่าแฮช SHA-256 จะแสดงในกล่องผลลัพธ์ทันทีขณะที่พิมพ์ ไม่ต้องกดปุ่มสร้าง
3. ดูจำนวนไบต์ของข้อความเหนือกล่องผลลัพธ์ เพื่อตรวจสอบขนาดข้อมูลที่ถูกแฮช
4. กดปุ่ม "คัดลอก" เพื่อคัดลอกค่าแฮชไปใช้งานต่อ เช่น วางในโค้ดหรือเอกสาร
5. กดปุ่ม "ล้าง" เพื่อลบข้อความและเริ่มใหม่

**เคล็ดลับ:** ข้อความที่ต่างกันเพียงตัวอักษรเดียว (รวมถึงช่องว่างหรือการขึ้นบรรทัดใหม่) จะให้ค่าแฮชที่ต่างกันโดยสิ้นเชิง ข้อความเดิมจะให้ค่าแฮชเดิมเสมอ การประมวลผลทั้งหมดเกิดขึ้นในเบราว์เซอร์ของคุณ ข้อความจึงไม่ถูกส่งไปยังเซิร์ฟเวอร์

## SHA-256 คืออะไร

SHA-256 สร้างข้อมูลสรุปขนาด 256 บิต เขียนเป็นเลขฐานสิบหก 64 ตัวอักษร หากไม่มีเหตุผลเฉพาะในการเลือกอย่างอื่น นี่คืออัลกอริทึมแฮชที่ควรเลือกใช้ เพราะยังไม่มีการโจมตีแบบการชนกันที่ใช้งานได้จริง

SHA-256 ถูกใช้งานในชีวิตประจำวันอยู่เสมอ เช่น การเซ็นใบรับรอง TLS การเชื่อมโยงบล็อกของ Bitcoin การตรวจสอบ JWT และไฟล์ `SHA256SUMS` สำหรับตรวจความถูกต้องของไฟล์ที่ดาวน์โหลด

## คำศัพท์ที่ควรรู้

| คำศัพท์ | คำอธิบายสั้น ๆ |
|---|---|
| แฮช (Hash) | ค่าที่ได้จากการนำข้อมูลเข้าฟังก์ชันทางคณิตศาสตร์ ได้ผลลัพธ์ความยาวคงที่ไม่ว่าข้อมูลจะยาวเท่าใด |
| SHA-256 | อัลกอริทึมแฮชในตระกูล SHA-2 ให้ผลลัพธ์ 256 บิต |
| ไดเจสต์ (Digest) | อีกชื่อหนึ่งของค่าแฮช หมายถึงข้อมูลสรุปที่ได้จากการแฮช |
| เลขฐานสิบหก (Hex) | ระบบตัวเลขที่ใช้ 0-9 และ a-f ใช้แสดงค่าแฮชให้อ่านและคัดลอกง่าย |
| บิต (Bit) | หน่วยข้อมูลที่เล็กที่สุด มีค่า 0 หรือ 1 |
| ไบต์ (Byte) | หน่วยข้อมูลขนาด 8 บิต ข้อความภาษาไทยหนึ่งตัวอักษรใน UTF-8 ใช้ 3 ไบต์ |
| UTF-8 | รูปแบบการเข้ารหัสตัวอักษรที่เครื่องมือนี้ใช้แปลงข้อความเป็นไบต์ก่อนแฮช |
| ฟังก์ชันทางเดียว (One-way) | แปลงข้อมูลเป็นแฮชได้ แต่ย้อนกลับจากแฮชไปเป็นข้อมูลเดิมไม่ได้ |
| การชนกัน (Collision) | เหตุการณ์ที่ข้อมูลสองชุดต่างกันให้ค่าแฮชเดียวกัน ซึ่งอัลกอริทึมที่ดีต้องหาได้ยากมาก |
| Checksum | ค่าที่ใช้ตรวจสอบว่าไฟล์หรือข้อมูลไม่ถูกแก้ไขหรือเสียหายระหว่างส่งต่อ |
| SubtleCrypto | API เข้ารหัสลับที่มีอยู่ในเบราว์เซอร์ ใช้คำนวณแฮชโดยไม่ต้องติดตั้งไลบรารีเพิ่ม |
| TLS | โปรโตคอลเข้ารหัสการเชื่อมต่อของเว็บไซต์ (HTTPS) ใช้ SHA-256 ในการเซ็นใบรับรอง |
| JWT | โทเค็นที่ใช้ยืนยันตัวตนบนเว็บและ API ซึ่งมักใช้ HMAC-SHA256 ในการเซ็น |
| SHA256SUMS | ไฟล์รายการค่าแฮชของไฟล์ต่าง ๆ ที่ผู้พัฒนาแนบมา ใช้เทียบกับไฟล์ที่ดาวน์โหลด |

## คำถามที่พบบ่อย

**ถอดรหัสแฮช SHA-256 กลับเป็นข้อความเดิมได้หรือไม่?**
ไม่ได้ เพราะเป็นฟังก์ชันทางเดียว สิ่งที่ทำได้คือแฮชข้อความที่คาดเดาแล้วนำมาเทียบกับค่าแฮชที่มี

**ใช้ SHA-256 เก็บรหัสผ่านได้หรือไม่?**
ไม่ควรใช้ SHA-256 เพียงอย่างเดียว ควรใช้อัลกอริทึมสำหรับรหัสผ่านโดยเฉพาะ เช่น bcrypt, scrypt หรือ Argon2 ร่วมกับ salt

**ต้องการแฮชไฟล์แทนข้อความ ทำอย่างไร?**
เครื่องมือนี้รองรับเฉพาะข้อความ สำหรับไฟล์ ให้ใช้คำสั่ง `sha256sum ชื่อไฟล์` บน Linux/macOS หรือ `certutil -hashfile ชื่อไฟล์ SHA256` บน Windows
