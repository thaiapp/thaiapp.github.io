<script>
(function () {
  'use strict';

  var CDNS = [
    {
      pdfLib: 'https://cdnjs.cloudflare.com/ajax/libs/pdf-lib/1.17.1/pdf-lib.min.js',
      pdfjs: 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.min.js',
      worker: 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js'
    },
    {
      pdfLib: 'https://cdn.jsdelivr.net/npm/pdf-lib@1.17.1/dist/pdf-lib.min.js',
      pdfjs: 'https://cdn.jsdelivr.net/npm/pdfjs-dist@3.11.174/build/pdf.min.js',
      worker: 'https://cdn.jsdelivr.net/npm/pdfjs-dist@3.11.174/build/pdf.worker.min.js'
    }
  ];

  function loadScript(src) {
    return new Promise(function (resolve, reject) {
      var s = document.createElement('script');
      s.src = src;
      s.async = true;
      s.onload = function () { resolve(); };
      s.onerror = function () { s.parentNode && s.parentNode.removeChild(s); reject(new Error('load failed')); };
      document.head.appendChild(s);
    });
  }

  var libsPromise = null;
  function ensureLibs() {
    if (libsPromise) return libsPromise;
    function tryAt(i) {
      if (i >= CDNS.length) return Promise.reject(new Error('libs'));
      var c = CDNS[i];
      var p1 = window.PDFLib ? Promise.resolve() : loadScript(c.pdfLib);
      var p2 = window.pdfjsLib ? Promise.resolve() : loadScript(c.pdfjs);
      return Promise.all([p1, p2]).then(function () {
        if (!window.PDFLib || !window.pdfjsLib) throw new Error('missing');
        window.pdfjsLib.GlobalWorkerOptions.workerSrc = c.worker;
      }).catch(function () { return tryAt(i + 1); });
    }
    libsPromise = tryAt(0);
    libsPromise.catch(function () { libsPromise = null; });
    return libsPromise;
  }

  function clamp(n, a, b) { return Math.min(b, Math.max(a, n)); }

  function init() {
    var root = document.getElementById('wm-app');
    if (!root) return;
    var $ = function (id) { return document.getElementById(id); };
    var el = {
      drop: $('wm-drop'), file: $('wm-file'), pick: $('wm-pick'), work: $('wm-work'),
      modeText: $('wm-mode-text'), modeImage: $('wm-mode-image'), text: $('wm-text'),
      imgBtn: $('wm-img-btn'), imgFile: $('wm-img-file'), imgName: $('wm-img-name'), imgClear: $('wm-img-clear'),
      rotate: $('wm-rotate'), opacity: $('wm-opacity'), opacityVal: $('wm-opacity-val'),
      size: $('wm-size'), width: $('wm-width'), widthVal: $('wm-width-val'),
      color: $('wm-color'), tile: $('wm-tile'), skip: $('wm-skip'),
      apply: $('wm-apply'), download: $('wm-download'), clear: $('wm-clear'), reset: $('wm-reset'),
      prev: $('wm-prev'), next: $('wm-next'), pageNo: $('wm-page-no'), pageTotal: $('wm-page-total'),
      stage: $('wm-stage'), pageCanvas: $('wm-page-canvas'), overlay: $('wm-overlay'),
      skipNote: $('wm-skip-note'), msg: $('wm-msg'), fileName: $('wm-file-name')
    };

    var state = {
      bytes: null, name: '', doc: null, page: 1, count: 0, mode: 'text',
      img: null, fx: 0.5, fy: 0.5, view: null, resultUrl: null, defaultSize: 80
    };

    function say(text, isErr) {
      el.msg.textContent = text || '';
      el.msg.className = 'wm-msg' + (isErr ? ' wm-err' : '');
    }

    function invalidate() {
      if (state.resultUrl) { URL.revokeObjectURL(state.resultUrl); state.resultUrl = null; }
      el.download.hidden = true;
    }

    function applyMode() {
      el.modeText.setAttribute('aria-pressed', state.mode === 'text' ? 'true' : 'false');
      el.modeImage.setAttribute('aria-pressed', state.mode === 'image' ? 'true' : 'false');
      var nodes = root.querySelectorAll('[data-only]');
      for (var i = 0; i < nodes.length; i++) {
        nodes[i].hidden = nodes[i].getAttribute('data-only') !== state.mode;
      }
    }

    function parseSkip() {
      var set = {};
      var str = String(el.skip.value || '').replace(/\s*[-\u2013]\s*/g, '-');
      str.split(/[,;\s]+/).forEach(function (tok) {
        if (!tok) return;
        var m = /^(\d+)(?:-(\d+))?$/.exec(tok);
        if (!m) return;
        var a = parseInt(m[1], 10);
        var b = m[2] ? parseInt(m[2], 10) : a;
        if (a > b) { var t = a; a = b; b = t; }
        for (var p = a; p <= b && p <= 5000; p++) set[p] = true;
      });
      return set;
    }

    var textCache = { key: '', value: null };
    function textWm() {
      var text = (el.text.value || '').trim();
      if (!text) return null;
      var size = clamp(Number(el.size.value) || state.defaultSize, 8, 400);
      var color = el.color.value || '#808080';
      var key = text + '|' + size + '|' + color;
      if (textCache.key === key) return textCache.value;
      var font = function (px) { return '700 ' + px + 'px "Noto Sans Thai","Sarabun","Segoe UI",Arial,sans-serif'; };
      var c = document.createElement('canvas');
      var ctx = c.getContext('2d');
      ctx.font = font(size);
      var w1 = ctx.measureText(text).width + size * 0.4;
      var s = Math.max(0.2, Math.min(3, 12000 / w1));
      c.width = Math.max(2, Math.ceil(w1 * s));
      c.height = Math.max(2, Math.ceil(size * 1.6 * s));
      ctx = c.getContext('2d');
      ctx.font = font(size * s);
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillStyle = color;
      ctx.fillText(text, c.width / 2, c.height / 2);
      var value = { canvas: c, w: c.width / s, h: c.height / s };
      textCache = { key: key, value: value };
      return value;
    }

    function getWm(pageW) {
      if (state.mode === 'text') return textWm();
      if (!state.img) return null;
      var pct = clamp(Number(el.width.value) || 40, 5, 100);
      var w = pageW * pct / 100;
      return { canvas: state.img, w: w, h: w * state.img.height / state.img.width };
    }

    function computeItems(pageW, pageH, wm) {
      var angle = Number(el.rotate.value) || 0;
      var a = angle * Math.PI / 180;
      var ax = state.fx * pageW;
      var ay = state.fy * pageH;
      if (!el.tile.checked) return [{ cx: ax, cy: ay, w: wm.w, h: wm.h, angle: angle }];
      var stepX = wm.w + Math.max(wm.w * 0.4, 24);
      var stepY = wm.h + Math.max(wm.h * 0.6, 24);
      var diag = Math.sqrt(pageW * pageW + pageH * pageH);
      var ni = Math.ceil(diag / stepX) + 1;
      var nj = Math.ceil(diag / stepY) + 1;
      if ((2 * ni + 1) * (2 * nj + 1) > 60000) return [];
      var ext = Math.sqrt(wm.w * wm.w + wm.h * wm.h) / 2;
      var cos = Math.cos(a), sin = Math.sin(a);
      var out = [];
      for (var i = -ni; i <= ni; i++) {
        for (var j = -nj; j <= nj; j++) {
          var ox = i * stepX, oy = j * stepY;
          var cx = ax + ox * cos - oy * sin;
          var cy = ay + ox * sin + oy * cos;
          if (cx < -ext || cx > pageW + ext || cy < -ext || cy > pageH + ext) continue;
          out.push({ cx: cx, cy: cy, w: wm.w, h: wm.h, angle: angle });
        }
      }
      return out;
    }

    function drawOverlay() {
      var v = state.view;
      if (!v) return;
      var ctx = el.overlay.getContext('2d');
      ctx.clearRect(0, 0, el.overlay.width, el.overlay.height);
      var skipped = !!parseSkip()[state.page];
      el.skipNote.hidden = !skipped;
      if (skipped) return;
      var wm = getWm(v.pageW);
      if (!wm) return;
      var op = clamp(Number(el.opacity.value) || 30, 5, 100) / 100;
      var items = computeItems(v.pageW, v.pageH, wm);
      for (var n = 0; n < items.length; n++) {
        var it = items[n];
        ctx.save();
        ctx.globalAlpha = op;
        ctx.translate(it.cx * v.k, it.cy * v.k);
        ctx.rotate(it.angle * Math.PI / 180);
        ctx.drawImage(wm.canvas, -it.w * v.k / 2, -it.h * v.k / 2, it.w * v.k, it.h * v.k);
        ctx.restore();
      }
    }

    var renderToken = 0;
    var renderTask = null;
    function renderPage() {
      if (!state.doc) return;
      var token = ++renderToken;
      if (renderTask) { try { renderTask.cancel(); } catch (e) {} }
      el.pageNo.textContent = String(state.page);
      el.pageTotal.textContent = String(state.count);
      el.prev.disabled = state.page <= 1;
      el.next.disabled = state.page >= state.count;
      state.doc.getPage(state.page).then(function (page) {
        if (token !== renderToken) return null;
        var base = page.getViewport({ scale: 1 });
        var maxW = Math.max(200, Math.min(el.stage.clientWidth || 520, 680));
        var scale = maxW / base.width;
        var dpr = window.devicePixelRatio || 1;
        var vp = page.getViewport({ scale: scale * dpr });
        var cw = Math.floor(vp.width), ch = Math.floor(vp.height);
        el.pageCanvas.width = cw; el.pageCanvas.height = ch;
        el.overlay.width = cw; el.overlay.height = ch;
        var cssW = base.width * scale, cssH = base.height * scale;
        el.pageCanvas.style.width = cssW + 'px'; el.pageCanvas.style.height = cssH + 'px';
        el.overlay.style.width = cssW + 'px'; el.overlay.style.height = cssH + 'px';
        state.view = { pageW: base.width, pageH: base.height, k: cw / base.width };
        var ctx = el.pageCanvas.getContext('2d');
        ctx.fillStyle = '#fff';
        ctx.fillRect(0, 0, cw, ch);
        renderTask = page.render({ canvasContext: ctx, viewport: vp });
        return renderTask.promise.then(function () {
          if (token === renderToken) drawOverlay();
        });
      }).catch(function (e) {
        if (e && e.name === 'RenderingCancelledException') return;
        say('ไม่สามารถแสดงตัวอย่างหน้านี้ได้', true);
      });
    }

    function showWork(show) {
      el.drop.hidden = show;
      el.work.hidden = !show;
    }

    function clearFile() {
      renderToken++;
      if (renderTask) { try { renderTask.cancel(); } catch (e) {} }
      if (state.doc) { try { state.doc.destroy(); } catch (e) {} }
      state.doc = null; state.bytes = null; state.view = null; state.count = 0; state.page = 1;
      invalidate();
      el.file.value = '';
      showWork(false);
      say('');
    }

    function resetSettings() {
      state.mode = 'text';
      state.fx = 0.5; state.fy = 0.5;
      state.img = null;
      el.imgName.textContent = ''; el.imgClear.hidden = true; el.imgFile.value = '';
      el.text.value = 'DRAFT';
      el.rotate.value = '-45';
      el.opacity.value = '30'; el.opacityVal.textContent = '30';
      el.size.value = String(state.defaultSize);
      el.width.value = '40'; el.widthVal.textContent = '40';
      el.color.value = '#808080';
      el.tile.checked = false;
      el.skip.value = '';
      applyMode(); invalidate(); drawOverlay();
    }

    function loadFile(file) {
      if (!file) return;
      if (!(file.type === 'application/pdf' || /\.pdf$/i.test(file.name))) {
        say('กรุณาเลือกไฟล์ PDF เท่านั้น', true);
        return;
      }
      say('กำลังโหลดไฟล์…');
      var bytes = null;
      Promise.all([ensureLibs(), file.arrayBuffer()]).then(function (r) {
        bytes = r[1];
        return window.pdfjsLib.getDocument({ data: new Uint8Array(bytes.slice(0)) }).promise;
      }).then(function (doc) {
        if (state.doc) { try { state.doc.destroy(); } catch (e) {} }
        state.bytes = bytes; state.name = file.name; state.doc = doc;
        state.count = doc.numPages; state.page = 1;
        var limit = Math.min(doc.numPages, 20);
        var jobs = [];
        for (var p = 1; p <= limit; p++) {
          jobs.push(doc.getPage(p).then(function (pg) { return pg.getViewport({ scale: 1 }).width; }));
        }
        return Promise.all(jobs);
      }).then(function (widths) {
        var sum = 0;
        widths.forEach(function (w) { sum += w; });
        var avg = widths.length ? sum / widths.length : 595;
        state.defaultSize = clamp(Math.round(80 * avg / 595), 20, 300);
        el.size.value = String(state.defaultSize);
        el.fileName.textContent = state.name;
        state.fx = 0.5; state.fy = 0.5;
        invalidate();
        showWork(true);
        applyMode();
        say('');
        renderPage();
      }).catch(function (e) {
        var name = e && e.name;
        if (name === 'PasswordException') {
          say('ไฟล์ PDF นี้ถูกล็อกด้วยรหัสผ่าน กรุณาปลดล็อกก่อนใส่ลายน้ำ', true);
        } else if (name === 'InvalidPDFException') {
          say('ไฟล์ PDF เสียหรืออ่านไม่ได้', true);
        } else {
          say('โหลดไฟล์ไม่สำเร็จ หรือโหลดไลบรารีไม่ได้ กรุณาตรวจสอบอินเทอร์เน็ตแล้วลองใหม่', true);
        }
      });
    }

    function loadImage(file) {
      if (!file) return;
      if (!/^image\/(png|jpeg)$/.test(file.type)) {
        say('รองรับเฉพาะรูปภาพ PNG หรือ JPG', true);
        return;
      }
      var url = URL.createObjectURL(file);
      var img = new Image();
      img.onload = function () {
        var max = 1600;
        var k = Math.min(1, max / Math.max(img.naturalWidth, img.naturalHeight));
        var c = document.createElement('canvas');
        c.width = Math.max(1, Math.round(img.naturalWidth * k));
        c.height = Math.max(1, Math.round(img.naturalHeight * k));
        c.getContext('2d').drawImage(img, 0, 0, c.width, c.height);
        URL.revokeObjectURL(url);
        state.img = c;
        el.imgName.textContent = file.name;
        el.imgClear.hidden = false;
        say('');
        invalidate();
        drawOverlay();
      };
      img.onerror = function () {
        URL.revokeObjectURL(url);
        say('เปิดรูปภาพนี้ไม่ได้', true);
      };
      img.src = url;
    }

    function canvasToBytes(canvas) {
      return new Promise(function (resolve, reject) {
        canvas.toBlob(function (blob) {
          if (!blob) { reject(new Error('blob')); return; }
          blob.arrayBuffer().then(resolve, reject);
        }, 'image/png');
      });
    }

    function pause() { return new Promise(function (r) { setTimeout(r, 0); }); }

    async function applyWatermark() {
      if (!state.bytes) return;
      if (state.mode === 'text' && !textWm()) { say('กรุณาพิมพ์ข้อความลายน้ำ', true); return; }
      if (state.mode === 'image' && !state.img) { say('กรุณาเลือกรูปภาพลายน้ำ', true); return; }
      el.apply.disabled = true;
      invalidate();
      say('กำลังประมวลผล…');
      try {
        await ensureLibs();
        var PL = window.PDFLib;
        var pdf = await PL.PDFDocument.load(state.bytes, { ignoreEncryption: true });
        var pages = pdf.getPages();
        var skip = parseSkip();
        var op = clamp(Number(el.opacity.value) || 30, 5, 100) / 100;
        var sample = getWm(595);
        var embedded = await pdf.embedPng(await canvasToBytes(sample.canvas));
        for (var i = 0; i < pages.length; i++) {
          if (!skip[i + 1]) {
            var page = pages[i];
            var cb = page.getCropBox();
            var W = cb.width, H = cb.height;
            var R = ((Math.round(page.getRotation().angle) % 360) + 360) % 360;
            var vw = (R === 90 || R === 270) ? H : W;
            var vh = (R === 90 || R === 270) ? W : H;
            var wm = getWm(vw);
            var items = computeItems(vw, vh, wm);
            for (var n = 0; n < items.length; n++) {
              var it = items[n];
              var px, py;
              if (R === 90) { px = it.cy; py = it.cx; }
              else if (R === 180) { px = W - it.cx; py = it.cy; }
              else if (R === 270) { px = W - it.cy; py = H - it.cx; }
              else { px = it.cx; py = H - it.cy; }
              px += cb.x; py += cb.y;
              var deg = -it.angle + R;
              var th = deg * Math.PI / 180;
              var cos = Math.cos(th), sin = Math.sin(th);
              page.drawImage(embedded, {
                x: px - (it.w / 2 * cos - it.h / 2 * sin),
                y: py - (it.w / 2 * sin + it.h / 2 * cos),
                width: it.w,
                height: it.h,
                rotate: PL.degrees(deg),
                opacity: op
              });
            }
          }
          if (i % 5 === 4) {
            say('กำลังประมวลผล ' + (i + 1) + ' / ' + pages.length);
            await pause();
          }
        }
        var out = await pdf.save();
        var blob = new Blob([out], { type: 'application/pdf' });
        state.resultUrl = URL.createObjectURL(blob);
        el.download.href = state.resultUrl;
        el.download.setAttribute('download', state.name.replace(/\.pdf$/i, '') + '-watermark.pdf');
        el.download.hidden = false;
        say('เสร็จแล้ว กด “ดาวน์โหลด” เพื่อบันทึกไฟล์');
      } catch (e) {
        say('เพิ่มลายน้ำไม่สำเร็จ ไฟล์อาจถูกป้องกันหรือเสียหาย', true);
      } finally {
        el.apply.disabled = false;
      }
    }

    el.pick.addEventListener('click', function () { el.file.click(); });
    el.file.addEventListener('change', function () { loadFile(el.file.files && el.file.files[0]); });
    el.imgBtn.addEventListener('click', function () { el.imgFile.click(); });
    el.imgFile.addEventListener('change', function () { loadImage(el.imgFile.files && el.imgFile.files[0]); });
    el.imgClear.addEventListener('click', function () {
      state.img = null; el.imgName.textContent = ''; el.imgClear.hidden = true; el.imgFile.value = '';
      invalidate(); drawOverlay();
    });

    el.modeText.addEventListener('click', function () { state.mode = 'text'; applyMode(); invalidate(); drawOverlay(); });
    el.modeImage.addEventListener('click', function () { state.mode = 'image'; applyMode(); invalidate(); drawOverlay(); });

    [el.text, el.rotate, el.size, el.color, el.tile, el.skip].forEach(function (node) {
      node.addEventListener('input', function () { invalidate(); drawOverlay(); });
      node.addEventListener('change', function () { invalidate(); drawOverlay(); });
    });
    el.opacity.addEventListener('input', function () {
      el.opacityVal.textContent = el.opacity.value; invalidate(); drawOverlay();
    });
    el.width.addEventListener('input', function () {
      el.widthVal.textContent = el.width.value; invalidate(); drawOverlay();
    });

    el.prev.addEventListener('click', function () { if (state.page > 1) { state.page--; renderPage(); } });
    el.next.addEventListener('click', function () { if (state.page < state.count) { state.page++; renderPage(); } });
    el.apply.addEventListener('click', function () { applyWatermark(); });
    el.clear.addEventListener('click', clearFile);
    el.reset.addEventListener('click', resetSettings);

    var drag = null;
    el.overlay.addEventListener('pointerdown', function (e) {
      if (!state.view) return;
      drag = { x: e.clientX, y: e.clientY, fx: state.fx, fy: state.fy };
      try { el.overlay.setPointerCapture(e.pointerId); } catch (err) {}
      el.overlay.style.cursor = 'grabbing';
      e.preventDefault();
    });
    el.overlay.addEventListener('pointermove', function (e) {
      if (!drag) return;
      var r = el.overlay.getBoundingClientRect();
      state.fx = clamp(drag.fx + (e.clientX - drag.x) / r.width, 0, 1);
      state.fy = clamp(drag.fy + (e.clientY - drag.y) / r.height, 0, 1);
      invalidate();
      drawOverlay();
    });
    function endDrag() { drag = null; el.overlay.style.cursor = 'grab'; }
    el.overlay.addEventListener('pointerup', endDrag);
    el.overlay.addEventListener('pointercancel', endDrag);
    el.overlay.addEventListener('keydown', function (e) {
      var d = { ArrowLeft: [-0.01, 0], ArrowRight: [0.01, 0], ArrowUp: [0, -0.01], ArrowDown: [0, 0.01] }[e.key];
      if (!d) return;
      e.preventDefault();
      state.fx = clamp(state.fx + d[0], 0, 1);
      state.fy = clamp(state.fy + d[1], 0, 1);
      invalidate();
      drawOverlay();
    });

    ['dragenter', 'dragover'].forEach(function (ev) {
      root.addEventListener(ev, function (e) {
        e.preventDefault();
        el.drop.classList.add('wm-over');
      });
    });
    ['dragleave', 'drop'].forEach(function (ev) {
      root.addEventListener(ev, function (e) {
        e.preventDefault();
        el.drop.classList.remove('wm-over');
      });
    });
    root.addEventListener('drop', function (e) {
      var f = e.dataTransfer && e.dataTransfer.files && e.dataTransfer.files[0];
      if (f) loadFile(f);
    });

    document.addEventListener('paste', function (e) {
      var files = e.clipboardData && e.clipboardData.files;
      if (!files || !files.length) return;
      for (var i = 0; i < files.length; i++) {
        if (files[i].type === 'application/pdf' || /\.pdf$/i.test(files[i].name)) {
          e.preventDefault();
          loadFile(files[i]);
          return;
        }
      }
    });

    var resizeTimer = null;
    window.addEventListener('resize', function () {
      clearTimeout(resizeTimer);
      resizeTimer = setTimeout(function () { if (state.doc) renderPage(); }, 200);
    });

    applyMode();
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})();
</script>

## วิธีใช้งานเครื่องมือใส่ลายน้ำ PDF

1. **อัปโหลดไฟล์ PDF** – ลากไฟล์มาวางในกรอบด้านบน กดปุ่ม “เลือกไฟล์ PDF…” หรือคัดลอกไฟล์แล้วกด `Ctrl`+`V`
2. **เลือกโหมดลายน้ำ** – เลือก “ข้อความ” เพื่อพิมพ์ข้อความเอง หรือ “รูปภาพ” เพื่อใช้โลโก้หรือตราประทับ (PNG หรือ JPG)
3. **ปรับการตั้งค่า** – กำหนดการหมุน ความทึบ ขนาด (หรือความกว้างของรูป) และสี ตามที่ต้องการ ดูผลได้ทันทีในตัวอย่าง
4. **ลากจัดตำแหน่ง** – ลากลายน้ำในตัวอย่างไปยังตำแหน่งที่ต้องการ ตำแหน่งเดียวกันจะใช้กับทุกหน้า หรือเปิด “ทำซ้ำ” เพื่อให้ลายน้ำเรียงเต็มหน้า
5. **ข้ามบางหน้า (ไม่บังคับ)** – เปิด “ขั้นสูง” แล้วระบุเลขหน้าที่ไม่ต้องการลายน้ำ เช่น `1, 5-7`
6. **เพิ่มลายน้ำและดาวน์โหลด** – กด “เพิ่มลายน้ำ” รอสักครู่ จากนั้นกด “ดาวน์โหลด” เพื่อบันทึกไฟล์ PDF ใหม่ (ไฟล์ต้นฉบับไม่ถูกแก้ไข)

> ไฟล์ทั้งหมดถูกประมวลผลในเบราว์เซอร์ของคุณ ไม่มีการอัปโหลดไปยังเซิร์ฟเวอร์

## คำอธิบายตัวเลือกและคำศัพท์

| ตัวเลือก / คำศัพท์ | คำอธิบายสั้น ๆ |
| --- | --- |
| ลายน้ำ (Watermark) | ข้อความหรือรูปภาพจาง ๆ ที่วางซ้อนบนเนื้อหา ใช้ระบุสถานะหรือความเป็นเจ้าของเอกสาร เช่น DRAFT, CONFIDENTIAL หรือโลโก้บริษัท |
| ข้อความ | โหมดที่พิมพ์ข้อความเป็นลายน้ำ ระบบแปลงข้อความเป็นภาพก่อนฝังลง PDF จึงแสดงผลเหมือนกันในทุกโปรแกรมอ่าน รวมถึงภาษาไทย |
| รูปภาพ | โหมดที่ใช้ไฟล์ PNG หรือ JPG เป็นลายน้ำ PNG คงพื้นหลังโปร่งใสไว้ได้ |
| หมุน | มุมเอียงของลายน้ำ ค่าเริ่มต้น -45° เหมาะกับตราประทับแนวเฉียง ค่าติดลบเอียงขึ้นทางขวา ค่าบวกเอียงลงทางขวา |
| ความทึบ (Opacity) | ความเข้มของลายน้ำ 5–100% ยิ่งค่าน้อยยิ่งจาง และเห็นเนื้อหาเดิมชัดขึ้น |
| ขนาด (pt) | ขนาดตัวอักษรเป็นจุด (point) ใช้กับโหมดข้อความ ค่าเริ่มต้นปรับตามความกว้างหน้ากระดาษอัตโนมัติ |
| กว้าง (%) | ความกว้างของรูปลายน้ำเทียบกับความกว้างของหน้า (5–100%) ใช้กับโหมดรูปภาพ ความสูงปรับตามสัดส่วนเดิม |
| สี | สีของข้อความลายน้ำ ใช้กับโหมดข้อความเท่านั้น |
| ทำซ้ำ (Tile) | วางลายน้ำเรียงซ้ำเต็มหน้า เหมาะกับเอกสารที่ต้องการกันการตัดต่อ เมื่อลากจะเลื่อนทั้งรูปแบบพร้อมกัน |
| ขั้นสูง | เมนูตั้งค่าเพิ่มเติม ปัจจุบันใช้ระบุหน้าที่ต้องการข้าม |
| ข้ามหน้า | รายการหน้าที่ไม่ต้องการลายน้ำ คั่นด้วยเครื่องหมายจุลภาค และใช้ขีดสำหรับช่วงหน้า เช่น `1, 5-7` เหมาะกับหน้าปกหรือหน้าที่มีตราประทับอยู่แล้ว |
| ตัวอย่าง (Preview) | ภาพหน้าเอกสารที่แสดงลายน้ำให้เห็นก่อนบันทึก ใช้ปุ่ม ‹ › เพื่อเปลี่ยนหน้า |
| เพิ่มลายน้ำ | ปุ่มเริ่มประมวลผล ระบบจะวางลายน้ำลงทุกหน้า ยกเว้นหน้าที่ข้าม |
| ดาวน์โหลด | ปุ่มบันทึกไฟล์ PDF ที่ใส่ลายน้ำแล้ว ไฟล์ใหม่ชื่อลงท้ายด้วย -watermark |
| ล้าง | ลบไฟล์ที่เลือกไว้และกลับไปหน้าอัปโหลด |
| ลบการตั้งค่าทั้งหมด | รีเซ็ตทุกตัวเลือก (ข้อความ มุม ความทึบ สี ทำซ้ำ ข้ามหน้า) กลับเป็นค่าเริ่มต้น |
| ประมวลผลในเครื่อง | การทำงานเกิดในเบราว์เซอร์ของคุณ ไฟล์ไม่ถูกส่งออกจากอุปกรณ์ จึงปลอดภัยสำหรับเอกสารสำคัญ |
