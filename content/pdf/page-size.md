<script>
(function () {
  'use strict';
  var PT = 72 / 25.4;
  var LIBS = {
    pdflib: [
      'https://cdnjs.cloudflare.com/ajax/libs/pdf-lib/1.17.1/pdf-lib.min.js',
      'https://cdn.jsdelivr.net/npm/pdf-lib@1.17.1/dist/pdf-lib.min.js'
    ],
    pdfjs: [
      'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.min.js',
      'https://cdn.jsdelivr.net/npm/pdfjs-dist@3.11.174/build/pdf.min.js'
    ]
  };
  var WORKERS = [
    'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js',
    'https://cdn.jsdelivr.net/npm/pdfjs-dist@3.11.174/build/pdf.worker.min.js'
  ];
  var PRESETS = { a4: [210, 297], a3: [297, 420], a5: [148, 210], a6: [105, 148], letter: [215.9, 279.4], legal: [215.9, 355.6] };
  var $ = function (id) { return document.getElementById(id); };
  if (!$('pps')) { return; }

  var S = { items: [], size: { w: 210, h: 297 }, mode: 'fit', margin: 0, border: false, bg: '#ffffff' };
  var editing = null;
  var busy = false;

  /* ---------- helpers ---------- */
  function setStatus(msg, type) {
    var el = $('pps-status');
    el.textContent = msg || '';
    el.className = 'pps-status' + (type === 'err' ? ' err' : '');
  }
  function fmt(n) { return String(Math.round(n * 10) / 10); }
  function sheet() { return { w: S.size.w * PT, h: S.size.h * PT }; }
  function hexToRgb(h) {
    var m = /^#?([0-9a-f]{6})$/i.exec(h || '');
    if (!m) { return [1, 1, 1]; }
    var v = parseInt(m[1], 16);
    return [((v >> 16) & 255) / 255, ((v >> 8) & 255) / 255, (v & 255) / 255];
  }
  function toBlob(c, type, q) {
    return new Promise(function (r) { c.toBlob(r, type, q); });
  }

  /* ---------- load libraries dynamically with fallback CDN, so a bad URL can never break the page ---------- */
  function loadLib(urls, test) {
    return new Promise(function (resolve, reject) {
      var i = 0;
      (function next() {
        if (test()) { resolve(Math.max(0, i - 1)); return; }
        if (i >= urls.length) { reject(new Error('lib')); return; }
        var s = document.createElement('script');
        s.src = urls[i++];
        s.async = true;
        s.onload = function () { if (test()) { resolve(i - 1); } else { next(); } };
        s.onerror = function () { if (s.parentNode) { s.parentNode.removeChild(s); } next(); };
        document.head.appendChild(s);
      })();
    });
  }
  var libsP = null;
  function ensureLibs() {
    if (!libsP) {
      libsP = Promise.all([
        loadLib(LIBS.pdflib, function () { return !!window.PDFLib; }),
        loadLib(LIBS.pdfjs, function () { return !!window.pdfjsLib; })
      ]).then(function (r) {
        window.pdfjsLib.GlobalWorkerOptions.workerSrc = WORKERS[r[1]] || WORKERS[0];
      });
      libsP.catch(function () { libsP = null; });
    }
    return libsP;
  }

  /* ---------- layout (shared by preview and export) ---------- */
  function layout(it) {
    var sh = sheet();
    var phi = (it.rot + it.angle) * Math.PI / 180;
    var c = Math.abs(Math.cos(phi)), sn = Math.abs(Math.sin(phi));
    var bw = it.bw * c + it.bh * sn;
    var bh = it.bw * sn + it.bh * c;
    var m = S.margin * PT;
    var boxW = Math.max(1, sh.w - 2 * m), boxH = Math.max(1, sh.h - 2 * m);
    var s = S.mode === 'fit' ? Math.min(boxW / bw, boxH / bh) : 1;
    s *= it.scale;
    return { s: s, phi: phi, cx: sh.w / 2 + it.dx, cy: sh.h / 2 + it.dy, w0: it.bw * s, h0: it.bh * s };
  }
  // lower-left corner of a w x h box centred on (cx,cy) after rotating it by t (counter-clockwise, radians) about that corner
  function corner(cx, cy, w, h, t) {
    var vx = w / 2, vy = h / 2, c = Math.cos(t), s = Math.sin(t);
    return { x: cx - (vx * c - vy * s), y: cy - (vx * s + vy * c) };
  }

  /* ---------- adding files ---------- */
  function newItem(o) {
    o.scale = 1; o.rot = 0; o.angle = 0; o.dx = 0; o.dy = 0; o.cache = null; o.cv = null;
    return o;
  }
  function isPdf(f) { return f.type === 'application/pdf' || /\.pdf$/i.test(f.name); }
  function isImg(f) { return /^image\/(jpeg|png|webp)$/.test(f.type) || /\.(jpe?g|png|webp)$/i.test(f.name); }

  async function addPdf(f) {
    var buf = await f.arrayBuffer();
    var bytes = new Uint8Array(buf);
    var doc = await window.pdfjsLib.getDocument({ data: bytes.slice(0) }).promise;
    var rec = { name: f.name, bytes: bytes };
    for (var n = 1; n <= doc.numPages; n++) {
      var pg = await doc.getPage(n);
      var vp = pg.getViewport({ scale: 1, rotation: 0 });
      var r0 = ((pg.rotate % 360) + 360) % 360;
      var sw = (r0 % 180 !== 0);
      S.items.push(newItem({
        kind: 'pdf', rec: rec, page: pg, pageNo: n,
        cw: vp.width, ch: vp.height, r0: r0,
        bw: sw ? vp.height : vp.width, bh: sw ? vp.width : vp.height
      }));
    }
  }
  async function addImage(f) {
    var url = URL.createObjectURL(f);
    var img = await new Promise(function (res, rej) {
      var im = new Image();
      im.onload = function () { res(im); };
      im.onerror = function () { rej(new Error('img')); };
      im.src = url;
    });
    var w = img.naturalWidth, h = img.naturalHeight;
    var c = document.createElement('canvas');
    c.width = w; c.height = h;
    c.getContext('2d').drawImage(img, 0, 0);
    var jpg = /jpe?g/i.test(f.type) || /\.jpe?g$/i.test(f.name);
    var blob = await toBlob(c, jpg ? 'image/jpeg' : 'image/png', 0.95);
    var bytes = new Uint8Array(await blob.arrayBuffer());
    URL.revokeObjectURL(url);
    S.items.push(newItem({
      kind: 'img', rec: { name: f.name }, img: c, bytes: bytes, fmt: jpg ? 'jpg' : 'png',
      cw: w * 0.75, ch: h * 0.75, r0: 0, bw: w * 0.75, bh: h * 0.75
    }));
  }
  async function addFiles(list) {
    list = Array.prototype.slice.call(list || []);
    if (!list.length || busy) { return; }
    busy = true;
    setStatus('กำลังโหลดไฟล์…');
    try { await ensureLibs(); }
    catch (e) {
      busy = false;
      setStatus('โหลดไลบรารีไม่สำเร็จ กรุณาตรวจสอบอินเทอร์เน็ตแล้วลองใหม่อีกครั้ง', 'err');
      return;
    }
    var bad = [];
    for (var i = 0; i < list.length; i++) {
      var f = list[i];
      try {
        if (isPdf(f)) { await addPdf(f); }
        else if (isImg(f)) { await addImage(f); }
        else { bad.push(f.name + ' (ไม่รองรับชนิดไฟล์)'); }
      } catch (e) {
        bad.push(f.name + (e && e.name === 'PasswordException' ? ' (ไฟล์มีรหัสผ่าน)' : ' (เปิดไฟล์ไม่ได้)'));
      }
    }
    busy = false;
    renderGrid();
    if (bad.length) { setStatus('ข้ามไฟล์: ' + bad.join(', '), 'err'); }
    else { updateSummary(); }
  }

  /* ---------- bitmaps & painting ---------- */
  var renderChain = Promise.resolve();
  function getBitmap(it, q) {
    if (it.kind === 'img') { return Promise.resolve(it.img); }
    if (it.cache && it.cache.q >= q) { return Promise.resolve(it.cache.c); }
    var p = renderChain.then(async function () {
      if (it.cache && it.cache.q >= q) { return it.cache.c; }
      var sc = q / Math.max(it.bw, it.bh);
      var vp = it.page.getViewport({ scale: sc });
      var c = document.createElement('canvas');
      c.width = Math.max(1, Math.round(vp.width));
      c.height = Math.max(1, Math.round(vp.height));
      var ctx = c.getContext('2d');
      ctx.fillStyle = '#fff';
      ctx.fillRect(0, 0, c.width, c.height);
      await it.page.render({ canvasContext: ctx, viewport: vp }).promise;
      it.cache = { q: q, c: c };
      return c;
    });
    renderChain = p.catch(function () {});
    return p;
  }
  function drawSheet(cv, it, maxPx, bmp, guide) {
    var sh = sheet();
    var k = maxPx / Math.max(sh.w, sh.h);
    var w = Math.max(1, Math.round(sh.w * k)), h = Math.max(1, Math.round(sh.h * k));
    if (cv.width !== w || cv.height !== h) { cv.width = w; cv.height = h; }
    var ctx = cv.getContext('2d');
    ctx.setTransform(1, 0, 0, 1, 0, 0);
    ctx.fillStyle = S.bg;
    ctx.fillRect(0, 0, w, h);
    if (bmp) {
      var L = layout(it);
      ctx.save();
      ctx.translate(L.cx * k, L.cy * k);
      ctx.rotate(L.phi);
      ctx.drawImage(bmp, -L.w0 * k / 2, -L.h0 * k / 2, L.w0 * k, L.h0 * k);
      if (S.border) {
        ctx.lineWidth = Math.max(1, k * 0.75);
        ctx.strokeStyle = '#666';
        ctx.strokeRect(-L.w0 * k / 2, -L.h0 * k / 2, L.w0 * k, L.h0 * k);
      }
      ctx.restore();
    }
    if (guide && S.margin > 0) {
      var m = S.margin * PT * k;
      ctx.save();
      ctx.setLineDash([6, 4]);
      ctx.strokeStyle = '#2563eb';
      ctx.lineWidth = 1;
      ctx.strokeRect(m, m, w - 2 * m, h - 2 * m);
      ctx.restore();
    }
  }
  function redrawCard(it) {
    if (!it.cv) { return; }
    drawSheet(it.cv, it, 360, null, false);
    getBitmap(it, 700).then(function (b) {
      if (it.cv) { drawSheet(it.cv, it, 360, b, false); }
    }).catch(function () {});
  }
  function redrawAll() {
    S.items.forEach(redrawCard);
    if (editing) { drawEditor(); }
  }

  /* ---------- grid ---------- */
  function updateSummary() {
    var n = S.items.length;
    setStatus(n ? ('ทั้งหมด ' + n + ' หน้า · กระดาษ ' + fmt(S.size.w) + ' × ' + fmt(S.size.h) + ' mm · โหมด' + (S.mode === 'fit' ? 'พอดีหน้า' : 'ขนาดจริง')) : '');
  }
  function btn(txt, act, id, title) {
    var b = document.createElement('button');
    b.type = 'button'; b.textContent = txt; b.title = title; b.setAttribute('aria-label', title);
    b.setAttribute('data-act', act); b.setAttribute('data-i', String(id));
    return b;
  }
  function renderGrid() {
    var g = $('pps-grid');
    g.innerHTML = '';
    S.items.forEach(function (it, i) {
      var card = document.createElement('div');
      card.className = 'pps-card';
      var cv = document.createElement('canvas');
      it.cv = cv;
      var cap = document.createElement('div');
      cap.className = 'pps-cap';
      cap.textContent = 'หน้า ' + (i + 1) + ' · ' + it.rec.name + (it.kind === 'pdf' ? ' (' + it.pageNo + ')' : '');
      var bs = document.createElement('div');
      bs.className = 'pps-btns';
      bs.appendChild(btn('◀', 'left', i, 'ย้ายไปก่อนหน้า'));
      bs.appendChild(btn('▶', 'right', i, 'ย้ายไปถัดไป'));
      bs.appendChild(btn('⟳', 'rot', i, 'หมุน 90°'));
      bs.appendChild(btn('ปรับแต่ง…', 'edit', i, 'ปรับแต่งหน้านี้'));
      bs.appendChild(btn('✕', 'del', i, 'ลบหน้านี้'));
      card.appendChild(cv); card.appendChild(cap); card.appendChild(bs);
      g.appendChild(card);
      redrawCard(it);
    });
    var has = S.items.length > 0;
    $('pps-clear').disabled = !has;
    $('pps-download').disabled = !has;
  }
  $('pps-grid').addEventListener('click', function (e) {
    var b = e.target.closest ? e.target.closest('button[data-act]') : null;
    if (!b) { return; }
    var i = parseInt(b.getAttribute('data-i'), 10), act = b.getAttribute('data-act'), it = S.items[i];
    if (!it) { return; }
    if (act === 'left' && i > 0) { S.items.splice(i - 1, 0, S.items.splice(i, 1)[0]); renderGrid(); }
    else if (act === 'right' && i < S.items.length - 1) { S.items.splice(i + 1, 0, S.items.splice(i, 1)[0]); renderGrid(); }
    else if (act === 'rot') { it.rot = (it.rot + 90) % 360; redrawCard(it); }
    else if (act === 'edit') { openEditor(it); }
    else if (act === 'del') { S.items.splice(i, 1); renderGrid(); updateSummary(); }
  });

  $('pps-escale').addEventListener('input', function () {

  $('pps-eapply').addEventListener('click', function () {
    if (!editing) { return; }

    S.items.forEach(function (it) {

  $('pps-edel').addEventListener('click', function () {

  $('pps-edone').addEventListener('click', closeEditor);
  $('pps-modal').addEventListener('mousedown', function (e) { if (e.target === this) { closeEditor(); } });

  var drag = null;
  var ecv = $('pps-ecv');

  ecv.addEventListener('pointerup', endDrag);

  /* ---------- settings ---------- */
  function applySize(w, h, fromPreset) {

  $('pps-download').addEventListener('click', async function () {
    if (!S.items.length || busy) { return; }
    busy = true;
    var dl = $('pps-download');
    dl.disabled = true;
    setStatus('กำลังสร้างไฟล์ PDF…');
    try {
      await ensureLibs();
      var P = window.PDFLib;
      var out = await P.PDFDocument.create();
      var sh = sheet(), W = sh.w, H = sh.h;
      var bgc = hexToRgb(S.bg);
      var srcDocs = new Map();
      for (var i = 0; i < S.items.length; i++) {
        var it = S.items[i];
        var page = out.addPage([W, H]);
        if (S.bg.toLowerCase() !== '#ffffff') {
          page.drawRectangle({ x: 0, y: 0, width: W, height: H, color: P.rgb(bgc[0], bgc[1], bgc[2]) });
        }
        var L = layout(it);
        var cyp = H - L.cy;
        if (it.kind === 'pdf') {
          var src = srcDocs.get(it.rec);
          if (!src) {
            src = await P.PDFDocument.load(it.rec.bytes, { ignoreEncryption: true, updateMetadata: false });
            srcDocs.set(it.rec, src);
          }
          var sp = src.getPage(it.pageNo - 1);
          var cb = sp.getCropBox(), mb = sp.getMediaBox();
          var x0 = Math.max(cb.x, mb.x), y0 = Math.max(cb.y, mb.y);
          var x1 = Math.min(cb.x + cb.width, mb.x + mb.width), y1 = Math.min(cb.y + cb.height, mb.y + mb.height);
          if (x1 <= x0 || y1 <= y0) { x0 = mb.x; y0 = mb.y; x1 = mb.x + mb.width; y1 = mb.y + mb.height; }
          var emb = await out.embedPage(sp, { left: x0, bottom: y0, right: x1, top: y1 }, [1, 0, 0, 1, -x0, -y0]);
          var th = -(it.r0 + it.rot + it.angle) * Math.PI / 180;
          var wC = it.cw * L.s, hC = it.ch * L.s;
          var p = corner(L.cx, cyp, wC, hC, th);
          page.drawPage(emb, { x: p.x, y: p.y, width: wC, height: hC, rotate: P.radians(th) });
        } else {
          var im = it.fmt === 'jpg' ? await out.embedJpg(it.bytes) : await out.embedPng(it.bytes);
          var ti = -(it.rot + it.angle) * Math.PI / 180;
          var pi2 = corner(L.cx, cyp, L.w0, L.h0, ti);
          page.drawImage(im, { x: pi2.x, y: pi2.y, width: L.w0, height: L.h0, rotate: P.radians(ti) });
        }
        if (S.border) {
          var tb = -(it.rot + it.angle) * Math.PI / 180;
          var pb = corner(L.cx, cyp, L.w0, L.h0, tb);
          page.drawRectangle({
            x: pb.x, y: pb.y, width: L.w0, height: L.h0, rotate: P.radians(tb),
            borderColor: P.rgb(0.4, 0.4, 0.4), borderWidth: 0.75
          });
        }
        setStatus('กำลังสร้างไฟล์ PDF… ' + (i + 1) + '/' + S.items.length);
      }
      var bytes = await out.save();
      var blob = new Blob([bytes], { type: 'application/pdf' });
      var a = document.createElement('a');
      var base = S.items[0].rec.name.replace(/\.[^.]+$/, '') || 'document';
      a.href = URL.createObjectURL(blob);
      a.download = base + '-resized.pdf';
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      setTimeout(function () { URL.revokeObjectURL(a.href); }, 10000);
      setStatus('สร้างไฟล์เรียบร้อยแล้ว · ' + S.items.length + ' หน้า');
    } catch (e) {
      setStatus('สร้างไฟล์ไม่สำเร็จ: ' + ((e && e.message) ? e.message : 'เกิดข้อผิดพลาด'), 'err');
    }
    busy = false;
    dl.disabled = !S.items.length;
  });
})();

## วิธีใช้งานเครื่องมือปรับขนาดหน้า PDF

1. **เพิ่มไฟล์** คลิกกรอบด้านบน ลากไฟล์มาวาง หรือกด Ctrl+V เพื่อวางไฟล์ รองรับ PDF และรูปภาพ JPG, PNG, WebP เพิ่มได้หลายไฟล์พร้อมกัน
2. **เลือกขนาดกระดาษ** เลือก A4, A3, A5, A6, Letter, Legal หรือพิมพ์ความกว้างและความสูงเองเป็นมิลลิเมตร กดปุ่มสลับแนวตั้ง/แนวนอนได้ตามต้องการ
3. **เลือกขนาดเนื้อหา** เลือก "พอดีหน้า" เพื่อย่อหรือขยายเนื้อหาให้เต็มกระดาษใหม่ หรือเลือก "ขนาดจริง" เพื่อคงขนาดเนื้อหาเดิมไว้และเปลี่ยนเฉพาะกระดาษ
4. **ตั้งค่าเพิ่มเติม** ใส่ระยะขอบ เปิดเส้นขอบรอบเนื้อหา และเลือกสีพื้นหลังตามต้องการ ตัวอย่างด้านล่างจะอัปเดตทันที
5. **ตรวจและปรับทีละหน้า** ใช้ปุ่ม ◀ ▶ เรียงลำดับหน้า ⟳ หมุนหน้า ✕ ลบหน้า หรือกด "ปรับแต่ง…" เพื่อย่อ/ขยาย ลากย้ายตำแหน่ง และปรับให้ตรงเฉพาะหน้านั้น
6. **ดาวน์โหลด** กด "ดาวน์โหลด PDF" ไฟล์ใหม่ที่ได้จะมีทุกหน้าขนาดเท่ากัน

## คำอธิบายศัพท์ในเครื่องมือ

- **ขนาดหน้ากระดาษ** ขนาดแผ่นกระดาษของ PDF ที่จะพิมพ์ออกมา ไม่เกี่ยวกับขนาดไฟล์เป็นเมกะไบต์
- **A4 / A3 / A5 / A6** ขนาดกระดาษมาตรฐานสากลที่ใช้กันทั่วไปในประเทศไทย A4 คือ 210 × 297 มม. A3 ใหญ่กว่า A4 สองเท่า ส่วน A5 และ A6 เล็กลงทีละครึ่ง
- **Letter / Legal** ขนาดกระดาษแบบอเมริกัน Letter กว้างกว่า A4 เล็กน้อยแต่สั้นกว่า ส่วน Legal ยาวกว่า Letter
- **กำหนดเอง** พิมพ์ความกว้างและความสูงของกระดาษเองในหน่วยมิลลิเมตร (มม. หรือ mm)
- **สลับแนวตั้ง/แนวนอน** สลับค่าความกว้างกับความสูง เพื่อเปลี่ยนกระดาษจากแนวตั้งเป็นแนวนอนหรือกลับกัน
- **พอดีหน้า** ย่อหรือขยายเนื้อหาให้พอดีกับกระดาษใหม่ เหมาะกับการแปลงเอกสารเช่น Letter เป็น A4
- **ขนาดจริง** คงขนาดเนื้อหาเท่าเดิมทุกประการ เปลี่ยนเฉพาะกระดาษรอบๆ เหมาะกับแบบแปลน ไฟล์สแกน และแบบสำหรับตัดที่ต้องวัดขนาดได้ ส่วนรูปภาพจะคำนวณขนาดจริงที่ความละเอียด 96 พิกเซลต่อนิ้ว
- **ระยะขอบ** ช่องว่างสีพื้นหลังระหว่างขอบกระดาษกับเนื้อหา มีผลกับโหมดพอดีหน้าโดยตรง เพราะเนื้อหาจะถูกย่อให้อยู่ภายในกรอบที่เหลือ
- **เส้นขอบ** เส้นบางสีเทาที่วาดรอบเนื้อหาของแต่ละหน้า ช่วยให้เห็นขอบของเอกสารต้นฉบับชัดขึ้น
- **พื้นหลัง** สีของกระดาษด้านหลังเนื้อหา ค่าเริ่มต้นคือสีขาว
- **ปรับแต่ง…** เปิดหน้าต่างปรับค่าเฉพาะหน้านั้น ได้แก่ ขนาด ตำแหน่ง การหมุน และการปรับให้ตรง
- **ขนาด (%)** อัตราส่วนเนื้อหาเทียบกับค่าเริ่มต้นของโหมดที่เลือก 100% ในโหมดพอดีหน้าคือขนาดที่พอดีกรอบ ส่วนในโหมดขนาดจริงคือขนาดเดิม
- **ปรับให้ตรง** หมุนเนื้อหาเล็กน้อยเป็นองศา ใช้แก้เอกสารสแกนที่เอียง
- **หมุน 90°** หมุนหน้าไปทางขวาทีละ 90 องศา ใช้กับหน้าที่สแกนมาตะแคง
- **ใช้กับทุกหน้า** คัดลอกค่าปรับแต่งของหน้าปัจจุบันไปใช้กับทุกหน้าในครั้งเดียว
- **รีเซ็ตทั้งหมด** คืนการตั้งค่ากระดาษและค่าปรับแต่งทุกหน้ากลับเป็นค่าเริ่มต้น

## ขนาดหน้ากระดาษ ไม่ใช่ขนาดไฟล์

เครื่องมือนี้เปลี่ยน**ขนาดแผ่นกระดาษ** ของ PDF เช่น จาก Letter เป็น A4 ไม่ได้ทำให้ไฟล์มีขนาดเมกะไบต์น้อยลง หากต้องการลดขนาดไฟล์เพื่ออัปโหลดหรือส่งอีเมล ให้ใช้เครื่องมือบีบอัด PDF แทน

## ข้อควรรู้

- เนื้อหาของหน้าต้นฉบับจะถูกนำมาวางบนกระดาษใหม่โดยไม่แปลงเป็นภาพ ข้อความจึงยังเลือกและค้นหาได้เหมือนเดิม
- ลิงก์และช่องกรอกแบบฟอร์มจะใช้งานไม่ได้ในไฟล์ใหม่ หากเอกสารเป็นแบบฟอร์ม แนะนำให้กรอกจากไฟล์ต้นฉบับ แล้วค่อยปรับขนาดสำเนา
- เครื่องพิมพ์หรือหน้าต่างสั่งพิมพ์อาจย่อหน้าให้พอดีโดยอัตโนมัติ หากต้องการพิมพ์ขนาดจริง ให้เลือกพิมพ์ที่ 100% หรือ "Actual size"
- ไฟล์ PDF ที่ตั้งรหัสผ่านไม่สามารถเปิดได้ ต้องปลดล็อกก่อนใช้งาน
