(function () {
  var CDNS = [
    "https://cdn.jsdelivr.net/npm/pdf-lib@1.17.1/dist/pdf-lib.min.js",
    "https://cdnjs.cloudflare.com/ajax/libs/pdf-lib/1.17.1/pdf-lib.min.js"
  ];
  var files = [];
  var state = { pos: "bc", busy: false, sizeTouched: false };
  var libPromise = null;
  function $(id) { return document.getElementById(id); }
  function esc(s) {
    return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
  }
  function toNum(v, def) {
    var n = parseFloat(v);
    return isNaN(n) ? def : n;
  }
  function toInt(v, def, min) {
    var n = parseInt(v, 10);
    if (isNaN(n)) { return def; }
    return min > n ? min : n;
  }
  function fmtSize(b) {
    if (b >= 1048576) { return (b / 1048576).toFixed(1) + " MB"; }
    return Math.max(1, Math.round(b / 1024)) + " KB";
  }
  function setStatus(msg, type) {
    var el = $("pn-status");
    el.textContent = msg;
    el.className = "pn-status" + (type ? " pn-" + type : "");
  }
  function loadLib() {
    if (window.PDFLib) { return Promise.resolve(window.PDFLib); }
    if (libPromise) { return libPromise; }
    libPromise = new Promise(function (resolve, reject) {
      var i = 0;
      function next() {
        if (i >= CDNS.length) { libPromise = null; reject(new Error("pdf-lib load failed")); return; }
        var s = document.createElement("script");
        s.src = CDNS[i];
        i++;
        s.onload = function () {
          if (window.PDFLib) { resolve(window.PDFLib); } else { next(); }
        };
        s.onerror = function () {
          if (s.parentNode) { s.parentNode.removeChild(s); }
          next();
        };
        document.head.appendChild(s);
      }
      next();
    });
    return libPromise;
  }
  function labelFor(fmt, num, last) {
    if (fmt === "p") { return "Page " + num; }
    if (fmt === "np") { return num + " / " + last; }
    if (fmt === "pn") { return "Page " + num + " of " + last; }
    return String(num);
  }
  function parseSkip(str, total) {
    var set = {};
    String(str || "").split(/[,;\s]+/).forEach(function (t) {
      if (!t) { return; }
      var m = t.match(/^(\d+)(?:-(\d+))?$/);
      if (!m) { return; }
      var a = parseInt(m[1], 10);
      var b = m[2] ? parseInt(m[2], 10) : a;
      if (a > b) { var x = a; a = b; b = x; }
      for (var p = a; b >= p && total >= p; p++) {
        if (p >= 1) { set[p] = true; }
      }
    });
    return set;
  }
  function updatePreview() {
    var sc = 150 / 595;
    var m = Math.max(0, toNum($("pn-margin").value, 28)) * sc;
    var fs = Math.max(6, toNum($("pn-size").value, 14) * sc);
    var first = toInt($("pn-first").value, 1, 0);
    var el = $("pn-num");
    el.textContent = labelFor($("pn-format").value, first, first + 9);
    el.style.fontSize = fs + "px";
    el.style.color = $("pn-color").value;
    el.style.top = "auto";
    el.style.bottom = "auto";
    el.style.left = "auto";
    el.style.right = "auto";
    el.style.transform = "none";
    var p = state.pos;
    if (p.charAt(0) === "t") { el.style.top = m + "px"; } else { el.style.bottom = m + "px"; }
    var h = p.charAt(1);
    if (h === "l") { el.style.left = m + "px"; }
    else if (h === "r") { el.style.right = m + "px"; }
    else { el.style.left = "50%"; el.style.transform = "translateX(-50%)"; }
  }
  function refreshButtons() {
    $("pn-go").disabled = state.busy || files.length === 0;
  }
  function render() {
    var html = "";
    var total = 0;
    for (var i = 0; i !== files.length; i++) {
      var f = files[i];
      total += f.pages;
      html += '<li class="pn-file"><span class="pn-fname">' + esc(f.name) + '</span>' +
        '<span class="pn-fmeta">' + f.pages + ' หน้า · ' + fmtSize(f.size) + '</span>' +
        '<span class="pn-facts">' +
        '<button type="button" data-i="' + i + '" data-act="up" aria-label="เลื่อนขึ้น"' + (i === 0 ? ' disabled' : '') + '>▲</button>' +
        '<button type="button" data-i="' + i + '" data-act="down" aria-label="เลื่อนลง"' + (i === files.length - 1 ? ' disabled' : '') + '>▼</button>' +
        '<button type="button" data-i="' + i + '" data-act="del" aria-label="ลบไฟล์">✕</button>' +
        '</span></li>';
    }
    $("pn-files").innerHTML = html;
    $("pn-summary").textContent = files.length > 0
      ? (files.length > 1 ? "รวม " + files.length + " ไฟล์ " : "") + "ทั้งหมด " + total + " หน้า" + (files.length > 1 ? " (รวมเป็นไฟล์เดียวตามลำดับ)" : "")
      : "";
    refreshButtons();
  }
  function autoSize() {
    if (state.sizeTouched || files.length === 0) { return; }
    var v = Math.round(files[0].avgW / 42);
    v = Math.min(36, Math.max(8, v));
    $("pn-size").value = v;
    updatePreview();
  }
  async function addFiles(list) {
    var arr = Array.prototype.slice.call(list || []);
    var pdfs = arr.filter(function (f) {
      return /\.pdf$/i.test(f.name) || f.type === "application/pdf";
    });
    if (pdfs.length === 0) { setStatus("กรุณาเลือกไฟล์ .pdf เท่านั้น", "err"); return; }
    setStatus("กำลังอ่านไฟล์...", "");
    var errs = [];
    var L;
    try {
      L = await loadLib();
    } catch (e) {
      setStatus("โหลดตัวประมวลผล PDF ไม่สำเร็จ กรุณาตรวจสอบอินเทอร์เน็ตแล้วลองใหม่", "err");
      return;
    }
    for (var i = 0; i !== pdfs.length; i++) {
      var f = pdfs[i];
      try {
        var buf = await f.arrayBuffer();
        var bytes = new Uint8Array(buf);
        var doc = await L.PDFDocument.load(bytes);
        var pgs = doc.getPages();
        var sum = 0;
        for (var j = 0; j !== pgs.length; j++) {
          var rot = Math.abs(pgs[j].getRotation().angle) % 180;
          sum += rot === 90 ? pgs[j].getHeight() : pgs[j].getWidth();
        }
        files.push({ name: f.name, size: f.size, bytes: bytes, pages: pgs.length, avgW: pgs.length > 0 ? sum / pgs.length : 595 });
      } catch (e) {
        errs.push(f.name);
      }
    }
    render();
    autoSize();
    if (errs.length > 0) {
      setStatus("เปิดไฟล์นี้ไม่ได้ (อาจมีรหัสผ่านหรือไฟล์เสียหาย): " + errs.join(", "), "err");
    } else {
      setStatus("พร้อมแล้ว เลือกตัวเลือกแล้วกดปุ่มดาวน์โหลดได้เลย", "ok");
    }
  }
  function download(bytes, name) {
    var blob = new Blob([bytes], { type: "application/pdf" });
    var url = URL.createObjectURL(blob);
    var a = document.createElement("a");
    a.href = url;
    a.download = name;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    setTimeout(function () { URL.revokeObjectURL(url); }, 4000);
  }
  function hexToRgb(hex, rgb) {
    var h = String(hex || "#000000").replace("#", "");
    if (h.length !== 6) { h = "000000"; }
    return rgb(parseInt(h.substr(0, 2), 16) / 255, parseInt(h.substr(2, 2), 16) / 255, parseInt(h.substr(4, 2), 16) / 255);
  }
  async function run() {
    if (state.busy || files.length === 0) { return; }
    state.busy = true;
    refreshButtons();
    setStatus("กำลังเพิ่มเลขหน้า...", "");
    try {
      var L = await loadLib();
      var PDFDocument = L.PDFDocument;
      var out;
      if (files.length === 1) {
        out = await PDFDocument.load(files[0].bytes);
      } else {
        out = await PDFDocument.create();
        for (var i = 0; i !== files.length; i++) {
          var src = await PDFDocument.load(files[i].bytes);
          var copied = await out.copyPages(src, src.getPageIndices());
          for (var c = 0; c !== copied.length; c++) { out.addPage(copied[c]); }
        }
      }
      var font = await out.embedFont(L.StandardFonts.Helvetica);
      var pages = out.getPages();
      var total = pages.length;
      var fmt = $("pn-format").value;
      var margin = Math.max(0, toNum($("pn-margin").value, 28));
      var fs = Math.max(4, toNum($("pn-size").value, 14));
      var start = toInt($("pn-start").value, 1, 1);
      var first = toInt($("pn-first").value, 1, 0);
      var skip = parseSkip($("pn-skip").value, total);
      var color = hexToRgb($("pn-color").value, L.rgb);
      var numbered = 0;
      for (var k = 1; total >= k; k++) {
        if (k >= start && !skip[k]) { numbered++; }
      }
      if (numbered === 0) {
        setStatus("ไม่มีหน้าที่จะได้เลขหน้า ลองตรวจสอบค่า เริ่มจากหน้า และ ข้ามหน้า", "err");
        state.busy = false;
        refreshButtons();
        return;
      }
      var last = first + numbered - 1;
      var counter = first;
      var pos = state.pos;
      for (var n = 1; total >= n; n++) {
        if (n < start || skip[n]) { continue; }
        var page = pages[n - 1];
        var box;
        try { box = page.getCropBox(); } catch (e1) { box = page.getMediaBox(); }
        var w = box.width;
        var h = box.height;
        var rot = ((Math.round(page.getRotation().angle) % 360) + 360) % 360;
        var vw = (rot === 90 || rot === 270) ? h : w;
        var vh = (rot === 90 || rot === 270) ? w : h;
        var text = labelFor(fmt, counter, last);
        var tw = font.widthOfTextAtSize(text, fs);
        var vx;
        if (pos.charAt(1) === "l") { vx = margin; }
        else if (pos.charAt(1) === "r") { vx = vw - margin - tw; }
        else { vx = (vw - tw) / 2; }
        var vy = pos.charAt(0) === "t" ? vh - margin - fs * 0.72 : margin;
        var x, y;
        if (rot === 90) { x = w - vy; y = vx; }
        else if (rot === 180) { x = w - vx; y = h - vy; }
        else if (rot === 270) { x = vy; y = h - vx; }
        else { x = vx; y = vy; }
        page.drawText(text, {
          x: box.x + x,
          y: box.y + y,
          size: fs,
          font: font,
          color: color,
          rotate: L.degrees(rot)
        });
        counter++;
      }
      var bytes = await out.save();
      var base = files.length === 1 ? files[0].name.replace(/\.pdf$/i, "") : "merged";
      download(bytes, base + "-numbered.pdf");
      setStatus("เสร็จแล้ว เพิ่มเลขหน้า " + numbered + " หน้า จากทั้งหมด " + total + " หน้า", "ok");
    } catch (err) {
      setStatus("เกิดข้อผิดพลาดระหว่างประมวลผล: " + (err && err.message ? err.message : "ไม่ทราบสาเหตุ"), "err");
    }
    state.busy = false;
    refreshButtons();
  }
  var drop = $("pn-drop");
  var input = $("pn-file");
  drop.addEventListener("click", function () { input.click(); });

  $("pn-files").addEventListener("click", function (e) {
    var b = e.target.closest ? e.target.closest("button") : null;
    if (!b || b.disabled) { return; }
    var i = parseInt(b.getAttribute("data-i"), 10);
    var act = b.getAttribute("data-act");
    if (act === "del") { files.splice(i, 1); }
    else if (act === "up" && i > 0) { var t = files[i - 1]; files[i - 1] = files[i]; files[i] = t; }
    else if (act === "down" && files.length - 1 > i) { var u = files[i + 1]; files[i + 1] = files[i]; files[i] = u; }
    render();
    autoSize();
  });
  $("pn-pos").addEventListener("click", function (e) {
    var b = e.target.closest ? e.target.closest("button") : null;
    if (!b) { return; }
    state.pos = b.getAttribute("data-pos");
    var all = $("pn-pos").getElementsByTagName("button");
    for (var i = 0; i !== all.length; i++) { all[i].className = all[i] === b ? "pn-on" : ""; }
    updatePreview();
  });
  ["pn-format", "pn-margin", "pn-color", "pn-first"].forEach(function (id) {
    $(id).addEventListener("input", updatePreview);
    $(id).addEventListener("change", updatePreview);
  });
  $("pn-size").addEventListener("input", function () { state.sizeTouched = true; updatePreview(); });
  $("pn-go").addEventListener("click", run);
  $("pn-clear").addEventListener("click", function () {
    files = [];
    state.sizeTouched = false;
    render();
    setStatus("", "");
  });
  render();
  updatePreview();
})();
</script>

## วิธีใช้งานเครื่องมือใส่เลขหน้า PDF

1. **เลือกไฟล์ PDF** คลิกที่กรอบด้านบน ลากไฟล์มาวาง หรือกด Ctrl+V เพื่อวางไฟล์ หากเพิ่มหลายไฟล์ ระบบจะรวมเป็นไฟล์เดียวตามลำดับในรายการ และคุณกดลูกศรเพื่อสลับลำดับได้
2. **เลือกตำแหน่ง** กดหนึ่งในหกตำแหน่ง ได้แก่ ซ้ายบน กลางบน ขวาบน ซ้ายล่าง กลางล่าง และขวาล่าง ค่าเริ่มต้นคือกลางล่าง
3. **ตั้งค่ารูปแบบ** เลือกรูปแบบเลขหน้า ปรับระยะขอบ ขนาด และสีของตัวเลข แล้วดูตำแหน่งจากภาพตัวอย่างด้านขวา
4. **ตั้งค่าขั้นสูง (ถ้าต้องการ)** กำหนดหน้าที่เริ่มใส่เลข เลขเริ่มต้น หรือหน้าที่ต้องการข้าม เช่น หน้าปกและหน้าสารบัญ
5. **กดเพิ่มเลขหน้าและดาวน์โหลด** ระบบจะประมวลผลในเบราว์เซอร์แล้วดาวน์โหลดไฟล์ใหม่ที่ลงท้ายด้วย -numbered.pdf ให้ทันที

เคล็ดลับ: ถ้าเอกสารมีหน้าปก ให้ตั้ง เริ่มจากหน้า เป็น 2 และ เลขแรก เป็น 1 หน้าแรกของเนื้อหาจะเริ่มที่เลข 1 พอดี

## คำอธิบายคำศัพท์

- **เลขหน้า** ตัวเลขที่แสดงลำดับของแต่ละหน้าในเอกสาร ช่วยให้อ้างอิงและเปิดหาหน้าได้ง่าย
- **ตำแหน่ง** จุดที่วางเลขหน้าบนกระดาษ มีหกแบบ คือมุมซ้าย กลาง และขวา ทั้งด้านบนและด้านล่าง
- **ระยะขอบ** ระยะห่างระหว่างเลขหน้ากับขอบกระดาษ ตั้งเป็นพอยต์ ยิ่งค่ามากเลขยิ่งเข้ามาใกล้เนื้อหา
- **pt (พอยต์)** หน่วยวัดในงานพิมพ์และ PDF โดย 72 pt เท่ากับ 1 นิ้ว กระดาษ A4 กว้างประมาณ 595 pt
- **รูปแบบ** หน้าตาของเลขหน้า มี 1, Page 1, 1 / N และ Page 1 of N โดยข้อความในไฟล์ใช้ฟอนต์ Helvetica จึงแสดงเป็นคำภาษาอังกฤษ
- **N** จำนวนเลขหน้าทั้งหมดที่นับเฉพาะหน้าที่ได้รับเลข หน้าที่ข้ามจะไม่ถูกนับ
- **ขนาดตัวเลข** ความสูงของตัวอักษรเลขหน้า ตั้งเป็นพอยต์ ค่าเริ่มต้นปรับตามความกว้างหน้าอัตโนมัติ ประมาณ 14 pt สำหรับ A4
- **สีตัวเลข** สีของเลขหน้า ค่าเริ่มต้นคือสีดำ
- **ขั้นสูง** กลุ่มตัวเลือกเพิ่มเติมสำหรับเอกสารที่ไม่ได้เริ่มนับเลขจากหน้าแรก
- **เริ่มจากหน้า** หน้าแรกที่จะได้รับเลขหน้า เช่น ตั้งเป็น 2 เพื่อเว้นหน้าปกไว้ไม่มีเลข
- **เลขแรก** ตัวเลขที่จะแสดงบนหน้าแรกที่ได้รับเลข ใช้ต่อเลขหน้าจากไฟล์อื่นได้
- **ข้ามหน้า** รายการหน้าที่ไม่ต้องการใส่เลข คั่นด้วยจุลภาคและใช้ช่วงได้ เช่น 1, 5-7
- **ประมวลผลในเบราว์เซอร์** ไฟล์ถูกแก้ไขบนอุปกรณ์ของคุณโดยตรง ไม่มีการอัปโหลดไปยังเซิร์ฟเวอร์
