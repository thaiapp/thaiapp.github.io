  function shuffle(a) {
    for (var i = a.length - 1; i > 0; i--) {

  function load() {
    try {
      var raw = window.localStorage.getItem(KEY);
      if (!raw) { return; }

      if (['hira', 'kata', 'both'].indexOf(o.script) !== -1) { S.script = o.script; }
      if (Array.isArray(o.groups)) {
        var g = o.groups.filter(function (x) { return ORDER.indexOf(x) !== -1; });

      });
      if (o.stars && typeof o.stars === 'object') {
        Object.keys(o.stars).forEach(function (c) {
          if (typeof o.stars[c] === 'string') { S.stars[c] = o.stars[c]; }
        });
      }
    } catch (e) { /* storage unavailable */ }
  }

  }

  function loadVoices() {
    try {
      if (!('speechSynthesis' in window)) { noJa = true; return; }
      voices = window.speechSynthesis.getVoices() || [];
      if (voices.length > 0) {
        noJa = !voices.some(function (v) { return v.lang && v.lang.toLowerCase().replace('_', '-').indexOf('ja') === 0; });
      }
    } catch (e) { /* ignore */ }
  }
  function pickVoice() {
    for (var i = 0; i !== voices.length; i++) {
      if (voices[i].lang && voices[i].lang.toLowerCase().replace('_', '-').indexOf('ja') === 0) { return voices[i]; }
    }
    return null;
  }
  function speak(t) {
    if (!S.sound) { return; }
    try {
      if (!('speechSynthesis' in window)) { return; }
      var ss = window.speechSynthesis;
      if (voices.length === 0) { loadVoices(); }
      ss.cancel();
      if (ss.paused) { ss.resume(); }
      window.setTimeout(function () {
        var u = new SpeechSynthesisUtterance(t);
        u.lang = 'ja-JP';
        u.rate = 0.8;
        var v = pickVoice();
        if (v) { u.voice = v; }
        curUtter = u;
        ss.speak(u);
      }, 60);
    } catch (e) { /* ignore */ }
  }

    ORDER.forEach(function (g) {

      scripts.forEach(function (s) {
        SETS[g][s].forEach(function (c, i) { out.push({ c: c, r: SETS[g].r[i] }); });
      });
    });
    return out;
  }

  function advance(viaKey) {

    run.idx++;
    run.shown = (run.mode === 0);
    renderCard(true);

    if (run.idx === 0) { return; }

## วิธีใช้งานการ์ดคำคานะ

1. **เลือกโหมดฝึก** กดปุ่ม "เริ่ม" ในโหมด #1 ฝึกความจำ, #2 ฝึกการอ่าน หรือ #3 ฝึกการเขียน
2. **ตั้งค่าก่อนเริ่ม (ไม่บังคับ)** เลือกชุดตัวอักษร (ฮิรางานะ คาตาคานะ หรือทั้งสอง) และกลุ่มเสียงที่ต้องการ เลือกได้มากกว่าหนึ่งกลุ่ม
3. **ดูการ์ดและเฉลย** ในโหมดอ่านและเขียน ให้แตะการ์ดหรือกด Space เพื่อเฉลยคำตอบ
4. **ฟังเสียงอ่าน** กดปุ่ม 🔊 หรือแตะการ์ดอีกครั้งหลังเฉลย หรือเปิด "เล่นเสียงอัตโนมัติ" ให้เล่นเสียงทุกครั้งที่เห็นคำตอบ
5. **ติดดาวตัวที่จำยาก** กดปุ่มดาวบนการ์ด หรือกด S แล้วเปิด "ฝึกเฉพาะตัวที่ติดดาว" เพื่อทบทวนเฉพาะตัวเหล่านั้น
6. **ไปการ์ดถัดไปหรือย้อนกลับ** ใช้ปุ่ม "ก่อนหน้า" "ถัดไป" หรือปุ่มลูกศร และกด "เสร็จสิ้น" เมื่อฝึกครบทุกใบ
7. **เปิดดูตารางอ้างอิง** เลื่อนไปที่ "อ้างอิงคานะ" เพื่อดูตัวอักษรทั้งหมด ฟังเสียง และติดดาวทีละตัวหรือทั้งกลุ่ม

ระบบจะจดจำการตั้งค่าและตัวอักษรที่ติดดาวไว้ในเบราว์เซอร์ของคุณโดยอัตโนมัติ

## โหมดฝึกทั้ง 3 แบบ

- **#1 ฝึกความจำ** แสดงตัวอักษรพร้อมเสียงอ่านพร้อมกัน เหมาะกับผู้เริ่มต้นที่ยังไม่คุ้นกับคานะ
- **#2 ฝึกการอ่าน** เห็นตัวอักษรแล้วต้องนึกเสียงอ่านเอง จากนั้นเฉลยเพื่อตรวจคำตอบ ช่วยเพิ่มความเร็วในการจดจำ
- **#3 ฝึกการเขียน** เห็นโรมาจิแล้วนึกหรือเขียนตัวอักษรลงกระดาษ จากนั้นเฉลยเพื่อเทียบกับตัวจริง

## คำอธิบายศัพท์ที่ควรรู้

- **คานะ (仮名)** ชื่อเรียกรวมของอักษรพยางค์ภาษาญี่ปุ่น 2 ชุด คือฮิรางานะและคาตาคานะ
- **ฮิรางานะ (ひらがな)** อักษรทรงโค้งมน ใช้เขียนคำญี่ปุ่นทั่วไป คำช่วย และส่วนท้ายของคำกริยา
- **คาตาคานะ (カタカナ)** อักษรทรงเหลี่ยม ใช้เขียนคำยืมจากต่างประเทศ ชื่อชาวต่างชาติ และคำที่ต้องการเน้น
- **โรมาจิ (ローマ字)** การเขียนเสียงอ่านภาษาญี่ปุ่นด้วยตัวอักษรละติน เช่น あ = a, か = ka
- **เสียงพื้นฐาน (清音 เซอิอง)** ชุดอักษรหลัก 46 ตัว เช่น あ い う え お และแถว か さ た นอกจากนี้ยังรวมตัว ん
- **เสียงก้อง (濁音 ดาคุอง)** เสียงที่เติมเครื่องหมายสองขีด ゛ เช่น か → が, さ → ざ ในกลุ่มนี้ยังรวมเสียงกึ่งก้อง (半濁音 ฮันดาคุอง) ที่เติมวงกลม ゜ เช่น は → ぱ
- **เสียงผสม (拗音 โยอง)** เสียงที่เกิดจากอักษรแถว い รวมกับ ゃ ゅ ょ ตัวเล็ก เช่น き + ゃ = きゃ (kya)
- **การ์ดคำ** บัตรที่ด้านหนึ่งเป็นโจทย์และอีกด้านเป็นคำตอบ ใช้ทบทวนซ้ำเพื่อช่วยจำ
- **สุ่มลำดับ** สลับลำดับการ์ดในแต่ละรอบ เพื่อให้จำตัวอักษรได้จริง ไม่ใช่จำลำดับ
- **ติดดาว** ทำเครื่องหมายตัวอักษรที่จำยากเพื่อกลับมาทบทวนภายหลัง
- **เล่นเสียง / เล่นเสียงอัตโนมัติ** ใช้เสียงอ่านภาษาญี่ปุ่นของเบราว์เซอร์ อัตโนมัติคือเล่นทันทีเมื่อเห็นคำตอบ
- **โหมดกลางคืน** เปลี่ยนสีหน้าจอเป็นโทนเข้ม ถนอมสายตาเมื่อใช้ในที่แสงน้อย
- **อ้างอิงคานะ** ตารางรวมอักษรทั้งหมดให้ดู ฟังเสียง และติดดาวได้ทันที
- **รีเซ็ตข้อมูลทั้งหมด** ล้างการตั้งค่าและตัวอักษรที่ติดดาวที่บันทึกไว้ในเบราว์เซอร์

## เคล็ดลับการเรียนรู้

- เริ่มจากเสียงพื้นฐาน แล้วค่อยเพิ่มเสียงก้องและเสียงผสมเมื่อพร้อม
- ฝึกวันละ 10-15 นาทีทุกวัน ได้ผลดีกว่าฝึกนานๆ ครั้งต่อสัปดาห์
- ติดดาวตัวที่ผิดบ่อย แล้วเปิด "ฝึกเฉพาะตัวที่ติดดาว" เมื่อมีเวลาจำกัด
- เปิดสุ่มลำดับเพื่อไม่ให้จำเป็นแถว
- เมื่อคล่องแล้ว ลองสลับโหมดจากจำ → อ่าน → เขียนตามลำดับ
