  }

  }
  function outName(it) {
    var suf = String(suffixEl.value || '').replace(/[\\\/:*?"<>|]+/g, '_').trim();

  }

  var CRC = (function () {
    var t = [], c, n, k;
    for (n = 0; n < 256; n++) {
      c = n;
      for (k = 0; k < 8; k++) { c = (c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1); }
      t[n] = c >>> 0;
    }
    return t;
  })();
  function crc32(u8) {
    var c = 0xFFFFFFFF;
    for (var i = 0; i < u8.length; i++) { c = CRC[(c ^ u8[i]) & 255] ^ (c >>> 8); }
    return (c ^ 0xFFFFFFFF) >>> 0;
  }
  function makeZip(entries) {
    var enc = new TextEncoder(), parts = [], central = [], offset = 0, cdSize = 0;
    var d = new Date();
    var dt = ((d.getHours() << 11) | (d.getMinutes() << 5) | (d.getSeconds() >> 1)) & 0xFFFF;
    var dd = (((d.getFullYear() - 1980) << 9) | ((d.getMonth() + 1) << 5) | d.getDate()) & 0xFFFF;
    entries.forEach(function (e) {
      var nameB = enc.encode(e.name), crc = crc32(e.data), size = e.data.length;
      var lh = new DataView(new ArrayBuffer(30));
      lh.setUint32(0, 0x04034b50, true); lh.setUint16(4, 20, true); lh.setUint16(6, 0x0800, true);
      lh.setUint16(8, 0, true); lh.setUint16(10, dt, true); lh.setUint16(12, dd, true);
      lh.setUint32(14, crc, true); lh.setUint32(18, size, true); lh.setUint32(22, size, true);
      lh.setUint16(26, nameB.length, true); lh.setUint16(28, 0, true);
      parts.push(lh.buffer, nameB, e.data);
      var ch = new DataView(new ArrayBuffer(46));
      ch.setUint32(0, 0x02014b50, true); ch.setUint16(4, 20, true); ch.setUint16(6, 20, true);
      ch.setUint16(8, 0x0800, true); ch.setUint16(10, 0, true); ch.setUint16(12, dt, true);
      ch.setUint16(14, dd, true); ch.setUint32(16, crc, true); ch.setUint32(20, size, true);
      ch.setUint32(24, size, true); ch.setUint16(28, nameB.length, true); ch.setUint16(30, 0, true);
      ch.setUint16(32, 0, true); ch.setUint16(34, 0, true); ch.setUint16(36, 0, true);
      ch.setUint32(38, 0, true); ch.setUint32(42, offset, true);
      central.push(ch.buffer, nameB);
      cdSize += 46 + nameB.length;
      offset += 30 + nameB.length + size;
    });
    var end = new DataView(new ArrayBuffer(22));
    end.setUint32(0, 0x06054b50, true); end.setUint16(4, 0, true); end.setUint16(6, 0, true);
    end.setUint16(8, entries.length, true); end.setUint16(10, entries.length, true);
    end.setUint32(12, cdSize, true); end.setUint32(16, offset, true); end.setUint16(20, 0, true);
    return new Blob(parts.concat(central, [end.buffer]), { type: 'application/zip' });
  }

  function saveUrl(url, filename) {
    var a = document.createElement('a');
    a.href = url;
    a.download = filename;
    document.body.appendChild(a);
    a.click();

  }
  function addFiles(files) {
    var arr = Array.prototype.slice.call(files || []).filter(function (f) {
      return /^image\//.test(f.type) || /\.(png|jpe?g|gif|bmp|heic|heif|avif|webp)$/i.test(f.name);
    });
    if (!arr.length) {
      setStatus('ไม่พบไฟล์รูปภาพ กรุณาเลือกไฟล์ PNG, JPG หรือรูปภาพอื่น', true);
      return;
    }
    if (!canEncode) {
      setStatus('เบราว์เซอร์นี้สร้างไฟล์ WebP ไม่ได้ กรุณาใช้ Chrome, Edge หรือ Firefox รุ่นล่าสุด', true);
      return;
    }
    setStatus('');
    arr.forEach(function (f) {

  function loadImage(it) {
    var img = new Image();
    img.onload = function () {
      if (it.removed) { return; }
      it.img = img;
      it.ow = img.naturalWidth;
      it.oh = img.naturalHeight;

    };

      updateSummary();
    };
    img.src = it.origUrl;
  }
  function scheduleConvert(delay) {
    clearTimeout(timer);
    timer = setTimeout(convertAll, delay);
  }
  function encode(img, q, max, keepAlpha, bg) {
    return new Promise(function (resolve, reject) {
      var w = img.naturalWidth, h = img.naturalHeight, s = 1;
      if (max > 0 && Math.max(w, h) > max) { s = max / Math.max(w, h); }
      var cw = Math.max(1, Math.round(w * s)), ch = Math.max(1, Math.round(h * s));
      var c = document.createElement('canvas');
      c.width = cw;
      c.height = ch;
      var ctx = c.getContext('2d');
      if (!ctx) { reject(new Error('ไม่สามารถประมวลผลรูปนี้ได้ รูปอาจมีขนาดใหญ่เกินไป')); return; }
      if (!keepAlpha) { ctx.fillStyle = bg; ctx.fillRect(0, 0, cw, ch); }
      ctx.imageSmoothingEnabled = true;
      ctx.imageSmoothingQuality = 'high';
      ctx.drawImage(img, 0, 0, cw, ch);
      c.toBlob(function (b) {
        if (!b || b.type !== 'image/webp') {
          reject(new Error('เบราว์เซอร์นี้สร้างไฟล์ WebP ไม่ได้ กรุณาใช้ Chrome, Edge หรือ Firefox รุ่นล่าสุด'));
          return;
        }
        resolve({ blob: b, w: cw, h: ch });
      }, 'image/webp', q);
    });
  }
  async function convertAll() {

    todo.forEach(function (i) { i.state = 'busy'; renderItem(i); });
    updateSummary();

      if (g !== gen) { return; }
      if (it.removed) { continue; }

        if (g !== gen) { return; }

        it.blob = res.blob;
        it.url = URL.createObjectURL(res.blob);
        it.w = res.w;
        it.h = res.h;
        it.sig = sig;
        it.state = 'done';
      } catch (e) {
        if (g !== gen) { return; }
        it.state = 'error';
        it.msg = (e && e.message) ? e.message : 'แปลงไฟล์ไม่สำเร็จ';
      }
      renderItem(it);
      updateSummary();
      await new Promise(function (r) { setTimeout(r, 0); });
    }
    updateSummary();
  }

      var dims = it.ow + '×' + it.oh + (it.w !== it.ow || it.h !== it.oh ? ' → ' + it.w + '×' + it.h + ' px' : ' px');
      html = '<div class="pw-cmp"><img alt="ภาพหลังแปลงเป็น WebP" src="' + it.url + '">' +

## วิธีใช้งานเครื่องมือแปลง PNG เป็น WebP

1. **เพิ่มรูปภาพ** ลากไฟล์ PNG (หรือ JPG) มาวางในกรอบด้านบน คลิกเพื่อเลือกไฟล์ หรือกด Ctrl/Cmd + V เพื่อวางรูปที่คัดลอกไว้ เลือกได้หลายไฟล์พร้อมกัน
2. **เลือกพรีเซ็ตหรือปรับคุณภาพ** เริ่มจาก "สมดุล" ถ้าไม่แน่ใจ ต้องการไฟล์เล็กที่สุดให้เลือก "ไฟล์เล็กที่สุด" หรือถ้าเป็นภาพที่ต้องคมชัดมาก เช่น โลโก้และสกรีนช็อตข้อความ ให้เลือก "คุณภาพสูง" หรือ "สูงสุด"
3. **ย่อขนาดภาพ (ไม่บังคับ)** เลือกความกว้างหรือความสูงสูงสุด เช่น 1920 px เครื่องมือจะย่อภาพโดยรักษาสัดส่วนเดิมและไม่ขยายภาพที่เล็กอยู่แล้ว
4. **ตั้งค่าความโปร่งใส** ปล่อยติ๊ก "คงพื้นหลังโปร่งใส" ไว้ถ้าต้องการรักษาส่วนโปร่งใสของ PNG หากเอาติ๊กออกจะมีช่องเลือกสีพื้นหลังมาแทนที่
5. **ตรวจผลลัพธ์** เลื่อนแถบบนภาพตัวอย่างเพื่อเทียบภาพก่อนและหลัง ดูขนาดไฟล์ที่ลดลงต่อไฟล์ และสรุปรวมด้านบนรายการ
6. **ตั้งชื่อไฟล์ (ไม่บังคับ)** ใส่คำต่อท้ายชื่อไฟล์ เช่น `-small` เพื่อแยกจากไฟล์ต้นฉบับ
7. **ดาวน์โหลด** กด "ดาวน์โหลด" ที่ไฟล์ที่ต้องการ หรือ "ดาวน์โหลด ZIP" เพื่อรับทุกไฟล์ในครั้งเดียว
8. **นำไปใช้บนเว็บไซต์** กด "คัดลอกโค้ด HTML" เพื่อได้แท็ก `<picture>` ที่ใช้ WebP พร้อมไฟล์เดิมเป็นตัวสำรองสำหรับเบราว์เซอร์เก่า

## อธิบายศัพท์ที่ใช้ในเครื่องมือ

- **PNG** รูปแบบไฟล์ภาพที่เก็บรายละเอียดครบและรองรับพื้นหลังโปร่งใส เหมาะกับโลโก้ ไอคอน และสกรีนช็อต แต่ไฟล์มักมีขนาดใหญ่
- **WebP** รูปแบบไฟล์ภาพที่ Google พัฒนา บีบอัดได้เล็กกว่า PNG และ JPG ในคุณภาพที่ใกล้เคียงกัน รองรับทั้งพื้นหลังโปร่งใสและภาพเคลื่อนไหว
- **Lossy (บีบอัดแบบสูญเสียข้อมูล)** ตัดรายละเอียดที่ตาเห็นยากออกเพื่อให้ไฟล์เล็กลงมาก ยิ่งคุณภาพต่ำ ไฟล์ยิ่งเล็กแต่ภาพยิ่งเสียรายละเอียด
- **Lossless (บีบอัดแบบไม่สูญเสียข้อมูล)** ลดขนาดโดยไม่ทำให้พิกเซลเปลี่ยน คุณภาพ 100 ในเครื่องมือนี้ให้ผลใกล้เคียง lossless แต่ผลจริงขึ้นกับเบราว์เซอร์ที่ใช้
- **คุณภาพ (Quality)** ค่า 1 ถึง 100 ที่กำหนดความละเอียดของภาพหลังบีบอัด ค่า 75 ถึง 85 มักเหมาะกับภาพทั่วไปบนเว็บ
- **พรีเซ็ต (Preset)** ชุดค่าคุณภาพที่ตั้งไว้ล่วงหน้า เลือกได้ในคลิกเดียว หากขยับแถบคุณภาพเองจะเปลี่ยนเป็น "กำหนดเอง"
- **ความโปร่งใส (Alpha)** ช่องข้อมูลที่บอกว่าพิกเซลใดโปร่งใสมากน้อยแค่ไหน ถ้าไม่เก็บไว้ ส่วนโปร่งใสจะถูกเติมด้วยสีที่เลือก
- **ย่อขนาด (Resize)** ลดจำนวนพิกเซลของภาพ ช่วยให้ไฟล์เล็กลงอีกมาก เหมาะเมื่อภาพต้นฉบับใหญ่กว่าที่แสดงบนเว็บจริง
- **แปลงเป็นชุด (Batch)** แปลงหลายไฟล์พร้อมกันในครั้งเดียว
- **ก่อน/หลัง (Before/After)** แถบเลื่อนเทียบภาพต้นฉบับกับภาพที่แปลงแล้ว ช่วยตัดสินใจว่าคุณภาพที่เลือกเพียงพอหรือไม่
- **ZIP** ไฟล์บีบอัดที่รวมหลายไฟล์ไว้ด้วยกัน สะดวกเมื่อแปลงหลายรูป
- **ต่อท้ายชื่อไฟล์ (Suffix)** ข้อความที่เพิ่มท้ายชื่อไฟล์ใหม่ เช่น `logo` เป็น `logo-small.webp`
- **`<picture>`** แท็ก HTML ที่ให้เบราว์เซอร์เลือกโหลดรูปแบบภาพที่รองรับ ใช้ WebP ก่อนและถอยไปใช้ PNG เมื่อไม่รองรับ
- **HEIC** รูปแบบภาพที่ iPhone ใช้ เครื่องมือจะอ่านได้เมื่อเบราว์เซอร์ถอดรหัสไฟล์นั้นได้เอง เช่น Safari
- **Core Web Vitals / LCP** ตัวชี้วัดความเร็วของ Google โดย LCP คือเวลาที่เนื้อหาหลักของหน้าแสดงเสร็จ รูปที่เล็กลงช่วยให้ LCP ดีขึ้น
- **ประมวลผลในเบราว์เซอร์ (Client-side)** งานแปลงทำบนเครื่องของคุณทั้งหมด ไฟล์จึงไม่ถูกส่งออกไปที่ใด

## เลือกคุณภาพอย่างไรดี

| ประเภทภาพ | คุณภาพที่แนะนำ | หมายเหตุ |
|---|---|---|
| ภาพถ่าย ภาพประกอบบทความ | 75 ถึง 85 | ลดขนาดได้มาก ความต่างแทบมองไม่เห็น |
| ภาพขนาดย่อ (thumbnail) | 60 ถึง 70 | เน้นไฟล์เล็กและโหลดเร็ว |
| สกรีนช็อตที่มีตัวอักษร | 90 ถึง 100 | ขอบตัวอักษรยังคมชัด |
| โลโก้ ไอคอน งานกราฟิก | 100 หรือเก็บเป็น PNG/SVG | ถ้าไฟล์ใหญ่ขึ้น ให้ใช้ไฟล์เดิม |

## PNG กับ WebP ต่างกันอย่างไร

| หัวข้อ | PNG | WebP |
|---|---|---|
| การบีบอัด | Lossless เท่านั้น | ทั้ง Lossy และ Lossless |
| ขนาดไฟล์ | ใหญ่กว่า | Google ระบุว่า WebP แบบ lossless เล็กกว่า PNG ประมาณ 26% |
| พื้นหลังโปร่งใส | รองรับ | รองรับ |
| ภาพเคลื่อนไหว | ไม่รองรับ (ยกเว้น APNG) | รองรับ |
| การรองรับเบราว์เซอร์ | ทุกเบราว์เซอร์ | เบราว์เซอร์รุ่นใหม่ส่วนใหญ่ ควรมีไฟล์สำรอง |

## คำถามที่พบบ่อย

### ไฟล์ของฉันถูกอัปโหลดไปที่เซิร์ฟเวอร์หรือไม่

ไม่ การแปลงทำในเบราว์เซอร์ของคุณทั้งหมด ไฟล์ไม่ออกจากอุปกรณ์ คุณตรวจสอบได้ด้วยแท็บ Network ในเครื่องมือนักพัฒนาของเบราว์เซอร์

### ทำไมบางไฟล์ถึงใหญ่ขึ้นหลังแปลง

PNG ขนาดเล็กมากหรือไฟล์ที่ผ่านการบีบอัดมาอย่างดีอยู่แล้ว (เช่น ไอคอนสีเรียบ) อาจใหญ่ขึ้นเมื่อเป็น WebP ลองลดค่าคุณภาพ ย่อขนาด หรือใช้ไฟล์เดิมต่อไป เครื่องมือจะแสดงป้าย "ใหญ่ขึ้น" ให้เห็นชัดเจน

### WebP ใช้ได้กับทุกเบราว์เซอร์หรือไม่

เบราว์เซอร์รุ่นใหม่ เช่น Chrome, Edge, Firefox และ Safari รุ่นล่าสุดแสดงผล WebP ได้ แต่เพื่อความปลอดภัยควรใช้แท็ก `<picture>` ที่เครื่องมือสร้างให้ เพื่อมีไฟล์เดิมเป็นตัวสำรอง

### WebP ช่วยเรื่อง SEO หรือไม่

ช่วยทางอ้อม รูปที่เล็กลงทำให้หน้าเว็บโหลดเร็วขึ้นและช่วยให้ค่า LCP ใน Core Web Vitals ดีขึ้น ซึ่งเป็นส่วนหนึ่งของประสบการณ์ผู้ใช้ที่ Google พิจารณา อย่างไรก็ตาม ควรตั้งชื่อไฟล์และใส่ alt ให้สื่อความหมายด้วย

### ข้อมูล EXIF หรือโปรไฟล์สีจะยังอยู่หรือไม่

ไม่ เพราะภาพผ่านการวาดบน canvas ของเบราว์เซอร์ ข้อมูลเมทาดาทา เช่น EXIF และโปรไฟล์สีเดิมจะไม่ถูกเก็บ ภาพที่ได้ใช้สี sRGB ตามมาตรฐานของเบราว์เซอร์ ซึ่งช่วยให้ไฟล์เล็กลงอีกเล็กน้อย

### เครื่องมือแจ้งว่าสร้างไฟล์ WebP ไม่ได้

เบราว์เซอร์รุ่นเก่าบางตัวไม่รองรับการสร้างไฟล์ WebP ให้อัปเดตเบราว์เซอร์ หรือเปลี่ยนไปใช้ Chrome, Edge หรือ Firefox รุ่นล่าสุด

### รองรับไฟล์ HEIC จาก iPhone หรือไม่

ขึ้นอยู่กับเบราว์เซอร์ Safari เปิดไฟล์ HEIC ได้เอง จึงแปลงได้ ส่วนเบราว์เซอร์อื่นอาจแจ้งว่าอ่านไฟล์ไม่ได้ ในกรณีนั้นให้แปลงเป็น JPG หรือ PNG ก่อน
