(function () {
  var $ = function (id) { return document.getElementById(id); };
  var root = $("tn");
  var S = { theme: "default", h12: false, sec: true, date: true, dark: false };
  var offset = 0;
  var lastSync = 0;
  var wake = null;
  var extra = [];
  var presets = ["Asia/Bangkok", "UTC", "Asia/Tokyo", "Europe/London", "America/New_York", "America/Los_Angeles", "Australia/Sydney", "Asia/Dubai"];
  var localTz = "";
  try { localTz = Intl.DateTimeFormat().resolvedOptions().timeZone || ""; } catch (e) { localTz = ""; }
  function load() {
    try {
      var raw = localStorage.getItem("tn-settings");
      if (raw) { var o = JSON.parse(raw); Object.keys(S).forEach(function (k) { if (o[k] !== undefined) { S[k] = o[k]; } }); return true; }
    } catch (e) {}
    return false;
  }
  function save() { try { localStorage.setItem("tn-settings", JSON.stringify(S)); } catch (e) {} }
  var had = load();
  if (!had) {
    try {
      var hc = Intl.DateTimeFormat(undefined, { hour: "numeric" }).resolvedOptions().hour12;
      S.h12 = hc === true;
      S.dark = window.matchMedia && window.matchMedia("(prefers-color-scheme: dark)").matches;
    } catch (e) {}
  }
  function pad(n) { return String(n).padStart(2, "0"); }
  function now() { return new Date(Date.now() + offset); }
  function pressed(el, v) { el.setAttribute("aria-pressed", v ? "true" : "false"); }
  function applySettings() {
    root.setAttribute("data-theme", S.theme);
    root.classList.toggle("dark", S.dark);
    Array.prototype.forEach.call(document.querySelectorAll("#tnThemes button"), function (b) { pressed(b, b.getAttribute("data-t") === S.theme); });
    pressed($("b24"), !S.h12); pressed($("b12"), S.h12);
    pressed($("bSec"), S.sec); pressed($("bDate"), S.date); pressed($("bDark"), S.dark);
    $("tnS1").style.display = S.sec ? "" : "none";
    $("tnS").style.display = S.sec ? "" : "none";
    $("fC").style.display = S.sec ? "" : "none";
    $("fS").style.display = S.sec ? "" : "none";
    $("aS").style.display = S.sec ? "" : "none";
    $("tnDate").style.display = S.date ? "" : "none";
    $("tnTz").textContent = localTz ? "เขตเวลา: " + localTz : "";
    render(true);
  }
  function setCard(el, v, anim) {
    if (el.textContent !== v) {
      el.textContent = v;
      if (anim) { el.classList.remove("go"); void el.offsetWidth; el.classList.add("go"); }
    }
  }
  function render(first) {
    var d = now();
    var h = d.getHours();
    var m = d.getMinutes();
    var s = d.getSeconds();
    var ap = "";
    var hh = h;
    if (S.h12) { ap = h >= 12 ? "PM" : "AM"; hh = h % 12; if (hh === 0) { hh = 12; } }
    var hs = S.h12 ? String(hh) : pad(hh);
    $("tnH").textContent = hs;
    $("tnM").textContent = pad(m);
    $("tnS").textContent = pad(s);
    $("tnAp").textContent = ap;
    var anim = !first;
    setCard($("fH"), hs, anim); setCard($("fM"), pad(m), anim); setCard($("fS"), pad(s), anim);
    $("aH").setAttribute("transform", "rotate(" + ((h % 12) * 30 + m * 0.5) + " 100 100)");
    $("aM").setAttribute("transform", "rotate(" + (m * 6 + s * 0.1) + " 100 100)");
    $("aS").setAttribute("transform", "rotate(" + (s * 6) + " 100 100)");
    $("tnDate").textContent = d.toLocaleDateString("th-TH", { weekday: "long", day: "numeric", month: "long", year: "numeric" });
    document.title = (S.h12 ? hs + ":" + pad(m) + " " + ap : hs + ":" + pad(m)) + " | เวลาตอนนี้";
  }
  function buildTicks() {
    var g = $("tnTicks");
    var ns = "http://www.w3.org/2000/svg";
    for (var i = 0; i !== 60; i++) {
      var big = i % 5 === 0;
      var ln = document.createElementNS(ns, "line");
      ln.setAttribute("x1", "100"); ln.setAttribute("x2", "100");
      ln.setAttribute("y1", "8"); ln.setAttribute("y2", big ? "20" : "14");
      ln.setAttribute("class", big ? "tick big" : "tick");
      ln.setAttribute("transform", "rotate(" + (i * 6) + " 100 100)");
      g.appendChild(ln);
    }
  }
  function tzOffsetMin(tz, d) {
    try {
      var f = new Intl.DateTimeFormat("en-US", { timeZone: tz, hourCycle: "h23", year: "numeric", month: "numeric", day: "numeric", hour: "numeric", minute: "numeric", second: "numeric" });
      var p = {};
      f.formatToParts(d).forEach(function (x) { p[x.type] = x.value; });
      var asUtc = Date.UTC(+p.year, +p.month - 1, +p.day, +p.hour % 24, +p.minute, +p.second);
      return Math.round((asUtc - Math.floor(d.getTime() / 1000) * 1000) / 60000);
    } catch (e) { return 0; }
  }
  function fmtOff(min) {
    var sign = min >= 0 ? "+" : "-";
    var a = Math.abs(min);
    return sign + pad(Math.floor(a / 60)) + ":" + pad(a % 60);
  }
  function fmtDiff(min) {
    if (min === 0) { return "เท่ากัน"; }
    var a = Math.abs(min);
    var t = Math.floor(a / 60) + " ชม.";
    if (a % 60 !== 0) { t += " " + (a % 60) + " นาที"; }
    return (min > 0 ? "เร็วกว่า " : "ช้ากว่า ") + t;
  }
  function renderRows() {
    if (!$("tnCmp").classList.contains("open")) { return; }
    var d = now();
    var localOff = -d.getTimezoneOffset();
    var body = $("tnRows");
    var list = presets.concat(extra.filter(function (z) { return presets.indexOf(z) === -1; }));
    while (body.firstChild) { body.removeChild(body.firstChild); }
    list.forEach(function (tz) {
      var off = tzOffsetMin(tz, d);
      var tr = document.createElement("tr");
      var tm = d.toLocaleTimeString("th-TH", { timeZone: tz, hour12: S.h12, hour: "2-digit", minute: "2-digit", second: "2-digit" });
      var dt = d.toLocaleDateString("th-TH", { timeZone: tz, day: "numeric", month: "short", weekday: "short" });
      [tz.replace(/_/g, " "), tm, dt, "UTC" + fmtOff(off), fmtDiff(off - localOff)].forEach(function (t) {
        var td = document.createElement("td"); td.textContent = t; tr.appendChild(td);
      });
      body.appendChild(tr);
    });
  }
  function buildSelect() {
    var sel = $("tnSel");
    var zones = [];
    try { if (Intl.supportedValuesOf) { zones = Intl.supportedValuesOf("timeZone"); } } catch (e) { zones = []; }
    if (zones.length === 0) { zones = ["Asia/Jakarta", "Asia/Singapore", "Asia/Seoul", "Asia/Kolkata", "Europe/Paris", "Europe/Moscow", "America/Chicago", "America/Sao_Paulo", "Africa/Cairo", "Pacific/Auckland"]; }
    var first = document.createElement("option");
    first.value = ""; first.textContent = "เลือกเขตเวลา...";
    sel.appendChild(first);
    zones.forEach(function (z) { var o = document.createElement("option"); o.value = z; o.textContent = z.replace(/_/g, " "); sel.appendChild(o); });
    sel.addEventListener("change", function () {
      if (sel.value && extra.indexOf(sel.value) === -1) { extra.push(sel.value); }
      sel.value = "";
      renderRows();
    });
  }
  function sync() {
    var t0 = Date.now();
    var url = location.pathname + (location.search ? location.search + "&" : "?") + "_t=" + t0;
    fetch(url, { method: "HEAD", cache: "no-store" }).then(function (res) {
      var t1 = Date.now();
      var h = res.headers.get("Date");
      if (!h) { throw new Error("no date"); }
      var server = new Date(h).getTime() + 500 + (t1 - t0) / 2;
      offset = server - t1;
      lastSync = t1;
      showSync(true);
    }).catch(function () {
      showSync(false);
    });
  }
  function showSync(ok) {
    var el = $("tnSync");
    while (el.firstChild) { el.removeChild(el.firstChild); }
    if (ok) {
      var b = document.createElement("b"); b.textContent = "ซิงค์แล้ว";
      el.appendChild(b);
      el.appendChild(document.createTextNode(" ปรับเวลากับเซิร์ฟเวอร์ (คลาดเคลื่อน " + (Math.round(offset / 100) / 10) + " วินาที) ทุก 2 นาที"));
    } else {
      el.textContent = "ใช้เวลาจากอุปกรณ์ของคุณ (ซิงค์เซิร์ฟเวอร์ไม่ได้)";
    }
  }
  function requestWake() {
    try {
      if (navigator.wakeLock && navigator.wakeLock.request) {
        navigator.wakeLock.request("screen").then(function (l) { wake = l; }).catch(function () {});
      }
    } catch (e) {}
  }
  function releaseWake() { try { if (wake) { wake.release(); wake = null; } } catch (e) {} }
  function isFs() { return !!(document.fullscreenElement || root.classList.contains("fs")); }
  function toggleFs() {
    if (isFs()) {
      if (document.fullscreenElement && document.exitFullscreen) { document.exitFullscreen(); }
      root.classList.remove("fs");
      releaseWake();
      $("bFs").textContent = "เต็มหน้าจอ";
      return;
    }
    if (root.requestFullscreen) {
      root.requestFullscreen().catch(function () { root.classList.add("fs"); });
    } else {
      root.classList.add("fs");
    }
    requestWake();
    $("bFs").textContent = "ออกจากเต็มหน้าจอ";
  }
  document.addEventListener("fullscreenchange", function () {
    if (!document.fullscreenElement) { root.classList.remove("fs"); releaseWake(); $("bFs").textContent = "เต็มหน้าจอ"; }
  });
  document.addEventListener("visibilitychange", function () {
    if (document.visibilityState === "visible") { if (isFs()) { requestWake(); } sync(); }
  });
  Array.prototype.forEach.call(document.querySelectorAll("#tnThemes button"), function (b) {
    b.addEventListener("click", function () { S.theme = b.getAttribute("data-t"); save(); applySettings(); });
  });
  $("b24").addEventListener("click", function () { S.h12 = false; save(); applySettings(); });
  $("b12").addEventListener("click", function () { S.h12 = true; save(); applySettings(); });
  $("bSec").addEventListener("click", function () { S.sec = !S.sec; save(); applySettings(); });
  $("bDate").addEventListener("click", function () { S.date = !S.date; save(); applySettings(); });
  $("bDark").addEventListener("click", function () { S.dark = !S.dark; save(); applySettings(); });
  $("bCmp").addEventListener("click", function () {
    var c = $("tnCmp");
    c.classList.toggle("open");
    pressed($("bCmp"), c.classList.contains("open"));
    renderRows();
  });
  $("bFs").addEventListener("click", toggleFs);
  buildTicks();
  buildSelect();
  applySettings();
  sync();
  setInterval(function () { render(false); renderRows(); }, 250);
  setInterval(sync, 120000);
})();

## วิธีใช้งานเครื่องมือเวลาตอนนี้

1. **เปิดหน้านี้** นาฬิกาจะแสดงเวลาปัจจุบันตามเขตเวลาของอุปกรณ์ทันที และตรวจรูปแบบ 12 หรือ 24 ชั่วโมงให้อัตโนมัติ
2. **เลือกธีม** กดปุ่มธีม เช่น ดิจิทัล การ์ตูน LED นาฬิกาพลิก นาฬิกาเข็ม หรือสแตนด์บาย เพื่อเปลี่ยนหน้าตานาฬิกา
3. **ปรับรูปแบบเวลา** กด "24 ชม." หรือ "12 ชม. AM/PM" ตามที่ต้องการ
4. **ซ่อนหรือแสดงรายละเอียด** กดปุ่ม "วินาที" หรือ "วันที่" เพื่อเปิดหรือปิดส่วนนั้น
5. **สลับโหมดมืด** กด "โหมดมืด" เพื่อถนอมสายตาเมื่อใช้งานในที่แสงน้อย
6. **เทียบเขตเวลา** กด "เทียบเขตเวลา" เพื่อดูเวลาในหลายเมืองพร้อมกัน แล้วเลือกเขตเวลาเพิ่มจากรายการได้
7. **เปิดเต็มหน้าจอ** กด "เต็มหน้าจอ" เพื่อแสดงนาฬิกาใหญ่ๆ เหมาะกับการสอน การแข่งขัน หรือการพูดบนเวที กด Esc หรือปุ่มเดิมเพื่อออก
8. **ใช้บนมือถือ** หมุนโทรศัพท์เป็นแนวนอนเพื่อให้ตัวเลขใหญ่ขึ้น

ค่าที่ตั้งไว้ เช่น ธีม โหมดมืด และรูปแบบเวลา จะถูกจำไว้ในเบราว์เซอร์ของคุณสำหรับการเปิดครั้งถัดไป

## คำศัพท์ที่ควรรู้

- **เวลาตอนนี้ (เวลาปัจจุบัน)** คือเวลา ณ ขณะที่เปิดดู อ้างอิงจากนาฬิกาของอุปกรณ์ที่ปรับให้ตรงกับเซิร์ฟเวอร์
- **เขตเวลา (Time Zone)** คือพื้นที่บนโลกที่ใช้เวลามาตรฐานเดียวกัน เช่น ประเทศไทยใช้ UTC+7
- **UTC / GMT** คือเวลามาตรฐานสากลที่ใช้เป็นจุดอ้างอิง เขตเวลาอื่นแสดงเป็นส่วนต่างจากค่านี้
- **12 ชั่วโมง (AM/PM)** คือการแบ่งวันเป็นสองช่วง AM คือก่อนเที่ยง PM คือหลังเที่ยง
- **24 ชั่วโมง** คือการนับเวลาต่อเนื่อง 00:00 ถึง 23:59 ไม่ต้องใช้ AM/PM
- **ซิงโครไนซ์ (Sync)** คือการปรับเวลาในเบราว์เซอร์ให้ตรงกับเวลาของเซิร์ฟเวอร์ เครื่องมือนี้ปรับซ้ำทุก 2 นาที
- **NTP (Network Time Protocol)** คือโปรโตคอลที่ใช้ซิงก์เวลาของอุปกรณ์กับเซิร์ฟเวอร์เวลาทั่วโลก
- **ออฟเซ็ต (Offset)** คือส่วนต่างของเวลา เช่น +07:00 หมายถึงเร็วกว่า UTC เจ็ดชั่วโมง
- **เวลาออมแสง (DST)** คือการปรับเวลาเดินหน้าหรือถอยหลัง 1 ชั่วโมงตามฤดูกาล ใช้ในบางประเทศ ประเทศไทยไม่ใช้
- **โหมดเต็มหน้าจอ (Fullscreen)** คือการขยายนาฬิกาให้เต็มจอ และพยายามไม่ให้หน้าจอดับเอง
- **โหมดมืด (Dark Mode)** คือการเปลี่ยนพื้นหลังเป็นสีเข้ม ช่วยลดแสงจ้าในที่มืด
- **ธีม (Theme)** คือรูปแบบหน้าตานาฬิกา เช่น ดิจิทัล การ์ตูน LED
- **นาฬิกาพลิก (Flip Clock)** คือนาฬิกาที่ตัวเลขเป็นแผ่นการ์ดพลิกเมื่อเวลาเปลี่ยน
- **นาฬิกาเข็ม (Analog)** คือนาฬิกาแบบมีหน้าปัดและเข็มชั่วโมง นาที วินาที
- **สแตนด์บาย (Standby)** คือหน้าจอพื้นดำแสดงเวลาตัวใหญ่ เหมาะวางบนโต๊ะข้างเตียง
- **LED** คือรูปแบบตัวเลขเรืองแสงสีแดงบนพื้นดำ เหมือนนาฬิกาปลุกดิจิทัล

## คำถามที่พบบ่อย

**เวลาแม่นยำแค่ไหน?** เครื่องมือเทียบเวลากับเซิร์ฟเวอร์ทุก 2 นาที ความคลาดเคลื่อนมักไม่เกินราว 1 วินาที หากเชื่อมต่อไม่ได้ จะใช้เวลาจากอุปกรณ์แทน

**ทำไมเวลาไม่ตรงกับที่คาดไว้?** ตรวจสอบเขตเวลาที่แสดงใต้นาฬิกา และตั้งค่าเขตเวลาของอุปกรณ์ให้ถูกต้อง
