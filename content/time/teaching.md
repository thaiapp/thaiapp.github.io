  function makeChoices(T) {
    var is = is24(), M = is ? 1440 : 720, st = step();
    var h = Math.floor(T / 60) % 12, m = T % 60, pm = (T >= 720) ? 720 : 0;
    var swH = Math.round(m / 5) % 12, swM = h * 5;
    var pri = [], rest = shuffle([T + 60, T - 60, T + st, T - st, T + 2 * st, T - 2 * st, T + 120, T - 120]);
    var cand, out = [], seen = {}, i, v, f, tries = 0;
    if (is) { pri.push(T + 720); }
    pri.push(swH * 60 + swM + pm);
    cand = pri.concat(rest);
    seen[fmt(T)] = 1;
    for (i = 0; cand.length > i && 3 > out.length; i++) {
      v = mod(cand[i], M);
      f = fmt(v);
      if (!seen[f]) { seen[f] = 1; out.push(v); }
    }
    while (3 > out.length && 200 > tries) {
      tries++;
      v = Math.floor(Math.random() * (M / st)) * st;
      f = fmt(v);
      if (!seen[f]) { seen[f] = 1; out.push(v); }
    }
    out.push(T);
    return shuffle(out);

  /* ---------- dragging ---------- */
  function pointerAngle(e) {
    var r = el.svg.getBoundingClientRect();
    var x = e.clientX - (r.left + r.width / 2);
    var y = e.clientY - (r.top + r.height / 2);
    var a = Math.atan2(x, -y) * 180 / Math.PI;
    if (0 > a) { a += 360; }
    return { a: a, r: Math.sqrt(x * x + y * y) / (r.width / 200) };
  }
  function angDiff(a, b) {
    var d = Math.abs(a - b) % 360;
    return d > 180 ? 360 - d : d;
  }
  function moveTo(a) {
    var d = a - drag.prev;
    if (d > 180) { d -= 360; }
    if (-180 > d) { d += 360; }
    drag.prev = a;
    drag.raw += (drag.which === 'hour') ? d * 2 : d / 6;
    C.t = mod(drag.raw, 720);
    shown = false;
    refresh();
  }
  function onDown(e) {
    if (!canDrag()) { return; }
    e.preventDefault();
    var p = pointerAngle(e);
    var ha = mod(C.t, 720) * 0.5, ma = mod(C.t, 60) * 6;
    var vis = (S.tab === 'play') ? S.hands : 'both';
    var which, dh, dm;
    if (vis === 'hour') { which = 'hour'; }
    else if (vis === 'min') { which = 'min'; }
    else {
      dh = angDiff(p.a, ha);
      dm = angDiff(p.a, ma);
      if (Math.abs(dh - dm) > 8) { which = (dh > dm) ? 'min' : 'hour'; }
      else { which = (p.r > 52) ? 'min' : 'hour'; }
    }
    drag = { which: which, prev: (which === 'hour') ? ha : ma, raw: C.t };
    try { el.svg.setPointerCapture(e.pointerId); } catch (x) { /* ignore */ }
    moveTo(p.a);
  }
  function onMove(e) {
    if (!drag) { return; }
    e.preventDefault();
    moveTo(pointerAngle(e).a);
  }
  function onUp() {
    if (!drag) { return; }
    C.t = snap(drag.raw, step());
    drag = null;
    refresh();
  }

  /* ---------- fullscreen, idle, wake lock ---------- */

    if (f) {
      try {
        pr = f.call(root);

      } catch (x) { fallbackFs(); }
    } else { fallbackFs(); }

  /* ---------- keyboard ---------- */
  function inView() {
    var r = root.getBoundingClientRect();
    var vh = window.innerHeight || document.documentElement.clientHeight;
    return r.bottom > 0 && vh > r.top;
  }
  function onKey(e) {
    var tg = e.target, tn = tg && tg.tagName, k, used = true, q;
    if (e.ctrlKey || e.metaKey || e.altKey) { return; }
    if (tn === 'INPUT' || tn === 'TEXTAREA' || tn === 'SELECT' || (tg && tg.isContentEditable)) { return; }
    if (tn === 'BUTTON' && (e.key === 'Enter' || e.key === ' ')) { return; }
    if (!inView() && !isFs()) { return; }
    if (!el.cf.hidden) {
      if (e.key === 'Escape') { closeCf(); e.preventDefault(); }
      return;
    }
    if (/^Key[A-Z]$/.test(e.code || '')) { k = e.code.slice(3).toLowerCase(); }
    else if (/^Digit[1-4]$/.test(e.code || '')) { k = e.code.slice(5); }
    else { k = e.key; }

    if (k === 'f') { toggleFs(); }
    else if (k === 't') { toggleDark(); }
    else if (k === 'p') { goTab(S.tab === 'play' ? 'prac' : 'play'); }
    else if (k === 'l') { setLevel(cycle(levelsFor(), S.level)); }
    else if (k === 'm') { setNum(cycle(NUMS, S.num)); }
    else if (k === 'ArrowRight' || k === 'ArrowUp') { if (canDrag()) { nudge(1, e.shiftKey); } else { used = false; } }
    else if (k === 'ArrowLeft' || k === 'ArrowDown') { if (canDrag()) { nudge(-1, e.shiftKey); } else { used = false; } }
    else if (S.tab === 'play') {
      if (k === 'n') { setNow(); }
      else if (k === 'r') { setRand(); }
      else if (k === 'd') { toggleHide(); }
      else if (k === 'h') { setHands(cycle(HANDS, S.hands)); }
      else { used = false; }
    } else {
      if (k === 'n' || k === 'Enter' || k === ' ') { if (P.done) { restart(); } else { nextQ(); } }
      else if (k === 'c') { check(); }
      else if (k === 'q') { setKind(cycle(KINDS, S.kind)); }
      else if (k.length === 1 && '1234'.indexOf(k) !== -1) {
        q = P.done ? null : P.qs[P.i];
        if (q && q.type === 'read' && q.choices[Number(k) - 1] !== undefined) { pick(q.choices[Number(k) - 1]); }
      } else { used = false; }
    }
    if (used) { e.preventDefault(); }
  }

  /* ---------- init ---------- */
  load();
  build();
  (function () {
    var d = new Date();
    C.t = snap((d.getHours() % 12) * 60 + d.getMinutes(), step());

  el.svg.addEventListener('pointerdown', onDown);

  root.addEventListener('pointerdown', onFirstTouch);
  root.addEventListener('pointermove', poke);

})();

## วิธีใช้งานเครื่องมือสอนดูนาฬิกา

เครื่องมือนี้ช่วยให้เด็กและผู้เริ่มต้นเรียนรู้การอ่านเวลาจากนาฬิกาแบบเข็มได้ด้วยตัวเอง เริ่มจากการลากเข็มเล่นเพื่อดูว่าเวลาเปลี่ยนอย่างไร แล้วค่อยฝึกทำโจทย์เป็นรอบละ 10 ข้อ ไม่ต้องติดตั้งและไม่ต้องสมัครสมาชิก

### โหมดลองเล่น

1. **ลากเข็ม** แตะหรือลากบนหน้าปัดได้ทุกจุด เข็มที่อยู่ใกล้นิ้วมากกว่าจะเลื่อนตามไป เมื่อปล่อยนิ้ว เข็มจะดีดไปหยุดที่ขีดใกล้ที่สุดตามระดับที่เลือก
2. **ดูเวลาดิจิทัล** ตัวเลขด้านขวาเปลี่ยนตามเข็มทันที ชั่วโมงเป็นสีน้ำเงินตามเข็มสั้น และนาทีเป็นสีส้มตามเข็มยาว
3. **เลือกระดับ** เริ่มจากโมงตรง แล้วไล่ไปที่ครึ่งชั่วโมง 15 นาที 5 นาที จนถึงทุกนาที
4. **เลือกตัวเลขบนหน้าปัด** แสดงเฉพาะชั่วโมง แสดงชั่วโมงพร้อมนาที (00 ถึง 55) หรือซ่อนตัวเลขทั้งหมดเพื่อท้าทายผู้เรียน
5. **เลือกเข็ม** แสดงทั้งสองเข็ม หรือเหลือเข็มสั้นหรือเข็มยาวเพียงเข็มเดียว เพื่อฝึกอ่านทีละเข็ม
6. **ปุ่มช่วย** ปุ่ม "ตอนนี้" ตั้งเป็นเวลาปัจจุบัน ปุ่ม "สุ่ม" ตั้งเวลาแบบสุ่มในระดับที่เลือก และปุ่มลูกศร ◀ ▶ เลื่อนทีละหนึ่งขีด
7. **ซ่อนดิจิทัล** เปิดปุ่ม "ซ่อนดิจิทัล" แล้วให้เด็กทายเวลาจากเข็มก่อน จากนั้นแตะกล่องเครื่องหมายคำถามเพื่อตรวจคำตอบ

### โหมดฝึกหัด

1. กดแท็บ **ฝึกหัด** เพื่อเริ่มรอบใหม่ มีทั้งหมด 10 ข้อ แถวจุดด้านบนจะเปลี่ยนเป็นสีเขียวเมื่อตอบถูก และสีแดงเมื่อตอบผิด
2. เลือกประเภทโจทย์ **ดูนาฬิกา** ให้เลือกคำตอบจากตัวเลือก 4 ข้อ **ตั้งนาฬิกา** ให้ลากเข็มให้ตรงกับเวลาที่โจทย์กำหนดแล้วกด "ตรวจ" หรือเลือก **ผสม** เพื่อสลับโจทย์ทั้งสองแบบ
3. หากตั้งนาฬิกาผิด เข็มของผู้เรียนจะคงอยู่ และเข็มที่ถูกต้องจะปรากฏเป็นเส้นประซ้อนทับให้เห็นความต่าง
4. กด "ถัดไป" เพื่อไปข้อต่อไป เมื่อครบ 10 ข้อ ระบบจะสรุปคะแนนเต็ม 10 พร้อมดาวสูงสุด 3 ดวง และมีปุ่ม "เล่นอีกครั้ง"
5. ถ้าเปลี่ยนแท็บ ระดับ หรือประเภทโจทย์ระหว่างทำ ระบบจะถามยืนยันก่อนจบรอบ

### ใช้ในห้องเรียน

- ปุ่ม **เต็มหน้าจอ** ขยายนาฬิกาให้เหมาะกับไวต์บอร์ด โปรเจกเตอร์ หรือแท็บเล็ต และแถบปุ่มจะซ่อนเองเมื่อไม่ได้ใช้งาน แตะอีกครั้งเพื่อเรียกกลับมา
- ปุ่ม **โหมดมืด** เปลี่ยนหน้าจอเป็นสีเข้ม สบายตาในห้องที่แสงน้อย
- ค่าที่เลือกไว้ เช่น ระดับ ตัวเลข และเข็ม จะถูกจำไว้ในอุปกรณ์เครื่องเดิม กด "ลบข้อมูลทั้งหมด" ใต้เครื่องมือเพื่อล้างค่าเหล่านี้

## ความหมายของคำศัพท์

- **นาฬิกาเข็ม (อนาล็อก)** นาฬิกาที่บอกเวลาด้วยเข็มที่หมุนบนหน้าปัด
- **เวลาดิจิทัล** เวลาที่แสดงเป็นตัวเลข เช่น 3:25 โดยตัวเลขหน้าคือชั่วโมงและตัวเลขหลังคือนาที
- **หน้าปัด** แผ่นกลมที่มีตัวเลข 1 ถึง 12 และขีดแบ่งเวลาเรียงรอบวง
- **เข็มสั้น (เข็มชั่วโมง)** เข็มที่บอกชั่วโมง เคลื่อนช้า ใช้เวลา 12 ชั่วโมงจึงครบหนึ่งรอบ
- **เข็มยาว (เข็มนาที)** เข็มที่บอกนาที ใช้เวลา 60 นาทีจึงครบหนึ่งรอบ
- **ขีด** เครื่องหมายเล็ก ๆ รอบหน้าปัด มีทั้งหมด 60 ขีด แต่ละขีดแทน 1 นาที
- **ตัวเลขนาที (00 ถึง 55)** ตัวเลขสีส้มรอบขอบหน้าปัด ช่วยให้รู้ว่าตัวเลขแต่ละตำแหน่งแทนกี่นาที เช่น เลข 3 คือ 15 นาที
- **โมงตรง** เวลาที่เข็มยาวชี้เลข 12 หรือ 0 นาที เช่น 3:00
- **ครึ่งชั่วโมง** เวลาที่เข็มยาวชี้เลข 6 หรือ 30 นาที เช่น 3:30 หรือ "3 โมงครึ่ง"
- **15 นาที** ระดับที่เข็มยาวหยุดที่เลข 3, 6, 9 และ 12 หรือที่ 15, 30, 45 และ 0 นาที ซึ่งแต่ละช่วงคือหนึ่งในสี่ของชั่วโมง
- **5 นาที** ระดับที่เข็มยาวหยุดที่เลขแต่ละตัว เพราะเลข 1 ถึง 12 แทน 5, 10, 15 ไปจนถึง 60 นาที
- **ทุกนาที** ระดับที่อ่านเวลาละเอียดถึงทุกขีด ต้องนับขีดระหว่างตัวเลข
- **โมงเศษ** เวลาที่เลยชั่วโมงไปเล็กน้อย เช่น 3:10 เข็มสั้นจึงอยู่ถัดจากเลข 3 เล็กน้อย
- **นาฬิกา 24 ชั่วโมง** ระบบที่นับชั่วโมงตั้งแต่ 00 ถึง 23 เช่น 15:00 คือบ่าย 3 โมง ใช้ดวงอาทิตย์หรือดวงจันทร์ช่วยแยกช่วงกลางวันกับกลางคืน
- **ช่วงเวลาของวัน** เช้า เที่ยง–บ่าย เย็น และกลางคืน ช่วยแยกเวลาที่เข็มเหมือนกัน เช่น 3:00 ตอนตีสามกับบ่ายสาม
- **ตอบถูก/เกือบถูก** ผลตรวจในโหมดฝึกหัด ตัวเลือกที่ผิดมักเป็นข้อผิดพลาดที่พบบ่อย เช่น อ่านเข็มสลับกัน หรืออ่านเป็นชั่วโมงถัดไป
