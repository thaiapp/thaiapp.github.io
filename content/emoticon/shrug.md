## วิธีใช้งาน

1. เปิดหน้านี้บนมือถือหรือคอมพิวเตอร์ได้เลย ไม่ต้องติดตั้งแอปหรือสมัครสมาชิก
2. ดูอีโมติคอนยักไหล่ในกล่อง "คัดลอก / วาง" แล้วกดปุ่ม "คัดลอก" ระบบจะคัดลอกไปที่คลิปบอร์ดทันที
3. อยากได้แบบอื่น ให้เลือกจากหัวข้อ "อีโมติคอนที่คล้ายกัน" เพียงคลิกที่แบบที่ชอบ ระบบจะคัดลอกให้อัตโนมัติและแสดงในกล่องด้านบน
4. เปิดแอปที่ต้องการใช้ เช่น LINE, Facebook, Instagram, X, WhatsApp หรือ Discord แล้วแตะค้างที่ช่องพิมพ์ข้อความ เลือก "วาง" (บนคอมพิวเตอร์กด Ctrl+V หรือ ⌘+V)
5. กดส่งหรือโพสต์ได้เลย

## คำอธิบายอีโมติคอนแต่ละแบบ

<dd>อีโมติคอนยักไหล่แบบดั้งเดิม ใช้เมื่อไม่รู้ ไม่แน่ใจ หรือช่างมัน</dd>

<dd>แบบเดียวกับอันแรกแต่ใช้ตัว ヅ ให้หน้าดูยิ้มกวนๆ มากขึ้นเล็กน้อย</dd>

## คำศัพท์ที่ควรรู้

- **ยักไหล่ (Shrug)** ท่าทางยกไหล่ขึ้น สื่อว่าไม่รู้ ไม่แน่ใจ หรือไม่ใส่ใจ
- **อีโมติคอน (Emoticon)** ใบหน้าหรือท่าทางที่สร้างจากตัวอักษรและสัญลักษณ์บนคีย์บอร์ด
- **คาโอโมจิ (Kaomoji)** อีโมติคอนสไตล์ญี่ปุ่นที่มองตรงๆ ได้โดยไม่ต้องเอียงคอ ยักอีโมติคอนก็จัดอยู่ในกลุ่มนี้
- **อีโมจิ (Emoji)** รูปภาพเล็กๆ ที่เป็นมาตรฐานยูนิโค้ด เช่น 🤷 ต่างจากอีโมติคอนที่สร้างจากตัวอักษร
- **ยูนิโค้ด (Unicode)** มาตรฐานรหัสอักขระสากล ทำให้สัญลักษณ์พิเศษแสดงผลได้ในหลายอุปกรณ์
- **คาตาคานะ (Katakana)** ตัวอักษรญี่ปุ่นชุดหนึ่ง ตัว ツ และ ヅ ใช้แทนใบหน้ายิ้มในยักอีโมติคอน
- **คลิปบอร์ด (Clipboard)** พื้นที่พักข้อความชั่วคราวของอุปกรณ์ เมื่อคัดลอกแล้วจึงนำไปวางที่อื่นได้
- **คัดลอกและวาง (Copy & Paste)** การนำข้อความหรือสัญลักษณ์จากที่หนึ่งไปใส่อีกที่หนึ่ง

<script>
(function () {
  var root = document.getElementById('shrug-tool');
  var output = document.getElementById('shrug-output');
  var copyBtn = document.getElementById('shrug-copy');
  var status = document.getElementById('shrug-status');
  var timer = null;
  if (!root || !output || !copyBtn || !status) { return; }
  function notify(message) {
    status.textContent = message;
    status.className = 'shrug-status shrug-show';
    clearTimeout(timer);
    timer = setTimeout(function () { status.className = 'shrug-status'; }, 1800);
  }
  function fallbackCopy(text) {
    var area = document.createElement('textarea');
    var ok = false;
    area.value = text;
    area.setAttribute('readonly', '');
    area.style.position = 'fixed';
    area.style.opacity = '0';
    document.body.appendChild(area);
    area.select();
    try { ok = document.execCommand('copy'); } catch (err) { ok = false; }
    document.body.removeChild(area);
    return ok;
  }
  function done(ok) {
    notify(ok ? 'คัดลอกแล้ว ✓' : 'คัดลอกไม่สำเร็จ กรุณาเลือกข้อความแล้วคัดลอกเอง');
  }
  function copyText(text) {
    if (navigator.clipboard && window.isSecureContext) {
      navigator.clipboard.writeText(text).then(
        function () { done(true); },
        function () { done(fallbackCopy(text)); }
      );
    } else {
      done(fallbackCopy(text));
    }
  }
  copyBtn.addEventListener('click', function () {
    copyText(output.textContent);
  });
  root.addEventListener('click', function (event) {
    var node = event.target;
    while (node && node !== root) {
      if (node.className && String(node.className).indexOf('shrug-item') !== -1) {
        output.textContent = node.textContent;
        copyText(node.textContent);
        return;
      }
      node = node.parentNode;
    }
  });
})();
</script>
